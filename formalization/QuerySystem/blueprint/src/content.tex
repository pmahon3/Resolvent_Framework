\chapter{Query Systems and Observational Foundations}

This formalization develops query systems: preordered families of measurable spaces with compatible refinement maps.

\section{Core Definitions}

\begin{definition}
\label{def:query}
\lean{Query}
\leanok
A \emph{query} is a measurable space representing the possible outcomes of asking a single question or making a single observation.
\end{definition}

\begin{definition}
\label{def:refine}
\lean{Refine}
\leanok
A \emph{refinement} between queries is a measurable function showing how outcomes of one query determine outcomes of another.
\end{definition}

\begin{definition}
\label{def:query-system}
\lean{QuerySystem}
\leanok
\uses{def:query, def:refine}
A \emph{query system} is a preordered family of queries with coherent refinement maps. It consists of:
\begin{itemize}
\item An index type $\iota$
\item A family of queries $q : \iota \to \text{Query}$
\item A preorder relation $\le$ on indices
\item Refinement maps $\pi : \forall \{i\ j\}, i \le j \to \text{Refine}(q_i, q_j)$
\end{itemize}
subject to functoriality conditions.
\end{definition}

\begin{definition}
\label{def:omega}
\lean{QuerySystem.Omega}
\leanok
The \emph{projective limit} $\Omega$ is the space of coherent families of query outcomes.
An element $\omega : \Omega$ assigns to each query $i$ an outcome $\omega(i)$, subject to the coherence condition:
whenever $i \le j$, we have $\omega(j) = \pi_{ij}(\omega(i))$.
\end{definition}

\begin{definition}
\label{def:cylinder}
\lean{QuerySystem.Cyl}
\leanok
A \emph{cylinder set} $\text{Cyl}(i, A)$ is the set of realizations where query $i$ yields an outcome in $A$.
These generate the observable $\sigma$-algebra.
\end{definition}

\begin{definition}
\label{def:compatible-marginals}
\lean{QuerySystem.CompatibleMarginals}
\leanok
\uses{def:query-system, def:refine}
A family $\nu$ of measures on query outcomes is \emph{compatible} if the measures commute with refinement maps:
whenever $i \le j$, the pushforward of $\nu_i$ along $\pi_{ij}$ equals $\nu_j$.
\end{definition}

\begin{definition}[Upper-directed]
\label{def:upper-directed}
\lean{QuerySystem.UpperDirected}
\leanok
A query system is \emph{upper-directed} if any two queries have a common refinement: $\forall i\,j, \exists k, i \le k \land j \le k$. This, not lower-directedness, is what the extension arguments actually use --- a query system pieces together finer information, and directedness has to run that direction.
\end{definition}

\begin{definition}[Sequentially upper-directed]
\label{def:seq-upper-directed}
\lean{QuerySystem.SequentiallyUpperDirected}
\leanok
\uses{def:upper-directed}
Every \emph{sequence} of queries has a common refinement:
$\forall u : \mathbb{N} \to \iota, \exists k, \forall n, u_n \le k$. Strictly
stronger than upper-directedness (which bounds pairs), and the analytic
hypothesis of the extension theorem: it lets a single level dominate a
countable cylinder cover. It says the completed query $\bigvee_n Q_n$ is
itself a query.
\end{definition}

\begin{definition}[Surjective evaluations]
\label{def:eval-surjective}
\lean{QuerySystem.EvalSurjective}
\leanok
\uses{def:omega}
Every evaluation $\mathrm{eval}_i : \Omega \to \mathrm{Outcome}_i$ is
surjective: each local outcome extends to a coherent global realization. A
realizability condition on the projective limit, needed to read off events at
a level from their cylinders ($\mathrm{Cyl}_k E_1 = \mathrm{Cyl}_k E_2
\Rightarrow E_1 = E_2$), which the well-definedness and additivity of the
cylinder content require.
\end{definition}

\begin{lemma}
\label{lem:pi-system}
\lean{QuerySystem.isPiSystem_CylGen}
\leanok
\uses{def:cylinder, def:upper-directed}
Under upper-directedness, the cylinder family forms a $\pi$-system.
\end{lemma}

\begin{proof}
\leanok
Upper-directedness gives a common refinement of any two cylinder-generating queries; intersect the pulled-back events there.
\end{proof}

\begin{theorem}[Observational Determination]
\label{thm:observational-determination}
\lean{QuerySystem.observational_determination}
\leanok
\uses{lem:pi-system, def:omega}
Two (finite) measures on the projective limit $\Omega$ are equal if and only if they have equal marginals on all queries. (``Equal marginals'' means equal pushforwards along every $\mathrm{eval}_i$; no compatibility hypothesis is consumed.)
\end{theorem}

\begin{proof}
\leanok
Apply the $\pi$-$\lambda$ theorem to the cylinder $\sigma$-algebra. The cylinders form a $\pi$-system by \ref{lem:pi-system}, and the hypothesis gives agreement on all cylinders.
\end{proof}

\label{lem:premu-well-defined}
\lean{QuerySystem.preμ_wellDefined}
\leanok

\chapter{The observational extension theorem, and the obstruction}

From a compatible family of finitely additive charges to a single probability
measure on $\Omega$. The extension theorem holds under \emph{sequential}
upper-directedness (\ref{thm:observational-extension-seq},
\ref{thm:stone-observational-extension}); under plain upper-directedness it is
false, and \S\ref{sec:sequentiality} exhibits the counterexample. The
mechanism of that failure --- an escaping tower --- is machine-checked
(\ref{thm:no-extension-tower}). On the Stone space there is a genuinely
finitely additive theorem, needing no collective exhaustion and no
directedness beyond upper (\ref{thm:stone-measure-exists}).

\section{Charges, collective exhaustion, and the Stone space}

\begin{definition}[Stone space of a query system]
\label{def:stone-space}
\lean{stoneSpace}
\leanok
$\mathrm{St}(S)$ is the space of ultrafilters on $\Omega$, i.e. the
Stone--\v{C}ech compactification of $\Omega$ regarded as a discrete set. It is
compact, Hausdorff and totally disconnected.
\end{definition}

\begin{definition}[Cylinder clopens]
\label{def:stone-clopens}
\lean{stoneClopens}
\leanok
\uses{def:stone-space}
The sets $\{u : E \in u\}$ for $E$ a cylinder event. These are clopen, form a
set semiring, and generate the $\sigma$-algebra used on $\mathrm{St}(S)$.
\end{definition}

\begin{definition}[Normalized compatible contents]
\label{def:ncc}
\lean{QuerySystem.NormalizedCompatibleContents}
\leanok
A family $\nu_i$ of finitely additive contents on the outcome spaces,
compatible under refinement, each of total mass $1$. Finitely additive only:
this is the whole point, since assuming $\sigma$-additivity would beg the
question.
\end{definition}

\begin{definition}[Collective exhaustion]
\label{def:collectively-exhaustive}
\lean{QuerySystem.CollectivelyExhaustive}
\leanok
For every index $i$ and every decreasing sequence of measurable events
shrinking to $\emptyset$, some finer level $j \geq i$ witnesses convergence:
the charge of the pulled-back sequence at $j$ tends to $0$. This is the
valuation-layer condition; index-layer conditions provably do not suffice
(\texttt{counterexampleQS\_no\_finitary\_condition}).
\end{definition}

\begin{lemma}[Collective exhaustion is per-level $\sigma$-additivity]
\label{lem:ce-levelwise}
\lean{QuerySystem.ce_iff_levelwise_continuity}
\leanok
\uses{def:collectively-exhaustive, def:ncc}
$\nu$ is collectively exhaustive if and only if every $\nu_i$ is continuous at
$\emptyset$, i.e. $\sigma$-additive.
\end{lemma}

\begin{proof}
\leanok
The definition reads as though the finer level $j$ carries content --- ``some
$j \geq i$ witnesses the convergence''. It does not. Compatibility gives
$\nu_i(A) = \nu_j(\pi^{-1}A)$ outright, so the quantity tested at $j$ is
\emph{equal} to the one at $i$ and the existential is cosmetic. CE is
therefore strictly a per-level condition; it cannot see the passage between
levels, and \S\ref{sec:sequentiality} shows the failure of extension lives
exactly there.
\end{proof}

\begin{theorem}[SP1]
\label{thm:sp1-iff}
\lean{QuerySystem.sp1_iff}
\leanok
\uses{def:collectively-exhaustive, def:ncc}
Collective exhaustion holds if and only if every level's content extends to a
genuine measure on that level's outcome space.
\end{theorem}

\begin{proof}
\leanok
This is the form the extension theorem consumes: collective exhaustion is
available as ``each $\nu_i$ is countably additive''. It characterises
\emph{per-level} extensibility only; read as global extensibility the
characterisation is false (\S\ref{sec:sequentiality}).
\end{proof}

\section{The extension theorem}

\begin{theorem}[Observational extension, sequential form]
\label{thm:observational-extension-seq}
\lean{QuerySystem.observational_extension}
\leanok
\uses{lem:cylgen-sigma-subadd, lem:premu-well-defined, thm:observational-determination, def:compatible-marginals, def:seq-upper-directed, def:eval-surjective}
If $S$ is \emph{sequentially} upper-directed with surjective evaluations and
$\nu_i$ is a compatible family of probability \emph{measures}, then there is a
unique probability measure $P$ on $\Omega$ with
$\mathrm{map}(\mathrm{eval}_i)\,P = \nu_i$ for every $i$.
\end{theorem}

\begin{proof}
\leanok
\uses{lem:cylgen-sigma-subadd, lem:premu-well-defined, thm:observational-determination}
Carath\'eodory on the cylinder semiring directly --- no Stone space. Build the
cylinder content --- its value on a cylinder is \texttt{pre}$\mu$ at a
singleton presentation, so \ref{lem:premu-well-defined} is what makes the
content well defined at all --- prove $\sigma$-subadditivity
(\ref{lem:cylgen-sigma-subadd}), extend, recover marginals; uniqueness is
\ref{thm:observational-determination}.
\end{proof}

\begin{lemma}[$\sigma$-subadditivity from a common upper bound]
\label{lem:cylgen-sigma-subadd}
\lean{QuerySystem.cylGen_addContent_isSigmaSubadditive}
\leanok
\uses{def:cylinder, def:compatible-marginals, def:seq-upper-directed, def:eval-surjective}
Under sequential upper-directedness (with surjective evaluations and
compatible marginal measures) the cylinder content is $\sigma$-subadditive.
\end{lemma}

\begin{proof}
\leanok
Sequentiality is used exactly once, and it is the crux of the whole chapter:
given a countable cover, it produces a \emph{single} index $m$ above every
index appearing in it, after which $\sigma$-subadditivity of the one measure
$\nu_m$ finishes the argument. Every other step is bookkeeping. Ordinary
upper-directedness bounds \emph{pairs} of indices, not countable families, so
no single $\nu_m$ dominates a countable cover --- and that gap is not a gap in
ingenuity: with ordinary upper-directedness plus collective exhaustion in
place of sequentiality, this lemma is \textbf{false}, by the same
Andersen--Jessen system that refutes the extension theorem
(\S\ref{sec:sequentiality}). Were it true, the Carath\'eodory pipeline
above --- which past this lemma needs only pairwise upper-directedness ---
would manufacture the extension that system provably lacks.
\end{proof}

\begin{theorem}[Observational extension from charges]
\label{thm:stone-observational-extension}
\lean{stone_observational_extension}
\leanok
\uses{thm:sp1-iff, thm:observational-extension-seq, def:ncc, def:collectively-exhaustive}
For a \emph{sequentially} upper-directed query system with surjective
evaluations and a collectively exhaustive normalized compatible family
$\nu$ of finitely additive charges, there is a unique
probability measure $\mu$ on $\Omega$ with
$\mu(\mathrm{Cyl}_i A) = \nu_i(A)$ for every query $i$ and measurable $A$.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:sp1-iff, thm:observational-extension-seq}
Proved, and not by the Stone route. \ref{thm:sp1-iff} converts collective
exhaustion into a genuine measure at each level; compatibility transfers
because each $\mu_i$ agrees with $\nu_i$ on every measurable set; and
\ref{thm:observational-extension-seq} finishes on $\Omega$ directly, its
uniqueness clause included. No Stone space and no charge theory are needed.
\end{proof}

\section{Sequentiality is not removable}
\label{sec:sequentiality}

Drop sequentiality to plain upper-directedness and the hypotheses of
\ref{thm:stone-observational-extension} read, via \ref{lem:ce-levelwise}:
compatible marginals, each $\sigma$-additive, plus upper-directedness and
surjective evaluations. That is Kolmogorov extension with no regularity or
compactness hypothesis of any kind --- the outcome spaces carry only a
$\sigma$-algebra --- and it is false. The Andersen--Jessen construction
(Sparre Andersen and Jessen 1948; exposition in K.~C.~Border,
\emph{Kolmogorov Extension Problem}, Prop.~17, following Halmos
\emph{Measure Theory} pp.~68--70) instantiates every hypothesis and refutes
the conclusion.

\emph{The construction.} Take $X_k \subseteq [0,1]$ \emph{thick} --- full
outer measure --- and decreasing with $X_k \downarrow \emptyset$. Thickness
makes $\Sigma_k = \{E \cap X_k : E \text{ Lebesgue measurable}\}$ with
$\lambda_k(E \cap X_k) = \lambda(E)$ a well defined \emph{$\sigma$-additive}
probability space. Index by $\mathbb{N}$ under $\leq$; put
$\mathrm{Outcome}_n = X_0 \times \cdots \times X_n$ with the projections as
bonding maps, so $\Omega = \prod_k X_k$; define $P_n$ by pushing $\lambda_n$
forward along the diagonal $x \mapsto (x,\dots,x)$. These are Kolmogorov
consistent.

\emph{The refutation.} The diagonal cylinders
$D_n = \{(x,\dots,x) \times \prod_{k>n} X_k : x \in X_n\}$ have
$P(D_n) = \lambda_n(X_n) = 1$ for every $n$, while $\bigcap_n D_n = \emptyset$
because $X_k \downarrow \emptyset$. No measure gives every term of a
decreasing sequence mass $1$ and the limit mass $0$.

\emph{Every hypothesis holds.} $\mathbb{N}$ is a chain, hence upper-directed,
and \emph{not} sequentially upper-directed, since $(0,1,2,\dots)$ has no upper
bound --- exactly the gap. Every $\mathrm{eval}_n$ is a projection onto an
initial segment, hence surjective (each $X_k$ is thick, so nonempty). Each
$P_n$ is a genuine measure, so collective exhaustion holds by
\ref{lem:ce-levelwise}. The variance matches: \texttt{QuerySystem} has
$\pi : \mathrm{Outcome}_j \to \mathrm{Outcome}_i$ for $i \leq j$, the
inverse-system direction.

So the failure lives entirely in the passage between levels, where a per-level
condition cannot see it: collective exhaustion characterises $\sigma$-additive
extensibility \emph{per level} (\ref{thm:sp1-iff}) and not globally. The
alternative to assuming sequentiality is a compact-class or perfect-measure
condition on the marginals (Marczewski), the classical hypothesis.

The hypothesis-match above is verified against the source hypothesis by
hypothesis. Its load-bearing step, the surjectivity clause, is no longer a hand
check: \texttt{staging/H1\_EvalSurjective.lean} proves \texttt{EvalSurjective}
for this shape against \texttt{Omega} and \texttt{eval} themselves, receipts
\texttt{[propext, Classical.choice, Quot.sound]}. Note $\Omega$ is the subtype
of \emph{coherent} families, not the bare product --- surjectivity onto a
projective limit can fail with every level nonempty --- but here the bonding
maps are initial-segment restrictions, so coherence is automatically
satisfiable and nonemptiness of each $X_k$ suffices. The thick tower is
machine-checked too (\ref{thm:aj-tower}); \ref{rmk:aj-remaining} states what
remains to make the whole refutation kernel-checked.

What sequentiality costs is that the completed query $\bigvee_n Q_n$ must
itself be a query. That is natural when a query is a $\sigma$-algebra of
resolvable events closed under countable joins, and it is not a fresh
idealisation --- making each query a $\sigma$-algebra already assumes
completed countable operations, and $\bigcup_n E_n$ is as unphysical as
$\bigvee_n Q_n$. What it excludes is precisely the finite-dimensional marginal
family above. Weakening instead to finitely additive contents keeps a theorem
without it (\ref{thm:stone-measure-exists}); assuming neither leaves nothing,
and the index layer alone does not suffice either
(\texttt{counterexampleQS\_no\_finitary\_condition}), so a valuation-layer and
an index-layer condition are both required, with a counterexample in each
direction.

\section{The finitely additive theorem on the Stone space}

\begin{theorem}[The Stone measure exists and extends the content]
\label{thm:stone-measure-exists}
\lean{stone_measure_exists}
\leanok
\uses{def:stone-clopens, def:ncc, def:upper-directed, def:eval-surjective}
There is a probability measure $\hat P$ on $\mathrm{St}(S)$ with
$\hat P(E) = m(E)$ for every cylinder clopen $E$, where $m$ is the additive
content transferred from $\nu$ along the Stone embedding. No collective
exhaustion and no directedness beyond upper: this is the theorem available to
an observer who never completes a countable family of queries, and on it free
ultrafilters carrying mass are not a pathology but the trace of exactly that.
\end{theorem}

\begin{proof}
\leanok
\uses{def:stone-clopens}
Transfer $\nu$ to an additive content on the cylinder clopens; prove
$\sigma$-subadditivity using compactness of the Stone space (every countable
cover of a clopen has a finite subcover); apply Carath\'eodory. The agreement
clause is the content of the statement: without it the conclusion never
mentions $\nu$ and is satisfied by a Dirac measure at any ultrafilter.
\end{proof}

\section{The obstruction, formalized}

\begin{definition}[Escaping tower]
\label{def:escaping-tower}
\lean{QuerySystem.EscapingTower}
\leanok
\uses{def:cylinder, def:ncc}
A decreasing sequence of cylinder events, each of full charge, with empty
intersection. Every finite stage carries all the mass; the limit carries none.
In Andersen--Jessen these are the diagonal cylinders over a decreasing sequence
of thick sets.
\end{definition}

\begin{theorem}[No extension over an escaping tower]
\label{thm:no-extension-tower}
\lean{QuerySystem.not_exists_extension_of_escapingTower}
\leanok
\uses{def:escaping-tower, lem:cont-above}
A query system carrying an escaping tower admits no extension of its charges to
a probability measure on $\Omega$.
\end{theorem}

\begin{proof}
\leanok
\uses{lem:cont-above}
The tower's cylinders are an antitone measurable sequence with empty
intersection and every stage of mass $1$, which is exactly what
\ref{lem:cont-above} forbids. Continuity from above is not re-derived here: it
was proved once, in the diagonal layer, and this is its consumer.

Note the hypotheses: there are none on $S$. No directedness, no surjectivity,
no collective exhaustion. The tower alone is fatal --- which is why
strengthening the index layer was the only available repair for
\ref{thm:stone-observational-extension}. One cannot patch the conclusion or the
valuation layer around a tower; one can only assume an index layer rich enough
that no tower exists.
\end{proof}

\begin{theorem}[The obstruction is not vacuous]
\label{thm:tower-inhabited}
\lean{QuerySystem.counterexampleTower}
\leanok
\uses{def:escaping-tower}
The finite--cofinite query system carries an escaping tower: with
$E_n = \mathbb{Q} \setminus \{q_0,\dots,q_n\}$, each $E_n$ is cofinite hence in
the hyperfilter hence of content $1$, while $\bigcap_n E_n = \emptyset$.
\end{theorem}

\begin{proof}
\leanok
Direct. The point is not the construction --- the same sequence already
appears in \texttt{counterexampleNCC\_not\_collectivelyExhaustive} --- but that
\ref{def:escaping-tower} is inhabited, without which
\ref{thm:no-extension-tower} would be empty content.
\end{proof}

\begin{corollary}[The finite--cofinite system admits no extension]
\label{cor:cex-no-extension}
\lean{QuerySystem.counterexample_no_extension}
\leanok
\uses{thm:tower-inhabited, thm:no-extension-tower}
Recovered from the abstract obstruction rather than re-argued, which is the
point: the tower is the reusable content of that counterexample.

Note what it does \emph{not} show. This system fails collective exhaustion
(\texttt{counterexampleNCC\_not\_collectivelyExhaustive}), so it does not
refute the plain-upper-directedness extension statement
(\S\ref{sec:sequentiality}). Refuting that needs a tower in a system
where every level \emph{is} $\sigma$-additive, which is what forces the thick
sets of \ref{thm:aj-tower} and why Andersen--Jessen is harder than this.
\end{corollary}

\begin{proof}
\leanok
Instantiate \ref{thm:no-extension-tower} at the finite--cofinite escaping tower of \ref{thm:tower-inhabited}.
\end{proof}

\chapter{Andersen--Jessen: the thick tower and the diagonal layer}

The counterexample that refutes observational extension under plain
upper-directedness needs one object: a decreasing family of subsets of $[0,1]$,
each of full outer measure, whose intersection is empty. Every level carries all
the measure, the limit carries none. That is the non-regularity phenomenon
itself, and it has no analogue over Polish spaces.

The tower is built from \emph{finiteness} rather than the parity split of the
source (Border, following Halmos pp.~68--70). Parity caps out: a set with all
even differences lies in a coset of $2\mathbb{Z}+\alpha\mathbb{Z}$, of which
there are two, so it can never separate more than two pieces and gives
thickness only at $k = 0$. A finite set has no such cap, and
\ref{thm:aj-finite} gives thickness at every level.
See \texttt{notes/andersen\_jessen\_brief.md}.

\begin{definition}[The group]
\label{def:aj-A}
\lean{AndersenJessen.A, AndersenJessen.Afull}
\leanok
$A = \mathbb{Z} + \alpha\mathbb{Z}$ for $\alpha$ irrational (as a set, and as
the additive subgroup \texttt{Afull} of $\mathbb{R}$). Countable.
\end{definition}

\begin{lemma}[Density]
\label{lem:aj-dense}
\lean{AndersenJessen.Afull_dense}
\leanok
\uses{def:aj-A}
$A$ is dense in $\mathbb{R}$.
\end{lemma}

\begin{proof}
\leanok
$2\mathbb{Z}+\alpha\mathbb{Z}$ is a subgroup, so by
\texttt{AddSubgroup.dense\_or\_cyclic} it is dense or cyclic; a generator $g$
would give $2 = pg$ and $\alpha = qg$, making $\alpha = 2q/p$ rational. Density
of $A$ follows by monotonicity.
\end{proof}

\begin{lemma}[The transversal]
\label{lem:aj-V}
\lean{AndersenJessen.V_unique}
\leanok
\uses{def:aj-A}
$V$, a set of coset representatives for $\mathbb{R}/A$, has the only property the
argument uses: two representatives congruent mod $A$ are equal.
\end{lemma}

\begin{proof}
\leanok
\texttt{Quotient.out} on $\mathbb{R} \mathbin{/} A$.
\end{proof}

\begin{theorem}[Finiteness beats parity]
\label{thm:aj-finite}
\lean{AndersenJessen.V_add_finite_measurable_null}
\leanok
\uses{lem:aj-V, lem:aj-dense}
For FINITE $F \subseteq A$, the set $V + F$ contains no measurable set of
positive measure.
\end{theorem}

\begin{proof}
\leanok
\uses{lem:aj-dense}
If it did, Steinhaus
(\texttt{Measure.sub\_mem\_nhds\_zero\_of\_addHaar\_pos}) puts a neighbourhood
of $0$ inside $F' - F'$. $F - F$ is finite while $A$ is dense, so some element of
$A$ lies in that neighbourhood and outside $F - F$; the transversal then forces
it into $F - F$ after all.

This is the replacement for the source's Proposition 13, and it is what gives
thickness at EVERY level rather than only at $k = 0$.
\end{proof}

\begin{theorem}[The thick tower]
\label{thm:aj-tower}
\lean{AndersenJessen.X_thick, AndersenJessen.X_antitone, AndersenJessen.X_iInter}
\leanok
\uses{thm:aj-finite, lem:aj-V}
With $A$ enumerated, set $X_k = V + \{a_j : j \geq k\}$. Then $X_k$ is
decreasing (\texttt{X\_antitone}), $\bigcap_k X_k = \emptyset$
(\texttt{X\_iInter}), and every $X_k$ is thick: any measurable set disjoint from
it is null.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:aj-finite, lem:aj-V}
$X_k^c = V + \{a_0,\dots,a_{k-1}\}$, a finite subset of $A$, so
\ref{thm:aj-finite} applies at every $k$. The empty intersection needs only
the enumeration covering $A$ and the transversal property
(\ref{lem:aj-V}): each real decomposes as $v + a$ with $v$ forced, and $a$
eventually falls out of every tail.
\end{proof}

\begin{remark}[What remains]
\label{rmk:aj-remaining}
\uses{thm:aj-tower, def:escaping-tower}
The tower was the hard half, and the layer above it --- the trace measure and
the projective system --- is now done too: see \ref{thm:trace-measure} and
\ref{thm:aj-no-extension}. The refutation is kernel-checked; nothing here rests
on a hand check any more.
\end{remark}

\section{The trace measure, and the refutation}

\begin{definition}[Thickness, relative to a base measure]
\label{def:thick-for}
\lean{ThickTrace.ThickFor}
\leanok
$X$ is \emph{thick for} $\mu$ when every measurable set disjoint from $X$ is
$\mu$-null. Stating it for an arbitrary base rather than for Lebesgue measure
is what makes the total mass finite: the marginals need mass $1$, and
$\lambda(\mathbb{R}) = \infty$. Confining the levels to $[0,1]$ does not work
--- $X_k \cap [0,1]$ is not thick in $\mathbb{R}$ --- but restricting the
\emph{ambient} measure to $\mathrm{Icc}\,0\,1$ does, and leaves the levels
exactly as \ref{thm:aj-tower} builds them.
\end{definition}

\begin{theorem}[The trace measure on a thick set]
\label{thm:trace-measure}
\uses{def:thick-for, thm:aj-tower}
\lean{ThickTrace.traceMeasure, ThickTrace.traceMeasure_apply}
\leanok
For $X$ thick for $\mu$, the traces $\{E \cap X\}$ carry a measure with
$\lambda_X(E \cap X) = \mu(E)$ (Border, Prop.~14).
\end{theorem}

\begin{proof}
\leanok
\uses{thm:aj-tower}
Not obtainable from \texttt{Measure.comap}: that is guarded by ``every
measurable subset has null-measurable image'' and returns $0$ when the guard
fails, which it does for a thick set with thick complement ---
\texttt{Measure.Subtype.measureSpace} is that comap, so it degenerates too. The
$\sigma$-algebra is fine (it is the comap $\sigma$-algebra); only the measure
must be built, via \texttt{Measure.ofMeasurable}. Well-definedness is the one
place thickness is spent: two measurable sets with the same trace differ by a
measurable set disjoint from $X$, hence null, and \ref{thm:aj-tower} is exactly
that hypothesis. $\sigma$-additivity follows because disjoint traces force
a.e.-disjoint representatives.
\end{proof}

\begin{theorem}[Andersen--Jessen admits no extension]
\label{thm:aj-no-extension}
\uses{thm:trace-measure, def:escaping-tower, thm:no-extension-tower, thm:diag-empty}
\lean{ASM.AJ_no_extension_unconditional}
\leanok
The Andersen--Jessen query system carries compatible $\sigma$-additive
probability marginals and admits no extension to a probability measure on
$\Omega$.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:trace-measure, thm:no-extension-tower, thm:diag-empty}
The levels carry trace measures (\ref{thm:trace-measure}) over the restricted
base, so each is a probability measure, and they are compatible along the
tower's inclusions. Pushing them along the diagonal gives Kolmogorov-consistent
marginals; the commutation $\mathrm{bond} \circ \mathrm{diag} = \mathrm{diag}
\circ \mathrm{incl}$ is definitional. The diagonal cylinders then form an
escaping tower: each carries full mass because the diagonal map lands in the
diagonal, and their intersection is empty by \ref{thm:diag-empty}.

That last step is where the diagonal layer is spent. A coherent family is not a
sequence, so it is carried to one by $\mathrm{toSeq}\,\omega : k \mapsto
(\omega_k)_k$, reading off the top coordinate at each level. Two facts about
that sequence are needed, and they come from different places: every entry lies
in its own $X_k$ by the subtype alone, while constancy needs both the coherence
(to move coordinate $0$ down from level $i$ to level $0$) and the base cylinders
(to move it across level $i$ from coordinate $0$ to coordinate $i$).
\ref{thm:diag-empty} then supplies the emptiness, against \ref{thm:aj-tower},
and \ref{thm:no-extension-tower} finishes.

This is what makes the falsity of the original \texttt{UpperDirected} statement
a theorem of the kernel rather than a hand check against the source.
\end{proof}

\begin{definition}[Diagonal cylinders]
\label{def:diag-cylinders}
\lean{Diagonal.Diag}
\leanok

The cylinders decrease (\texttt{Diag\_antitone}), and a point of $D_n$ is

Phrased in Mathlib vocabulary rather than over the project's own definitions.
\end{definition}

\begin{lemma}[The cylinders shrink to the constants]
\label{lem:diag-inter}
\lean{Diagonal.Diag_iInter}
\leanok
\uses{def:diag-cylinders}
$\bigcap_n D_n = \{\omega : \omega_i = \omega_0 \text{ for all } i\}$: the

\end{lemma}

\begin{proof}
\leanok
Both inclusions are immediate from the definition; the forward one reads off the
bound $i \leq n$ at $n = i$.
\end{proof}

\begin{theorem}[The diagonal escapes]
\label{thm:diag-empty}
\lean{Diagonal.pi_inter_diag_eq_empty}
\leanok
\uses{lem:diag-inter}
If $\bigcap_k Y_k = \emptyset$ then $\Pi'(Y) \cap \bigcap_n D_n = \emptyset$.
Generic in $Y$ --- the Lean statement consumes nothing from the tower. Applied
to the thick tower $Y_k = X_k$ of \ref{thm:aj-tower}, whose intersection
is empty, it is the second half of the contradiction: every finite stage
carries the diagonal, the limit contains nothing.
\end{theorem}

\begin{proof}
\leanok
\uses{lem:diag-inter}
A point of every $D_n$ is constant by \ref{lem:diag-inter}, so its common value
$\omega_0$ lies in $Y_k$ for every $k$, i.e.\ in $\bigcap_k Y_k = \emptyset$.
\end{proof}

\begin{lemma}[The cylinders are measurable]
\label{lem:diag-measurable}
\lean{Diagonal.measurableSet_Diag}
\leanok
\uses{def:diag-cylinders}
Each $D_n$ is measurable, so it makes sense to ask for its measure. Coordinate
evaluation is measurable (\texttt{measurable\_coord}), hence so is agreement of
two coordinates (\texttt{measurableSet\_eq\_coord}), and $D_n$ is a finite
intersection of those.
\end{lemma}

\begin{proof}
\leanok
$D_n = \bigcap_{i \leq n} \{\omega : \omega_i = \omega_0\}$, a finite
intersection of measurable sets.
\end{proof}

\begin{lemma}[Continuity from above, in the form the contradiction uses]
\label{lem:cont-above}
\lean{Diagonal.no_mass_one_of_iInter_empty}
\leanok
A probability measure cannot give every term of an antitone measurable sequence
mass one when the intersection is empty.
\end{lemma}

\begin{proof}
\leanok
The measures converge to that of the intersection, which is $0$; if every term
were $1$ the limit would be both $0$ and $1$.
\end{proof}

\chapter{Delay embeddings: a system on the wrong side of the hypothesis}

The extension theorem of the previous chapters needs
\ref{def:seq-upper-directed}, and \S\ref{sec:sequentiality} showed the plain
upper-directed form is false. That leaves a question the counterexample alone
does not answer: is the gap between the two hypotheses a pathology, reachable
only by a construction built to break things, or does an ordinary system fall
into it?

It is not a pathology. The delay query system --- the standard object of
time-series reconstruction, reading $d$ samples at lag $\tau$ --- is
upper-directed and \emph{not} sequentially upper-directed
(\ref{thm:delay-not-sud}). So the extension theorem does not apply to it, and
this is not a defect of the formalization but a fact about delay embeddings.
What does apply is the bounded fixed-lag subsystem (\ref{thm:delay-extension}),
and that is the honest scope of the result.

This chapter is otherwise independent of the $\sigma$-essential material: it
shares the query-system core of Chapter 1 and nothing below it.

\section{The delay system}

\begin{definition}[Delay query]
\label{def:delay-query}
\uses{def:query}
\lean{delayQuery, delayEval}
\leanok
Over a sensor alphabet $X$, the observation space is the bi-infinite stream
space $\Omega = X^{\mathbb{Z}}$. The query at $(d,\tau)$ reads $d$ samples
backwards at lag $\tau$:
\[
  \mathrm{eval}_{d,\tau}(\omega)
    = (\omega_0, \omega_{-\tau}, \omega_{-2\tau}, \dots, \omega_{-(d-1)\tau})
    \in X^d .
\]
\end{definition}

\begin{definition}[Delay refinement]
\label{def:delay-refine}
\uses{def:delay-query, def:refine}
\lean{delayLe, delayRefineMap}
\leanok
$(d,\tau) \leq (d',\tau')$ when $\tau' \mid \tau$ and $d' > (d-1)(\tau/\tau')$:
the finer lag divides the coarser one, and the finer window is long enough to
span the coarser one. The refinement map keeps the coordinates at positions
$0, \tau/\tau', \dots, (d-1)\tau/\tau'$.
\end{definition}

\begin{theorem}[The delay system is a query system]
\label{thm:delay-qs}
\uses{def:delay-query, def:delay-refine, def:query-system}
\lean{delayQuerySystem}
\leanok
$(d,\tau) \mapsto$ the delay query, with the refinement above, satisfies the
query-system axioms.
\end{theorem}

\begin{proof}
\leanok
\uses{def:delay-refine}
Reflexivity and transitivity of the order are divisibility and arithmetic;
$\pi_{\mathrm{refl}}$ and $\pi_{\mathrm{trans}}$ are index computations on
coordinate selections.
\end{proof}

\begin{theorem}[Upper-directed]
\label{thm:delay-upper}
\uses{thm:delay-qs, def:upper-directed}
\lean{delayQuerySystem.upperDirected}
\leanok
Any two delay queries have a common refinement.
\end{theorem}

\begin{proof}
\leanok
\uses{def:delay-refine}
Take lag $\gcd(\tau,\tau')$ and depth
$\max\{(d-1)(\tau/\gcd), (d'-1)(\tau'/\gcd)\} + 1$. The gcd divides both lags,
and the depth is chosen to span both windows.
\end{proof}

\section{The gap, exhibited}

\begin{theorem}[The delay system is NOT sequentially upper-directed]
\label{thm:delay-not-sud}
\uses{thm:delay-qs, def:seq-upper-directed}
\lean{delayQuerySystem.not_seqUpperDirected}
\leanok
There is no single query above the whole sequence $u_n = (n+2,\ 1)$.
\end{theorem}

\begin{proof}
\leanok
\uses{def:delay-refine}
An upper bound $(d'',\tau'')$ would need $\tau'' \mid 1$, hence $\tau''=1$, and
then $(n+2)-1 = n+1 < d''$ for every $n$ --- impossible for a finite $d''$.
Deepening the window is free, but no finite window contains all of them.

Note what separates this from \ref{thm:delay-upper}: upper-directedness asks
for a bound above any \emph{two} queries, and gcd-plus-max supplies one;
sequential upper-directedness asks for a bound above a countable family, and
the depths are unbounded. The two hypotheses differ exactly here, and the delay
system sits between them.
\end{proof}

\begin{remark}[A definitional mismatch with the source, recorded]
\label{prop:delay-mismatch}
\uses{thm:delay-not-sud}
Paper 1's Proposition 2 establishes that every \emph{finite} subfamily of delay
queries has a common refinement, and its Definition 3.3 calls this ``sequential
upper-directedness''. That is not the condition of
\ref{def:seq-upper-directed}, which quantifies over a countable family and is
strictly stronger; the finite-subfamily property is exactly
\ref{thm:delay-upper}. The Lean development uses the stronger reading, so the
paper's extension theorem is stated at a scope its hypothesis does not reach,
and \ref{thm:delay-not-sud} is the witness. The repair is the fixed-lag
restriction below, not a weakening of the theorem.
\end{remark}

\section{Where the extension theorem does apply}

\begin{definition}[Bounded fixed-lag subsystem]
\label{def:delay-fixed-lag}
\uses{def:delay-query}
\lean{delayFixedLagBoundedSystem}
\leanok
Fix a lag $\tau_0$ and a maximum depth $N$; index by $\{d \leq N\}$ with the
order $d \leq d'$ and coordinate-truncation refinement.
\end{definition}

\begin{theorem}[The bounded subsystem is sequentially upper-directed]
\label{thm:delay-fixed-sud}
\uses{def:delay-fixed-lag, def:seq-upper-directed}
\lean{delayFixedLagBoundedSystem.seqUpperDirected}
\leanok
$N$ is a universal upper bound.
\end{theorem}

\begin{proof}
\leanok
\uses{def:delay-fixed-lag}
Every index is $\leq N$, so the constant family at $N$ bounds any sequence. The
obstruction of \ref{thm:delay-not-sud} was unbounded depth, and bounding the
depth removes it.
\end{proof}

\begin{theorem}[Extension for bounded fixed-lag delay embeddings]
\label{thm:delay-extension}
\uses{thm:delay-fixed-sud, thm:observational-extension-seq, def:delay-fixed-lag}
\lean{delayFixedLagBoundedSystem.observational_extension_fixedLag}
\leanok
For fixed $\tau_0$ and $N$, any compatible family of probability marginals on
$\{X^d : d \leq N\}$ extends \emph{uniquely} to a probability measure on
$\Omega$.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:delay-fixed-sud, thm:observational-extension-seq}
\ref{thm:delay-fixed-sud} supplies sequential upper-directedness and
\texttt{evalSurjective} the realizability hypothesis; the extension theorem
applies directly.

This is the honest scope of the delay-embedding extension result:
finite-window, fixed-lag. \ref{thm:delay-not-sud} is why it cannot be stated for
the full system, and \S\ref{sec:sequentiality} is why the hypothesis cannot
simply be weakened to \ref{def:upper-directed} --- that form is refuted.
\end{proof}

\section{Reconstruction, and the join to the query system}

The reconstruction material and the delay query system have shared a file since
the delay chapter was written, and shared nothing else --- no definition, no
lemma, no dependency. They were two subjects with one filename. This section
states the reconstruction side and then joins it, which requires knowing what
the two sides mean by their variables.

The join is not automatic and it is not notation. It rests on two modelling
answers, both of which could have gone the other way:

\begin{itemize}
  \item \emph{Whose space is the query system's parameter?} The
    \textbf{observation} space. \ref{def:delay-query} samples streams
    $\mathbb{Z} \to Y$; taking $Y = \mathbb{R}$ makes its outcomes tuples of
    sensor readings. The state space never appears in it. A state is not an
    outcome of any delay query --- only its observations are.
  \item \emph{Which time direction?} \ref{def:delay-query} samples at
    $0, -\tau, \dots, -(d-1)\tau$ --- the past --- while reconstruction iterates
    $T$ forward. They agree once reading the stream backwards is reading the
    orbit forwards, which is the ordinary delay-coordinate convention. The
    alternative --- past times as \emph{backward} iterates --- needs $T$
    invertible; for invertible $T$ the two generate the same $\sigma$-algebra,
    and for non-invertible $T$ only the convention taken here exists at all.
\end{itemize}

\subsection{The reconstruction side}

\begin{definition}[The delay map $\Phi_h$]
  \label{def:delay-map}
  \lean{delayMap}
  \leanok
  $\Phi_h(x) = (h(T^n x))_{n \in \mathbb{N}}$, the forward orbit of $x$ seen
  through the observable $h$. Paper II indexes on $\mathbb{Z}$; the Lean
  development indexes on $\mathbb{N}$, which for invertible $T$ generates the
  same $\sigma$-algebra.
\end{definition}

\begin{definition}[The observable algebra $\mathcal{O}_h$]
  \label{def:observable-algebra}
  \uses{def:delay-map}
  \lean{observableAlgebra, delayObservableAlgebra}
  \leanok
  The $\sigma$-algebra on the state space generated by a family
  $f : \mathbb{Z} \to (X \to \mathbb{R})$;
  $\mathcal{O}_h := \mathrm{observableAlgebra}(n \mapsto h \circ T^{n})$ is the
  case that matters (\texttt{delayObservableAlgebra}). It is what the observer
  can resolve using $h$ and the dynamics alone.
\end{definition}

\begin{theorem}[Pullback identity]
  \label{thm:obs-eq-comap}
  \uses{def:delay-map, def:observable-algebra}
  \lean{observableAlgebra_eq_comap}
  \leanok
  $\mathcal{O}_h = \Phi_h^{-1}(\mathcal{B}(\mathbb{N} \to \mathbb{R}))$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{def:delay-map}
  $(\le)$ each generator $h \circ T^{n}$ is $\pi_n \circ \Phi_h$, so its level
  sets are pullbacks. $(\ge)$ the product $\sigma$-algebra is generated by the
  coordinate projections, and $\Phi_h^{-1}(\pi_n^{-1}(E)) = (h \circ T^n)^{-1}(E)$.
\end{proof}

\begin{theorem}[Density bridge, Paper II Lemma 4.3]
  \label{thm:density-bridge}
  \lean{density_bridge}
  \leanok
  The $m$-simple functions are dense in $L^2(X, m, \mu|_m)$.
\end{theorem}

\begin{proof}
  \leanok
  \texttt{Lp.simpleFunc.dense}.
\end{proof}

\begin{theorem}[Reconstruction, Paper II Theorem 4.1]
  \label{thm:reconstruction}
  \uses{def:observable-algebra, thm:density-bridge, thm:obs-eq-comap}
  \lean{reconstruction_iff_lpMeas}
  \leanok
  $\mathcal{O}_h = m_0$ modulo $\mu$ if and only if
  $\mathrm{lpMeas}\,\mathcal{O}_h$ is dense in $L^2(\mu)$; and by
  \ref{thm:obs-eq-comap} both are equivalent to $\Phi_h$ being a
  measure-theoretic embedding.
\end{theorem}

\begin{proof}
  \leanok
  \uses{thm:density-bridge}
  Closed-plus-dense on the \texttt{lpMeas} subspace, with
  \ref{thm:density-bridge} supplying density.
\end{proof}

\begin{theorem}[Reconstruction for the delay algebra]
  \label{thm:delay-reconstruction}
  \uses{thm:reconstruction, def:observable-algebra}
  \lean{delay_reconstruction_iff}
  \leanok
  \ref{thm:reconstruction} instantiated at the delay observable algebra.
\end{theorem}

\begin{proof}
  \leanok

\subsection{The join}

\begin{definition}[The observation stream of a state]
  \label{def:orbit-stream}
  \uses{def:delay-query, def:delay-map}
  \lean{orbitStream}
  \leanok
  $\mathrm{orbitStream}\,h\,T\,x$ is the sensor stream a state emits:
  at time $t$ it carries $h(T^{(-t)^{+}} x)$, the observation of the state
  $(-t)$ steps ahead. This is the map that takes a state to an object the delay
  queries can be applied to, and it is where the time convention above is
  discharged.
\end{definition}

\begin{theorem}[Sampling the orbit stream is taking delay coordinates]
  \label{thm:delay-eval-orbit}
  \uses{def:orbit-stream, def:delay-query}
  \lean{delayEval_orbitStream}
  \leanok
  $\mathrm{delayEval}\,d\,\tau\,(\mathrm{orbitStream}\,h\,T\,x)
   = (h(T^{k\tau} x))_{k < d}$. At $\tau = 1$ this is $\Phi_h$ truncated to its
  first $d$ coordinates (\texttt{delayEval\_orbitStream\_one}): the query
  system's outcomes and the reconstruction map are the same data, read at finite
  and at infinite depth.
\end{theorem}

\begin{proof}
  \leanok
  \uses{def:orbit-stream}
  Unfold both sides; the time index $-(k\tau)$ negates to $k\tau$.
\end{proof}

\begin{theorem}[The bridge: the queries resolve exactly $\mathcal{O}_h$]
  \label{thm:query-reconstruction-bridge}
  \uses{thm:delay-eval-orbit, def:observable-algebra, def:delay-query}
  \lean{delayQueryAlgebra_eq_delayObservableAlgebra}
  \leanok
  Let $\mathrm{delayQueryAlgebra}$ be the $\sigma$-algebra induced on the state
  space by pulling every delay query's outcome back along
  \ref{def:orbit-stream}. Then
  \[ \mathrm{delayQueryAlgebra}\,h\,T = \mathcal{O}_h. \]
  What the delay \emph{queries} can resolve about a state is exactly what the
  observable algebra contains --- so \ref{thm:reconstruction} is a statement
  about the query system, not merely about a map that resembles it.
\end{theorem}

\begin{proof}
  \leanok
  \uses{thm:delay-eval-orbit}
  $(\le)$ every coordinate of every query is some $h \circ T^{n}$ by
  \ref{thm:delay-eval-orbit}, so each query map is $\mathcal{O}_h$-measurable
  and its pullback sits below. $(\ge)$ conversely every generator
  $h \circ T^{n}$ is a coordinate of the query $(n+1, 1)$; the negative indices
  of the $\mathbb{Z}$-indexed family collapse to $h$ itself and are covered by
  $d = 1$. Unit lag alone therefore already generates everything --- the lag
  parameter adds resolution to no purpose here, which is worth knowing.
\end{proof}

\begin{remark}[What the join does and does not settle]
  \label{rmk:bridge-scope}
  \uses{thm:query-reconstruction-bridge}
  This connects the two halves of the delay chapter, and it connects the
  reconstruction lane to the query-system core --- the first edge between them.
  It does not transfer the extension theorem: \ref{thm:delay-not-sud} still puts
  the delay query system on the wrong side of sequential upper-directedness, and
  the bridge changes nothing about that. What it buys is that reconstruction is
  now a statement in the query system's own vocabulary.
\end{remark}

\subsection{Fixed lag, and where the pruning chapter's index lives}

\ref{thm:query-reconstruction-bridge} ranges over every lag, and its proof shows
unit lag alone already attains the supremum --- so $\tau$ is invisible to it.
That is a fact about the \emph{supremum}, and it is what has kept this chapter
and the pruning chapter apart: the reconstruction question takes the sup and
washes the lag out, while the pruning question is asked at a fixed lag, which
the sup destroys.

Fixing $\tau$ puts it back, and the two indices turn out to be the same index.
\ref{def:delay-query} samples at $0, -\tau, \dots, -(d-1)\tau$ --- stroboscopic
sampling --- while \ref{def:lisc}'s layered ring $R_L$ has vertex set
$\mathbb{Z}_L \times A$, its layer being position mod $L$. The delay chapter's
lag is the pruning chapter's ring length.

\begin{theorem}[Lag-$\tau$ data is unit-lag data of the power map]
  \label{thm:delay-eval-iterate}
  \uses{thm:delay-eval-orbit, def:delay-map}
  \lean{delayEval_orbitStream_iterate}
  \leanok
  $\mathrm{delayEval}\,d\,\tau\,(\mathrm{orbitStream}\,h\,T\,x)
   = (\Phi_h^{T^{\tau}}(x)_k)_{k<d}$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{thm:delay-eval-orbit}
  $T^{k\tau} = (T^{\tau})^{k}$.
\end{proof}

\begin{theorem}[Fixed lag resolves exactly the observable algebra of $T^{\tau}$]
  \label{thm:lag-algebra}
  \uses{thm:delay-eval-iterate, def:observable-algebra, def:delay-query}
  \lean{delayQueryAlgebraAtLag, delayQueryAlgebraAtLag_eq}
  \leanok
  Let $\mathrm{delayQueryAlgebraAtLag}\,h\,T\,\tau$ be the $\sigma$-algebra
  induced by the delay queries \emph{at the single lag} $\tau$. Then
  \[ \mathrm{delayQueryAlgebraAtLag}\,h\,T\,\tau = \mathcal{O}_h(T^{\tau}). \]
  Stroboscopic observation at lag $\tau$ is therefore not a degraded view of
  $T$; it is the same reconstruction question asked of $T^{\tau}$. In general
  $\mathcal{O}_h(T^{\tau}) < \mathcal{O}_h(T)$ strictly --- the intermediate
  times are genuinely lost --- which is the content
  \ref{thm:query-reconstruction-bridge} cannot see. At $\tau = 1$ nothing is
  lost (\texttt{delayQueryAlgebraAtLag\_one}).
\end{theorem}

\begin{proof}
  \leanok
  \uses{thm:delay-eval-iterate}
  Both inclusions as in \ref{thm:query-reconstruction-bridge}, with $T^{\tau}$
  in place of $T$: every coordinate at lag $\tau$ is some $h \circ (T^{\tau})^n$
  by \ref{thm:delay-eval-iterate}, and conversely each such generator is
  coordinate $n$ of the query $(n+1, \tau)$.
\end{proof}

\begin{remark}[What this buys against the pruning chapter]
  \label{rmk:lag-is-ring-length}
  \uses{thm:lag-algebra}
  $\mathrm{Safe}(\rho)$ (\ref{def:unsafe-safe}) is a statement about which ring
  lengths reconstruct; \ref{thm:lag-algebra} says what ``reconstructs at lag
  $\tau$'' means on the measure-theoretic side. The two chapters are asking one
  question in two vocabularies, and this is the dictionary entry.

  It is not yet a theorem connecting them, and the gap is specific. Pruning's
  $\rho \subseteq A \times A$ is a \emph{relation}; the delay side's $T$ is a
  \emph{function}. Specialising pruning to a deterministic $\rho$ looks to
  trivialise it --- a functional graph has out-degree one, so its cycles are
  disjoint and $\mathrm{LISC}_k(L)$ collapses to ``some cycle has length exactly
  $kL$'' --- so the content of \ref{thm:pruning} lives in the nondeterministic
  case. Closing the gap therefore means carrying the delay side to relations,
  not restricting pruning to maps.
\end{remark}

\section{The predictive layer, and what is not formalized}
\label{sec:delay-gaps}

The structural theory above is complete. The probabilistic half of Paper 1 ---
the predictive law map $\varphi_{d,\tau} : X^d \to \mathcal{P}(X)$, predictive
sufficiency as injectivity of $\varphi$ on the support, the $\tau$-Markov order
$m(\tau,P)$, the characterization $d \geq m(\tau,P) \iff$ sufficiency, and
stationarity of the delay distributions --- is not formalized.

The obstruction recorded when the file was written, that Lean lacked
conditional-distribution infrastructure, no longer holds: Mathlib now has
disintegration for standard Borel spaces
(\texttt{Measure.condKernel}, \texttt{Measure.disintegrate}). The live
constraint is different and worth stating precisely, because it is a question
about \emph{statements} rather than proofs: disintegration requires the target
to be standard Borel, while everything above is proved for an arbitrary
measurable sensor alphabet. Defining $\varphi_{d,\tau}$ therefore either
narrows the whole chapter or applies only to a subclass. The structural results
deserve their generality, so the second is the intended reading.

Takens' theorem is a separate matter: it needs generic-embedding machinery that
Mathlib does not have at all, and it is not a near-term target.

\chapter{From Boolean to orthomodular: where distributivity is spent}

Everything above is the distributive case. A query system's outcome
$\sigma$-algebras are Boolean, and the extension questions there are settled by
$\sigma$-additivity together with an index-layer condition.

This chapter is the hinge to the chapter that follows. It isolates the single
predicate separating that regime from the orthomodular one, and shows the
distributive side of the separation is closed. The point of stating it as one
predicate is that distributivity is not diffused through the theory: it is
spent in exactly one step, and what fails on a non-Boolean structure fails
because that step is unavailable.

\section{The pivot}

\begin{definition}[Block]
  \label{def:se-block}
  \uses{def:sigma-class}
  \lean{SigmaEssential.Block}
  \leanok
  A finite $\bot$-closed family of carrier sets --- a finite context, the
  orthomodular analogue of a finite sub-algebra. States are compared on blocks,
  never globally: the encoding discipline that keeps a local pattern from being
  a global state in disguise.
\end{definition}

\begin{definition}[Intersection-closure: the pivot]
  \label{def:interclosed}
  \uses{def:sigma-class}
  \lean{SigmaEssential.InterClosed}
  \leanok
  $\mathrm{InterClosed}(d)$ says $A, A' \in d \Rightarrow A \cap A' \in d$.

  A Dynkin system with this property is exactly a $\sigma$-algebra, by Dynkin's
  $\pi$--$\lambda$ theorem. So this one predicate is the Boolean/orthomodular
  divide: \textbf{with} it the carrier is distributive and the earlier chapters
  apply; \textbf{without} it the carrier is a concrete orthomodular structure in
  which lattice meet need not be set intersection, and need not exist at all.
\end{definition}

\begin{lemma}[Distributivity is spent here, and only here]
  \label{lem:val-inter}
  \uses{def:interclosed, def:two-valued-state}
  \lean{SigmaEssential.TwoValuedState.val_inter}
  \leanok
  On an intersection-closed carrier a two-valued state is \emph{multiplicative}:
  if $A$ and $A'$ are true, so is $A \cap A'$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{def:interclosed}
  $A = (A \cap A') \sqcup (A \cap A'^{c})$ is a disjoint union inside $d$. Since
  $A \cap A'^{c} \subseteq A'^{c}$ and $A'$ is true, monotonicity makes
  $A \cap A'^{c}$ false; additivity on the disjoint pair forces $A \cap A'$ true.

  This is the only place distributivity is used in the development, and exactly
  the step that fails on a non-Boolean orthomodular lattice: the disjoint
  decomposition of $A$ along $A'$ needs the meet to be set intersection and to
  lie in the carrier.
\end{proof}

\section{MO$_2$: the smallest non-distributive witness}

\begin{definition}[Orthomodular lattice]
  \label{def:oml}
  \lean{QuerySystem.OrthomodularLattice}
  \leanok
  A bounded lattice with an order-reversing involution $x \mapsto x^{\perp}$
  satisfying $x \wedge x^{\perp} = \bot$, $x \vee x^{\perp} = \top$, and the
  orthomodular law $x \le y \Rightarrow y = x \vee (y \wedge x^{\perp})$.
  Distributivity is not assumed; a distributive one is a Boolean algebra.
\end{definition}

\begin{definition}[MO$_2$]
  \label{def:mo2}
  \uses{def:oml}
  \lean{QuerySystem.MO2}
  \leanok
  The six-element lattice $\{\bot, a, a', b, b', \top\}$ with the four middle
  elements pairwise incomparable, $a^{\perp} = a'$ and $b^{\perp} = b'$.
\end{definition}

\begin{theorem}[MO$_2$ is an orthomodular lattice]
  \label{thm:mo2-oml}
  \uses{def:mo2, def:oml}
  \lean{QuerySystem.MO2.instOrthomodularLattice}
  \leanok
  MO$_2$ satisfies every orthomodular lattice axiom.
\end{theorem}

\begin{proof}
  \leanok
  \uses{def:mo2}
  All laws are decidable over a six-element type; the instance discharges them
  by \texttt{decide}.
\end{proof}

\begin{theorem}[The non-distributive gap]
  \label{thm:mo2-gap}
  \uses{thm:mo2-oml}
  \lean{QuerySystem.MO2.gap}
  \leanok
  In MO$_2$, $a \wedge b = \bot$ while $a \not\le b^{\perp}$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{def:mo2}
  By decision over the finite type.

  This is the concrete content of non-distributivity, and what \ref{lem:val-inter}
  needs and cannot get: meeting to $\bot$ no longer implies orthogonality, so the
  disjoint decomposition used there is unavailable. MO$_2$ is the smallest
  structure where the Boolean argument breaks.
\end{proof}

\section{The descent ladder: MO$_2$ carried to infinity}

MO$_2$ above is a six-element counterexample, and on its own that is all it is.
This section is what it was built for. The gap $a \wedge b = \bot$ with
$a \not\le b^{\perp}$ relates \emph{one} element to \emph{one} other; the
descent question asks for one element interleaved in the same way with an
\emph{infinite orthogonal family}. The construction that delivers this is
Navara's, and the route to it is a ladder: replicate the gap across finitely
many blocks, then across countably many.

Every rung is stated over a product of MO$_2$ copies with order, meet,
orthocomplement and bottom all computed coordinatewise. That is the whole
reason the ladder climbs: the star condition is a coordinatewise test, so it
survives replication, and each rung reduces to \ref{thm:mo2-gap} at a single
block.

\subsection{Rung 2: finite truncation}

\begin{definition}[Finite truncation $L_2^N$]
  \label{def:l2n}
  \uses{def:mo2}
  \lean{QuerySystem.L2N}
  \leanok
  $L_2^N := \mathrm{Fin}\,N \to \mathrm{MO}_2$, carrying the product lattice
  order: $x \le y$ iff $x_i \le y_i$ at every block $i$.
\end{definition}

\begin{definition}[Coordinatewise orthocomplement]
  \label{def:pi-ortho}
  \uses{def:l2n}
  \lean{QuerySystem.piOrtho}
  \leanok
  $(x^{\perp})_i := (x_i)^{\perp}$.
\end{definition}

\begin{definition}[The witness and the family]
  \label{def:descent-witness}
  \uses{def:l2n, def:pi-ortho}
  \lean{QuerySystem.p}
  \leanok
  $p := b$ on every block, and $a_i := a$ on block $i$, $\bot$ elsewhere
  (\texttt{QuerySystem.aFam}). The family is orthogonal because its members have
  disjoint supports; $p$ meets all of them.
\end{definition}

\begin{theorem}[$(\star)$ at every finite truncation]
  \label{thm:star-finite}
  \uses{def:descent-witness, thm:mo2-gap}
  \lean{QuerySystem.star_finite}
  \leanok
  For every $N$ and every block $i < N$,
  \[ p \sqcap a_i = \bot \quad\text{and}\quad p \not\le a_i^{\perp}. \]
\end{theorem}

\begin{proof}
  \leanok
  \uses{thm:mo2-gap}
  Both halves are coordinatewise tests. For the meet: off block $i$ the element
  $a_i$ is $\bot$, and on block $i$ the meet is $b \sqcap a = \bot$ by
  \ref{thm:mo2-gap}. For the non-orthogonality: $\le$ on the product is
  pointwise, so $p \le a_i^{\perp}$ would force $b \le a^{\perp}$ at block $i$,
  which \ref{thm:mo2-gap} refutes.

  One block suffices to refute an inequality; every block must be checked to
  confirm one. That asymmetry is why the second half costs a single instance and
  the first is where the work is.
\end{proof}

\subsection{Rung 3: the infinite object, and what is assumed}

Navara's $L_2$ is a genuine infinite object and a published construction
(PAMS 115 (1992) 425--431, p.~428). Per the repository's convention it is
\emph{assumed}, not reformalized. Eleven axioms record its carrier, the fact
that it is an orthomodular lattice, and --- the load-bearing point --- that it
is built as a \textbf{sublogic of the product}, so its order, orthocomplement,
bottom and meet are the restrictions of the coordinatewise ones. Two more name
the witness $p = \bigvee_n b|_{C_n}$ and the family $a_n = a|_{C_n}$.

\emph{None of the eleven mentions $(\star)$.} What is proved is that $(\star)$
follows from the blockwise description, and it follows for the reason it did at
Rung 2: reduction to \ref{thm:mo2-gap} at a single block. The content assumed is
the existence of the object; the content proved is what the object does.

\begin{definition}[Navara's $L_2$, blockwise]
  \label{def:navara-l2}
  \uses{def:mo2, def:oml}
  \lean{QuerySystem.L2}
  \leanok
  The carrier of the $(\beta)$-construction over a countable block index set,
  with block evaluation $\mathrm{blockVal} : L_2 \to \mathbb{N} \to
  \mathrm{MO}_2$ and the sublogic-of-product axioms
  \texttt{le\_iff\_blockwise}, \texttt{blockVal\_ortho}, \texttt{blockVal\_inf},
  \texttt{blockVal\_bot}. Cited, not proved.
\end{definition}

\begin{theorem}[$(\star)$ at infinity]
  \label{thm:star-infinite}
  \uses{def:navara-l2, thm:star-finite, thm:mo2-gap}
  \lean{QuerySystem.star_infinite}
  \leanok
  For Navara's witness $p$ and every $n$ in the countable orthogonal family,
  \[ p \sqcap a_n = \bot \quad\text{and}\quad p \not\le a_n^{\perp}. \]
  So $L_2$ is a concrete, $\sigma$-orthocomplete, non-Boolean orthomodular
  lattice in which one element is meet-zero against an infinite orthogonal
  family while orthogonal to no member of it.
\end{theorem}

\begin{proof}
  \leanok
  \uses{def:navara-l2, thm:mo2-gap}
  Verbatim the Rung-2 argument, with \texttt{le\_iff\_blockwise} supplying the
  coordinatewise order that the product structure gave for free. Antisymmetry of
  the blockwise order reduces the meet to a blockwise identity; the
  non-orthogonality is refuted at block $n$.
\end{proof}

\begin{remark}[What this refutes, and what it does not]
  \label{rmk:descent-refutation}
  \uses{thm:star-infinite}
  This is a second concrete witness alongside $L(H)$, and it tells against the
  reading on which richness starves concreteness: $L_2$ is rich enough to
  exercise descent and concrete all the same. It bears on $\Phi$ only
  indirectly. $\Phi$ is a question about \emph{states}, and nothing here
  constructs a finitely additive two-valued state with no $\sigma$-additive
  reproduction.
\end{remark}

\subsection{The axioms are consistent}

An axiom set with no model proves everything, so \ref{thm:star-infinite} is
worth exactly what the consistency of the eleven axioms behind it is worth.
That obligation is discharged by exhibiting a model, not argued.

\begin{theorem}[Products of orthomodular lattices are orthomodular]
  \label{thm:prod-oml}
  \uses{def:oml}
  \lean{QuerySystem.instPiOrthomodularLattice}
  \leanok
  If each $\alpha_i$ is an orthomodular lattice then so is $\prod_i \alpha_i$,
  with lattice operations, bounds and orthocomplement all coordinatewise.
\end{theorem}

\begin{proof}
  \leanok
  Every law of the class is an equation or an implication between elements, and
  the product order, lattice and bounded-order structures are pointwise; so each
  law follows from the same law in each factor, applied at each coordinate. The
  instance is built \emph{from} the existing product lattice and bounded-order
  structures rather than restating them, so no second, non-defeq copy of the
  operations is introduced.
\end{proof}

\begin{theorem}[A model of the Rung-3 axioms]
  \label{thm:descent-consistent}
  \uses{def:navara-l2, thm:prod-oml}
  \lean{QuerySystem.ConsistencyModel.axioms_consistent}
  \leanok
  Take $M := \mathbb{N} \to \mathrm{MO}_2$ with all operations coordinatewise,
  $p := b$ everywhere, $a_n := a$ on block $n$. Then every one of the eleven
  Rung-3 axioms is a \emph{theorem} about $M$, and $(\star)$ holds in it
  directly. Hence the axiom set is consistent and \ref{thm:star-infinite} is not
  vacuous.
\end{theorem}

\begin{proof}
  \leanok
  \uses{thm:prod-oml}
  Each blockwise axiom is the corresponding product lemma. The remaining
  obligation is the axiom asserting that $L_2$ \emph{is} an orthomodular
  lattice: without it the symbols $\sqcap$, $\le$, $^{\perp}$ and $\bot$ in the
  other ten are undefined, so a model supplying only those ten would establish
  the consistency of a fragment with no vocabulary. \ref{thm:prod-oml} supplies
  it, and the model's orthocomplement is definitionally that structure's
  (\texttt{mOrtho\_eq\_ortho}) rather than a similarly-defined function that
  happens to agree.
\end{proof}

\subsection{Closure: the family is genuinely infinite and orthogonal}

$(\star)$ is a descent statement only if $\{a_n\}$ really is an infinite
orthogonal family. Otherwise it is \ref{thm:mo2-gap} in disguise. These legs
check that, together with one structural fact about how far MO$_2$ can be
pushed.

\begin{lemma}[The tests localize per block]
  \label{lem:legs-localize}
  \uses{thm:descent-consistent}
  \lean{QuerySystem.ClosureLegs.meet_bot_iff_blockwise}
  \leanok
  In $M$, $x \sqcap y = \bot$ iff $x_n \sqcap y_n = \bot$ at every block, and
  $x \le y^{\perp}$ iff $x_n \le (y_n)^{\perp}$ at every block
  (\texttt{ortho\_iff\_blockwise}). Both hold for arbitrary supports, not only
  for single-block elements --- which is what licenses the blockwise reading of
  $p$, whose support is all of $\mathbb{N}$.
\end{lemma}

\begin{proof}
  \leanok
  Function extensionality and the pointwise product lemmas.
\end{proof}

\begin{lemma}[MO$_2$ caps orthogonal families at two]
  \label{lem:mo2-cap-two}
  \uses{def:mo2}
  \lean{QuerySystem.ClosureLegs.mo2_cap_two}
  \leanok
  No three nonzero elements of MO$_2$ are pairwise orthogonal; equivalently, any
  pairwise-orthogonal triple has a $\bot$ member
  (\texttt{mo2\_pairwise\_ortho\_one\_zero}).
\end{lemma}

\begin{proof}
  \leanok
  Finite case check over the six-element type.

  This is why the family has to be spread across blocks rather than found inside
  one: a single MO$_2$ admits no infinite orthogonal family, so the infinitude in
  \ref{thm:star-infinite} comes from the block index and from nowhere else.
\end{proof}

\begin{lemma}[The witness family is infinite and orthogonal]
  \label{lem:afam-ortho}
  \uses{lem:legs-localize, lem:mo2-cap-two}
  \lean{QuerySystem.ClosureLegs.aW_pairwise_ortho}
  \leanok
  For $n \ne m$, $a_n \le a_m^{\perp}$. The family is indexed by $\mathbb{N}$,
  so it is infinite --- which by \ref{lem:mo2-cap-two} it could not have been
  inside a single block.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:legs-localize}
  Blockwise by \ref{lem:legs-localize}. At block $k$ the only case that could
  fail is $k = n = m$, excluded by $n \ne m$; the rest are $a \le \top$,
  $\bot \le a^{\perp}$ and $\bot \le \top$.

  Closure here is \emph{witness-scoped}. The supports $C_n = \{n\}$ are
  singletons, on which Navara's constancy clause is vacuous, so the join
  $p = \bigvee_n b|_{C_n}$ lands in the sublogic automatically. The general
  $\sigma$-orthocompleteness of the construction, for arbitrary $C$, is the
  cited classical result assumed at \ref{def:navara-l2} and is not reproved.
\end{proof}

\section{The distributive case is closed}

\begin{definition}[Finite coherence]
  \label{def:fincoherent}
  \uses{def:local-pattern}
  \lean{SigmaEssential.Amended.FinitelyCoherent}
  \leanok
  The pattern is realized by some finitely additive global state. This is the
  amendment: coherence sits \emph{in} the definition, which is what repairs the
  Boolean baseline.
\end{definition}

\begin{theorem}[Boolean baseline]
  \label{thm:boolean-baseline}
  \uses{def:interclosed, def:fincoherent, lem:val-inter}
  \lean{SigmaEssential.Amended.boolean_baseline}
  \leanok
  On an intersection-closed carrier every finitely coherent local pattern has
  nonempty kernel: some Dirac state extends it.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:val-inter}
  Fold the pattern-true block sets under intersection. Multiplicativity
  (\ref{lem:val-inter}) keeps every partial intersection true, hence nonempty,
  and the block is finite, so the total intersection is nonempty. Any point of it
  carries the required Dirac state. Coherence supplies the state that
  distributivity then converts into a point.

  Its contrapositive, \ref{lem:boolean-no-witness} in the next chapter, is what
  forces any witness to be non-Boolean --- and \ref{lem:carrier-not-boolean}
  confirms the constructed witness is exactly that.
\end{proof}

\chapter{The $\sigma$-essential witness, and the OML gap}

The flagship result. \texttt{psiAmended\_ZFC} is a ZFC theorem with zero
sorries, axioms exactly \texttt{[propext, Classical.choice, Quot.sound]}: a
concrete $\sigma$-complete orthomodular \emph{poset} carrying a finite
two-valued pattern that is finitely coherent yet extends to no global
$\sigma$-additive two-valued state.

This chapter is the fidelity spine of \texttt{papers/sigma\_essential/}. Every
definition on the path from the paper's statement to the Lean theorem appears
as a node with its Lean name attached, so \texttt{checkdecls} enforces the
paper--Lean correspondence rather than leaving it to a hand-maintained
side-by-side. The chapter closes on what is \emph{not} settled: the same
question with latticehood adjoined.

\section{The carrier and its states}

\begin{definition}[$\sigma$-class = concrete $\sigma$-complete OMP]
\label{def:sigma-class}
\lean{MeasurableSpace.DynkinSystem}
\leanok
A family $L \subseteq \mathcal{P}(\Omega)$ with $\emptyset, \Omega \in L$,
closed under complement, and closed under countable \emph{disjoint} unions.
Formalized as Mathlib's \texttt{DynkinSystem}: \texttt{has\_iUnion\_nat} takes
\texttt{Pairwise (Disjoint on f)}, so this is a $\sigma$-class and \emph{not} a
$\sigma$-algebra --- the distinction the whole question turns on.
\end{definition}

\begin{definition}[$\sigma$-additive two-valued state]
\label{def:two-valued-state}
\lean{SigmaEssential.TwoValuedState}
\leanok
\uses{def:sigma-class}
$s : L \to \{0,1\}$ with $s(\Omega) = 1$, additive over countable orthogonal
joins. The paper's $s(\bigcup_n A_n) = \sum_n s(A_n)$ becomes ``union true
if and only if some member true''; the at-most-one half is \emph{derived}
(\texttt{val\_at\_most\_one}) rather than assumed, which makes the structure
weaker-or-equal and the derived lemma restores equality.
\end{definition}

\begin{definition}[Finitely additive two-valued state]
\label{def:finadd-state}
\lean{SigmaEssential.Amended.FinAddState}
\leanok
\uses{def:sigma-class}
Pair additivity only. Finite-family additivity is derived
(\texttt{val\_sUnion\_finset}). Nothing $\sigma$-additive is assumed, so the
coherence clause quantifies over exactly the paper's finitely additive states.
\end{definition}

\begin{definition}[The local pattern]
\label{def:local-pattern}
\lean{SigmaEssential.Amended.LocalState}
\leanok
\uses{def:sigma-class, def:se-block}
A two-valued assignment on a finite $\perp$-closed block $B \subseteq L$.
\textbf{The one substantive encoding choice:} \texttt{LocalState} demands only
complement-additivity inside $B$, weaker in general than the paper's Def 1.2
restricted to $B$. It is closed both ways below.
\end{definition}

\begin{lemma}[The weaker encoding is upgraded by coherence]
\label{lem:pattern-fully-additive}
\lean{SigmaEssential.Amended.coherent_pattern_fully_additive}
\leanok
\uses{def:local-pattern, def:finadd-state}
A finitely coherent \texttt{LocalState} satisfies \emph{every} Def-1.2
constraint on $B$. Since $\sigma$-essentiality requires coherence, the pattern
is a genuine paper-state wherever the definition applies.
\end{lemma}

\begin{proof}
\leanok
Coherence supplies a finitely additive extension on all of $L$; its additivity restricts to every Def-1.2 constraint on the finite block $B$.
\end{proof}

\begin{lemma}[On the witness block the encodings coincide outright]
\label{lem:coreblock-disjoint}
\lean{SigmaEssential.Ulam.coreBlock_disjoint_eq_compl}
\leanok

On the witness's own block the only disjoint pairs are the complement pairs

\end{lemma}

\begin{proof}
\leanok
The twelve cross-intersections of the three cores and their complements are each exhibited nonempty, so the only disjoint pairs are the complement pairs.
\end{proof}

\section{$\sigma$-essentiality and $\Psi$}

\begin{definition}[$\sigma$-essential contextual state]
\label{def:sigma-essential}
\lean{SigmaEssential.Amended.IsSigmaEssentialL}
\leanok
\uses{def:local-pattern, def:two-valued-state, def:finadd-state}
$s_0$ is finitely coherent (extends to \emph{some} finitely additive
two-valued state on all of $L$) and extends to \textbf{no} global
$\sigma$-additive two-valued state. Extension means agreement on $B$; the
negative clause quantifies over all $\sigma$-additive states, Dirac and
non-Dirac alike.
\end{definition}

\begin{definition}[The amended $\Psi$]
\label{def:psi-amended}
\lean{SigmaEssential.Amended.PsiAmended}
\leanok
\uses{def:sigma-essential}
The localized existence sentence: some $\Omega$, $\sigma$-class, block and
pattern with $s_0$ $\sigma$-essential. Admissibility predicates are not
bundled in --- non-Booleanness is \ref{lem:carrier-not-boolean}; essential
irreducibility and non-segregation are paper-level facts about the carrier and
do not affect the truth of \ref{thm:psi-amended-zfc}.
\end{definition}

\begin{theorem}[Localization --- the whole content is clause (ii)]
\label{thm:localization}
\lean{SigmaEssential.Amended.localization_amended}
\leanok
\uses{def:sigma-essential}
$s_0$ is a witness if and only if (0) it is finitely coherent, (i) no
point-evaluation extends it ($K(s_0) = \emptyset$), \emph{and} (ii) no
non-Dirac $\sigma$-additive two-valued state does. Clause (i) is freely
arrangeable by the Navara--Pt\'ak intersection device, so the entire
difficulty is clause (ii) --- Wall A. (The pre-amendment two-clause form is
\texttt{SigmaEssential.localization}.)
\end{theorem}

\begin{proof}
\leanok
A global $\sigma$-additive two-valued state either agrees with a point evaluation on the carrier or does not; splitting on that alternative gives clauses (i) and (ii). Clause (0) is the amendment's coherence conjunct, carried through unchanged.
\end{proof}

\begin{lemma}[No witness on a Boolean carrier]
\label{lem:boolean-no-witness}
\lean{SigmaEssential.Amended.boolean_no_witness_amended}
\leanok
\uses{def:sigma-essential, thm:boolean-baseline, def:interclosed}
On an intersection-closed (Boolean) carrier no $\sigma$-essential contextual
state exists: coherence alone folds the finitely many true sets of the block to
a nonempty intersection, and any point of it gives a Dirac extension. This is
the amended Prop 2.1 --- the statement the $\Omega_7$ example
(\ref{thm:omega7}) shows is \emph{false} under the literal, unamended
definition.
\end{lemma}

\begin{proof}
\leanok
Via \texttt{boolean\_baseline}: the finitely additive extension supplied by
coherence is multiplicative on the intersection-closed carrier, so the block's
true sets have the finite intersection property; a common point is a kernel
point.
\end{proof}

\section{Inner regularity: what forces $\sigma$-additivity, and what refuses it}

The witness above is a finitely additive two-valued state with no $\sigma$-additive
extension. That raises the complementary question: what would \emph{force}
$\sigma$-additivity, and how close does the boundary run? Two answers, in
opposite directions, and a counterexample that fixes the gap between them.

\subsection{Transport: a compact class forces the $\sigma$-condition}

\begin{definition}[Countably compact class]
  \label{def:countably-compact-class}
  \lean{SigmaEssential.Blocks.CountablyCompactClass}
  \leanok
  A family $\mathcal{K}$ of sets such that every sequence from $\mathcal{K}$
  with the finite intersection property has nonempty total intersection.
\end{definition}

\begin{definition}[Marczewski approximation, two-valued in-block form]
  \label{def:approximates-on}
  \uses{def:countably-compact-class, def:finadd-state}
  \lean{SigmaEssential.Blocks.ApproximatesOn}
  \leanok
  Every value-$1$ member $E$ of the family carries a value-$1$ inner witness
  sandwiched below a class member: $D \subseteq K \subseteq E$ with
  $K \in \mathcal{K}$. Two-valuedness is what collapses Marczewski's general
  $\eta$-form to this.
\end{definition}

\begin{theorem}[Compact transport]
  \label{thm:compact-transport}
  \uses{def:approximates-on, def:countably-compact-class, def:sigma-class}
  \lean{SigmaEssential.Blocks.compact_transport,
        SigmaEssential.Blocks.IsMaxBlock.isSigmaOn_of_compact_class}
  \leanok
  On a carrier where meets exist, a finitely additive two-valued state
  approximated on every maximal block by a countably compact class satisfies the
  $\sigma$-condition on the whole carrier.
\end{theorem}

\begin{proof}
  \leanok
  \uses{def:approximates-on}
  Run inside one maximal block and glue. Tails of a disjoint sequence are
  value-$1$, so their inner witnesses have value-$1$ --- hence nonempty ---
  finite intersections; the sandwiching compacts therefore have the finite
  intersection property, and countable compactness contradicts the empty tail
  intersection. Marczewski 1953, 4(i), two-valued.
\end{proof}

\subsection{Forcing: (8.1) makes a finitely additive state Dirac}

\begin{definition}[A field of sets and a two-valued state on it]

  \lean{SigmaEssential.Blocks.SetField, SigmaEssential.Blocks.FieldState}
  \leanok

  finitely additive statement and is stated where that is all there is.
\end{definition}

\begin{definition}[Inner regularity (8.1)]
  \label{def:inner-regular-81}
  \uses{def:set-field}
  \lean{SigmaEssential.Blocks.InnerRegular81}
  \leanok
  Every value-$1$ member of the field contains a value-$1$ in-field
  \emph{compact}.
\end{definition}

\begin{theorem}[Theorem-let R: (8.1) forces Dirac]
  \label{thm:dirac-forcing}
  \uses{def:inner-regular-81, def:set-field}
  \lean{SigmaEssential.Blocks.dirac_forcing}
  \leanok

  has nonempty $\mathcal{K}_1$ (its value-$1$ in-field compacts), nonempty
  $D = \bigcap \mathcal{K}_1$, and \emph{equals} $\delta_\omega$ on the field

\end{theorem}

\begin{proof}
  \leanok

  \textbf{No $\sigma$-additivity is used anywhere.}
\end{proof}

\section{The two sides: cycles in the layered ring, walks in the tuple digraph}

\begin{definition}[Layer-injective simple cycle]
\label{def:lisc}
\lean{Reconstruction.Pruning.IsLISC}
\leanok
For $L \geq 1$ the \emph{layered ring} $R_L(\rho)$ has vertex set
$\mathbb{Z}_L \times A$ and arcs $(i,a) \to (i+1 \bmod L, b)$ for every
$(a,b) \in \rho$. A simple cycle of \emph{winding} $k$ is a closed walk of
length $kL$ with all vertices distinct. $\mathrm{LISC}_k(L)$ holds when one
exists.

Encoded by the state sequence $W$ on positions $t < kL$: $\rho$-steps between
consecutive positions, a closing $\rho$-step, and the simplicity clause ---
same layer $(t \bmod L)$ plus same state implies same position --- which is
exactly layer-injectivity, automatic given simplicity.
\end{definition}

\begin{definition}[Tuple digraph and rotation walks]
\label{def:tr}
\lean{Reconstruction.Pruning.IsTR}
\leanok
$T_k(\rho)$ has as vertices the \emph{injective} $k$-tuples $u \in A^k$, with
arcs $u \to v$ iff $(u_j, v_j) \in \rho$ for every $j$. For $r \in \mathbb{Z}_k$
let $\sigma_r(u)_j = u_{j+r}$. Then $\mathrm{TR}_k^{(r)}(L)$ holds when some
length-$L$ walk in $T_k(\rho)$ runs from a $u$ to $\sigma_r(u)$.

Encoded as $\tau : \mathbb{N} \to \mathbb{Z}_k \to A$ with $\tau_i$ injective
for $i \leq L$, coordinatewise $\rho$-arcs, and endpoint
$\tau_L\,j = \tau_0\,(j+r)$.
\end{definition}

\begin{lemma}[Step 2: extraction]
\label{lem:extraction}
\lean{Reconstruction.Pruning.isTR_one_of_isLISC}
\leanok
\uses{def:lisc, def:tr}
For $0 < L$, $\mathrm{LISC}_k(L)$ implies $\mathrm{TR}_k^{(1)}(L)$.
\end{lemma}

\begin{proof}
\leanok
\uses{def:lisc, def:tr}
Read the $k$ strands at each layer: set $\tau_i[j]$ to the state of $W[jL+i]$.
Injectivity of each $\tau_i$ is forced by simplicity --- distinct vertices at
the same layer have distinct states --- so off-diagonality is not an extra
assumption. The endpoint computes to $\sigma_1(\tau_0)$.
\end{proof}

\begin{lemma}[Step 3: assembly]
\label{lem:assembly}
\lean{Reconstruction.Pruning.isLISC_of_isTR_one}
\leanok
\uses{def:lisc, def:tr}

\end{lemma}

\leanok
\uses{def:lisc, def:tr}

injectivity of each $\tau_i$. Ratification kit gap 5.1.

\begin{lemma}[Step 4: generator conjugation]
\label{lem:conjugation}
\lean{Reconstruction.Pruning.isTR_rot_iff_isTR_one}
\leanok
\uses{def:tr}
If $r$ is a unit of $\mathbb{Z}_k$, that is $\gcd(r,k)=1$, then

\begin{proof}
\leanok
\uses{def:tr}

\end{theorem}

\leanok

\begin{definition}[Eventually periodic subset of the naturals]
\label{def:evperiodic}
\lean{Reconstruction.Pruning.IsEvPeriodic}
\leanok

and a period $p > 0$ with $L \in S$ iff $L+p \in S$, for every $L \geq s$.
Finite unions, intersections and complements are again eventually periodic,
with threshold $\max$ and period $\mathrm{lcm}$.
\end{definition}

\leanok

\leanok

\uses{lem:ng}
A rotation-$r$ tuple walk certifies a simple cycle of winding
$\mathrm{addOrderOf}(r) = k/\gcd(r,k)$ --- equal to $k$ exactly when $r$
generates. No orbit map has to be supplied by hand.
\end{lemma}

\begin{proof}
\leanok
\uses{lem:ng}
The orbit map is \texttt{ZMod.lift} applied to \texttt{zmultiplesHom}, so
$f(j+1) = f(j) + r$ falls out of \texttt{map\_add} rather than a case split on
the wrap-around at $j = m-1$. Injectivity is the one place \texttt{.val} is
needed, and there it is a single divisibility step.

Feeding this to \ref{lem:ng} gives the winding-$m$ conclusion with no data
argument. Still open: pairwise disjointness of the $d$ orbit cycles.
\end{proof}

\begin{theorem}[Lemma NG, refutation half]
\label{thm:ng-refute}
\lean{Reconstruction.Pruning.NG_refutation}
\leanok
\uses{def:lisc, def:tr}
For the directed 4-cycle at $L = 2$ and the non-generator $r = 2$
($\gcd(2,4) = 2$): $\mathrm{TR}_4^{(2)}(2)$ holds, $\mathrm{LISC}_2(2)$ holds,
and $\mathrm{LISC}_4(2)$ \emph{fails}.
\end{theorem}

\begin{proof}
\leanok
\uses{lem:ng-orbit}
The relation is a FUNCTION, so every vertex of the layered ring has out-degree
one and a walk is determined by its start; it returns to its starting state
after four steps. A winding-4 simple cycle needs eight pairwise distinct
positions, but positions $0$ and $4$ share a layer and a state. The
winding-2 cycle and the rotation-2 slot walk $\tau_i(j) = j + i$ are exhibited
directly.

So a rotation witness at a non-generator certifies winding $k/d = 2$, not
winding $k$ --- Theorem P's equivalence genuinely needs $r$ to generate, and
\ref{lem:ng-orbit} is sharp.
\end{proof}

\section{Relational delay systems: the pruning objects as delay data}

\ref{rmk:lag-is-ring-length} left a dictionary and named the obstruction to
turning it into a theorem: the delay chapter observes a \emph{function}, the
pruning chapter studies a \emph{relation}, and specialising pruning to a
deterministic $\rho$ trivialises it. So the join carries the delay side up to
relations.

A relation has no orbit map, but it has a \emph{trajectory space}, and delay
queries apply to a trajectory space exactly as they apply to a stream.

\begin{definition}[The subshift of a relation]
  \label{def:subshift}
  \uses{def:delay-query}
  \lean{QuerySystem.RelationalDelay.Subshift}
  \leanok
  $\Sigma_{\rho} = \{\omega : \mathbb{Z} \to A \mid \rho(\omega_{t+1},
  \omega_t) \text{ for all } t\}$, the bi-infinite $\rho$-trajectories inside
  the sensor streams. The orientation is the one \ref{def:orbit-stream} already
  fixed and \ref{def:delay-query}'s backward sampling forces: reading the stream
  backwards runs the relation forwards.
\end{definition}

\begin{definition}[The trajectory of a cycle]
  \label{def:cycle-stream}
  \uses{def:subshift, def:lisc}
  \lean{QuerySystem.RelationalDelay.cycleStream}
  \leanok
  From a cycle witness $W$ and modulus $n$, the stream carrying $W$ at index
  $(-t) \bmod n$. The negation is the time convention; the modulus makes it
  bi-infinite and $n$-periodic (\texttt{cycleStream\_periodic}).
\end{definition}

\begin{lemma}[A cycle is a trajectory]
  \label{lem:cycle-stream-subshift}
  \uses{def:cycle-stream, def:subshift, def:lisc}
  \lean{QuerySystem.RelationalDelay.cycleStream_mem_subshift}
  \leanok
  $\mathrm{cycleStream}\,W\,(kL) \in \Sigma_{\rho}$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{def:cycle-stream}
  Interior positions are the cycle's own arcs. The one position where the index
  rolls over is where the closing arc $\rho(W_{kL-1}, W_0)$ is spent --- which is
  the only place the \emph{cycle} hypothesis, as opposed to the walk hypothesis,
  is used.
\end{proof}

\begin{theorem}[The bridge: layer-injectivity is delay-injectivity]
  \label{thm:lisc-delay-witness}
  \uses{lem:cycle-stream-subshift, def:lisc, def:delay-query, thm:lag-algebra}
  \lean{QuerySystem.RelationalDelay.isLISC_delay_witness}
  \leanok
  If $\mathrm{LISC}_k(L)$ then there is a $\rho$-trajectory $\omega$, periodic
  of period $kL$, whose depth-$k$, lag-$L$ delay query is \textbf{injective}.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:cycle-stream-subshift, def:lisc}
  Take $\omega = \mathrm{cycleStream}\,W\,(kL)$; \ref{lem:cycle-stream-subshift}
  gives membership and periodicity is immediate. For injectivity: the delay
  query at lag $L$ and depth $k$ samples $\omega$ at $0, -L, \dots, -(k-1)L$,
  and those are precisely the positions $0, L, \dots, (k-1)L$ of the cycle ---
  the $k$ positions sharing layer $0$. \ref{def:lisc}'s simplicity clause says
  exactly that these carry distinct states.

  So the two chapters' injectivity conditions are one condition. Layer
  injectivity in the layered ring \emph{is} injectivity of the lag-$L$ delay
  query on the trajectory, and this is why the two lanes kept meeting the same
  combinatorics from opposite sides.
\end{proof}

\begin{corollary}[$\mathrm{Unsafe}$, in the delay vocabulary]
  \label{cor:unsafe-delay}
  \uses{thm:lisc-delay-witness, def:unsafe-safe}
  \lean{QuerySystem.RelationalDelay.unsafe_delay_witness}
  \leanok
  $L \in \mathrm{Unsafe}(\rho)$ yields a periodic $\rho$-trajectory on which
  lag-$L$ sampling fails to collapse: it resolves $k \geq 2$ distinct strands
  where safety would see one.
\end{corollary}

\begin{proof}
  \leanok
  \uses{thm:lisc-delay-witness, def:unsafe-safe}
  \ref{def:unsafe-safe} unfolds to an $\mathrm{LISC}_k(L)$ with $k \geq 2$;
  apply \ref{thm:lisc-delay-witness}.
\end{proof}

\begin{lemma}[The bridge is not vacuous]
  \label{lem:lisc-inhabited}
  \uses{thm:lisc-delay-witness}
  \lean{QuerySystem.RelationalDelay.isLISC_complete_bool,
        QuerySystem.RelationalDelay.exists_delay_witness_complete_bool}
  \leanok
  The complete relation on $\{\bot,\top\}$ has $\mathrm{LISC}_2(1)$, so
  \ref{thm:lisc-delay-witness} fires and its conclusion is inhabited.
\end{lemma}

\begin{proof}
  \leanok
  Steps and closure are trivially available; simplicity is the two-case check
  that the cycle's two states differ. An implication with an unsatisfiable
  hypothesis proves nothing, so this is recorded rather than assumed.
\end{proof}

\subsection{The deterministic case, as a subclass}

The relational description is the general one, so the deterministic theory has
to sit \emph{inside} it at $\rho = \mathrm{graph}(T)$ rather than beside it.
It does.

\begin{lemma}[Delay queries see only non-positive times]
  \label{lem:delay-nonpos}
  \uses{def:delay-query}
  \lean{QuerySystem.RelationalDelay.delayEval_congr_nonpos}
  \leanok
  Streams agreeing at every $t \leq 0$ are indistinguishable to every delay
  query.
\end{lemma}

\begin{proof}
  \leanok
  \ref{def:delay-query} samples at $0, -\tau, \dots, -(d-1)\tau$.
\end{proof}

\begin{lemma}[The two halves of a deterministic trajectory]
  \label{lem:graph-subshift-halves}
  \uses{def:subshift, def:orbit-stream}
  \lean{QuerySystem.RelationalDelay.graphSubshift_neg,
        QuerySystem.RelationalDelay.graphSubshift_pos}
  \leanok
  For $\omega \in \Sigma_{\mathrm{graph}(T)}$: the non-positive half is the
  forward orbit of the base point, $\omega(-n) = T^n(\omega_0)$; the positive
  half is a coherent backward orbit, $T^n(\omega_n) = \omega_0$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{def:subshift}
  Induction in each direction; the trajectory condition read forwards and
  backwards respectively.
\end{proof}

\begin{theorem}[Deterministic is a subclass]
  \label{thm:deterministic-subclass}
  \uses{lem:graph-subshift-halves, lem:delay-nonpos, def:orbit-stream, thm:delay-eval-orbit}
  \lean{QuerySystem.RelationalDelay.delayEval_comp_eq_orbitStream}
  \leanok
  For $\omega \in \Sigma_{\mathrm{graph}(T)}$ and an observable $h$, the delay
  data the relational picture assigns to $\omega$ is literally the delay data
  the deterministic picture assigns to $\omega_0$.

  So \ref{def:orbit-stream} and everything proved from it ---
  \ref{thm:delay-eval-orbit}, \ref{thm:lag-algebra},
  \ref{thm:query-reconstruction-bridge} --- are the $\rho = \mathrm{graph}(T)$
  instance of this chapter, not a parallel development that resembles it.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:graph-subshift-halves}
  Both sides evaluate at $-(k\tau)$, and \ref{lem:graph-subshift-halves} says
  that is $T^{k\tau}(\omega_0)$.
\end{proof}

\begin{remark}[The honest residue]
  \label{rmk:deterministic-residue}
  \uses{thm:deterministic-subclass}
  \lean{QuerySystem.RelationalDelay.graphSubshift_base_mem_range}
  \leanok
  An earlier draft of this section claimed there was no deterministic embedding
  at all, on the grounds that the state stream $t \mapsto T^{(-t)^{+}}x$ is not
  in $\Sigma_{\mathrm{graph}(T)}$ --- the clamp makes its positive half
  constant, so the trajectory condition there would demand $x = Tx$. That is
  true and it is the wrong shape of objection: by \ref{lem:delay-nonpos} the
  positive half is never sampled, so agreement there was never something the
  theory needed.

  What genuinely does not transfer is smaller, and is a condition on $T$ rather
  than on observation. Being the base of a bi-infinite trajectory forces
  $\omega_0 \in \mathrm{range}(T^n)$ for every $n$, so a non-surjective $T$ has
  points no relational trajectory can be based at --- while the delay data of
  those points is perfectly well defined by \ref{def:orbit-stream}. The
  deterministic picture is the $\rho = \mathrm{graph}(T)$ case on delay data,
  which is what the theory observes, and is strictly larger in base points.
\end{remark}

\chapter{Query Systems and Observational Foundations}

This formalization develops query systems: preordered families of measurable spaces with compatible refinement maps.

\section{Core Definitions}

\begin{definition}
\label{def:query}
\lean{Query}
\leanok
A \emph{query} is a measurable space representing the possible outcomes of asking a single question or making a single observation.
\end{definition}

\begin{definition}
\label{def:refine}
\lean{Refine}
\leanok
A \emph{refinement} between queries is a measurable function showing how outcomes of one query determine outcomes of another.
\end{definition}

\begin{definition}
\label{def:query-system}
\lean{QuerySystem}
\leanok
A \emph{query system} is a preordered family of queries with coherent refinement maps. It consists of:
\begin{itemize}
\item An index type $\iota$
\item A family of queries $q : \iota \to \text{Query}$
\item A preorder relation $\le$ on indices
\item Refinement maps $\pi : \forall \{i\ j\}, i \le j \to \text{Refine}(q_i, q_j)$
\end{itemize}
subject to functoriality conditions.
\end{definition}

\begin{definition}
\label{def:omega}
\lean{QuerySystem.Omega}
\leanok
The \emph{projective limit} $\Omega$ is the space of coherent families of query outcomes.
An element $\omega : \Omega$ assigns to each query $i$ an outcome $\omega(i)$, subject to the coherence condition:
whenever $i \le j$, we have $\omega(j) = \pi_{ij}(\omega(i))$.
\end{definition}

\begin{definition}
\label{def:cylinder}
\lean{QuerySystem.Cyl}
\leanok
A \emph{cylinder set} $\text{Cyl}(i, A)$ is the set of realizations where query $i$ yields an outcome in $A$.
These generate the observable $\sigma$-algebra.
\end{definition}

\begin{definition}
\label{def:compatible-marginals}
\lean{QuerySystem.CompatibleMarginals}
\leanok
\uses{def:query-system, def:refine}
A family $\nu$ of measures on query outcomes is \emph{compatible} if the measures commute with refinement maps:
whenever $i \le j$, the pushforward of $\nu_i$ along $\pi_{ij}$ equals $\nu_j$.
\end{definition}

\begin{definition}
\label{def:lower-directed}
\lean{QuerySystem.LowerDirected}
\leanok
A query system is \emph{lower-directed} if any two queries have a common coarsening:
$\forall i\ j, \exists k, k \le i \land k \le j$.
\end{definition}

\section{The two sides: cycles in the layered ring, walks in the tuple digraph}

\begin{definition}[Layer-injective simple cycle]
\label{def:lisc}
\lean{Reconstruction.Pruning.IsLISC}
\leanok
For $L \geq 1$ the \emph{layered ring} $R_L(\rho)$ has vertex set
$\mathbb{Z}_L \times A$ and arcs $(i,a) \to (i+1 \bmod L, b)$ for every
$(a,b) \in \rho$. A simple cycle of \emph{winding} $k$ is a closed walk of
length $kL$ with all vertices distinct. $\mathrm{LISC}_k(L)$ holds when one
exists.

Encoded by the state sequence $W$ on positions $t < kL$: $\rho$-steps between
consecutive positions, a closing $\rho$-step, and the simplicity clause ---
same layer $(t \bmod L)$ plus same state implies same position --- which is
exactly layer-injectivity, automatic given simplicity.
\end{definition}

\begin{definition}[Tuple digraph and rotation walks]
\label{def:tr}
\lean{Reconstruction.Pruning.IsTR}
\leanok
$T_k(\rho)$ has as vertices the \emph{injective} $k$-tuples $u \in A^k$, with
arcs $u \to v$ iff $(u_j, v_j) \in \rho$ for every $j$. For $r \in \mathbb{Z}_k$
let $\sigma_r(u)_j = u_{j+r}$. Then $\mathrm{TR}_k^{(r)}(L)$ holds when some
length-$L$ walk in $T_k(\rho)$ runs from a $u$ to $\sigma_r(u)$.

Encoded as $\tau : \mathbb{N} \to \mathbb{Z}_k \to A$ with $\tau_i$ injective
for $i \leq L$, coordinatewise $\rho$-arcs, and endpoint
$\tau_L\,j = \tau_0\,(j+r)$.
\end{definition}

\section{Theorem P}

\begin{lemma}[Step 2: extraction]
\label{lem:extraction}
\lean{Reconstruction.Pruning.isTR_one_of_isLISC}
\leanok
\uses{def:lisc, def:tr}
For $0 < L$, $\mathrm{LISC}_k(L)$ implies $\mathrm{TR}_k^{(1)}(L)$.
\end{lemma}

\begin{proof}
\leanok
\uses{def:lisc, def:tr}
Read the $k$ strands at each layer: set $\tau_i[j]$ to the state of $W[jL+i]$.
Injectivity of each $\tau_i$ is forced by simplicity --- distinct vertices at
the same layer have distinct states --- so off-diagonality is not an extra
assumption. The endpoint computes to $\sigma_1(\tau_0)$.
\end{proof}

\begin{lemma}[Step 3: assembly]
\label{lem:assembly}
\lean{Reconstruction.Pruning.isLISC_of_isTR_one}
\leanok
\uses{def:lisc, def:tr}
For $0 < L$, $\mathrm{TR}_k^{(1)}(L)$ implies $\mathrm{LISC}_k(L)$.
\end{lemma}

\begin{proof}
\leanok
\uses{def:lisc, def:tr}
Place position $t = jL+i$ at vertex $(i, \tau_i[j])$. Arcs within a strand are
the tuple-walk arcs; at a strand boundary the rotation endpoint supplies
$\tau_L[j] = \tau_0[j+1]$, which is the state at position $(j+1)L$, and at
$j = k-1$ this is the closure. Simplicity follows from the layer clock plus
injectivity of each $\tau_i$. Ratification kit gap 5.1.
\end{proof}

\begin{lemma}[Step 4: generator conjugation]
\label{lem:conjugation}
\lean{Reconstruction.Pruning.isTR_rot_iff_isTR_one}
\leanok
\uses{def:tr}
If $r$ is a unit of $\mathbb{Z}_k$, that is $\gcd(r,k)=1$, then
$\mathrm{TR}_k^{(r)}(L)$ holds iff $\mathrm{TR}_k^{(1)}(L)$ does.
\end{lemma}

\begin{proof}
\leanok
\uses{def:tr}
The slot relabelling $P(u)[j] = u[j\bar r]$, with $\bar r = r^{-1}$, is a
bijection of injective tuples, preserves arcs (a fixed coordinate permutation),
and conjugates the rotations via $(j+r)\bar r = j\bar r + 1$. Ring algebra in
$\mathbb{Z}_k$, so no index chasing is possible. Ratification kit gap 5.2.
\end{proof}

\begin{theorem}[Theorem P: pruning, phase-parametrized]
\label{thm:pruning}
\lean{Reconstruction.Pruning.pruning}
\leanok
\uses{lem:extraction, lem:assembly, lem:conjugation}
For every $\rho$, every $L \geq 1$, every $k \geq 1$ and every $r$ generating
$\mathbb{Z}_k$, $\mathrm{LISC}_k(L)$ holds iff $\mathrm{TR}_k^{(r)}(L)$ does.
Exact for every $L$, not merely cofinitely, and with no strong connectivity.
\end{theorem}

\begin{proof}
\leanok
\uses{lem:extraction, lem:assembly, lem:conjugation}
Conjugate to $r=1$ by \ref{lem:conjugation}, then combine \ref{lem:extraction}
and \ref{lem:assembly}.
\end{proof}

\begin{lemma}[Lemma NG: assembly half]
\label{lem:ng}
\lean{Reconstruction.Pruning.isLISC_of_isTR_of_orbit}
\leanok
\uses{lem:assembly, def:tr, def:lisc}
Suppose the rotation $r$ closes an orbit of size $m$, witnessed by an injective
slot map $f : \mathbb{Z}_m \to \mathbb{Z}_k$ with $f(j+1) = f(j) + r$. Then a
rotation-$r$ tuple walk certifies $\mathrm{LISC}_m(L)$: winding $m$, not
winding $k$. For a generator $m = k$; for $\gcd(r,k) = d > 1$ it is the
strictly weaker $m = k/d$.
\end{lemma}

\begin{proof}
\leanok
\uses{lem:assembly}
Read the walk through $f$: this is a rotation-$1$ witness of width $m$, so
\ref{lem:assembly} assembles it. The instance $m=k$, $f = \mathrm{id}$, $r=1$
recovers Step 3 exactly, and that instance is checked in the Lean file so the
parametrization cannot drift.

Not formalized, and recorded as such: the construction of $f$ at
$m = k/\gcd(r,k)$; that the $d$ orbit cycles are pairwise vertex-disjoint; and
the refutation half, namely that $\mathrm{LISC}_k$ can genuinely fail while
$\mathrm{TR}_k^{(r)}$ holds, witnessed by C4dir at $L=2$, which stays
machine-checked in the oracle.
\end{proof}

\section{Theorem B}

\begin{definition}[Eventually periodic subset of the naturals]
\label{def:evperiodic}
\lean{Reconstruction.Pruning.IsEvPeriodic}
\leanok
$S \subseteq \mathbb{N}$ is eventually periodic when there are a threshold $s$
and a period $p > 0$ with $L \in S$ iff $L+p \in S$, for every $L \geq s$.
Finite unions, intersections and complements are again eventually periodic,
with threshold $\max$ and period $\mathrm{lcm}$.
\end{definition}

\begin{lemma}[The k-cap]
\label{lem:kcap}
\lean{Reconstruction.Pruning.le_card_of_isLISC}
\leanok
\uses{def:lisc}
If $0 < L$ and $\mathrm{LISC}_k(L)$ then $k \leq |A|$.
\end{lemma}

\begin{proof}
\leanok
\uses{def:lisc}
The positions $0, L, \dots, (k-1)L$ all sit at layer $0$ and are pairwise
distinct, so their states are $k$ distinct elements of $A$. Pigeonhole.
\end{proof}

\begin{definition}[Unsafe and Safe]
\label{def:unsafe-safe}
\lean{Reconstruction.Pruning.Safe}
\leanok
\uses{def:lisc}
$\mathrm{Unsafe}(\rho)$ is the set of $L \geq 1$ with $\mathrm{LISC}_k(L)$ for
some $k \geq 2$, and $\mathrm{Safe}(\rho)$ is its complement among the
positives. Ring reconstruction $C_L = R_L$ holds exactly on
$\mathrm{Safe}(\rho)$.
\end{definition}

\begin{lemma}[Unsafe is a finite union of reachability sets]
\label{lem:unsafe-biunion}
\lean{Reconstruction.Pruning.unsafe_eq_biUnion}
\leanok
\uses{lem:kcap, thm:pruning, def:unsafe-safe, def:tr}
For finite $A$, $\mathrm{Unsafe}(\rho)$ is the union of $\mathrm{TR}_k^{(1)}$
over $2 \leq k \leq |A|$.
\end{lemma}

\begin{proof}
\leanok
\uses{lem:kcap, thm:pruning}
The k-cap bounds the winding by $|A|$, making the union finite; Theorem P
converts each $\mathrm{LISC}_k$ into $\mathrm{TR}_k^{(1)}$.
\end{proof}

\begin{lemma}[Deterministic orbits in a finite type are eventually periodic]
\label{lem:iterate}
\lean{Reconstruction.Pruning.exists_evPeriodic_iterate}
\leanok
\uses{def:evperiodic}
For $f : \beta \to \beta$ with $\beta$ finite and any $b$, there are $s$ and
$p > 0$ with $f^{n+p}(b) = f^{n}(b)$ for all $n \geq s$.
\end{lemma}

\begin{proof}
\leanok
The sequence $n \mapsto f^n(b)$ cannot be injective, so it repeats; being
generated deterministically, it is periodic from the first repeat onward. This
is the first-repeat argument of the source note, with the Boolean-matrix monoid
packaging removed.
\end{proof}

\begin{lemma}[Reachability is an iterate, and matches the raw walk]
\label{lem:reach-walk}
\lean{Reconstruction.Pruning.reach_iff_walk}
\leanok
\uses{def:tr}
Length-$n$ reachability in $T_k(\rho)$ is the $n$-th iterate of a single step
operator on relations over injective tuples, and it holds between $u$ and $v$
exactly when a raw tuple walk of the shape in \ref{def:tr} runs from $u$ to
$v$.
\end{lemma}

\begin{proof}
\leanok
Induction on $n$, rebuilding the walk one layer at a time. This bridge is what
the iterate formulation costs, and it is where the work of the formalization
went.
\end{proof}

\begin{lemma}[Per-k eventual periodicity]
\label{lem:evper-tr}
\lean{Reconstruction.Pruning.isEvPeriodic_isTR}
\leanok
\uses{lem:iterate, lem:reach-walk, def:evperiodic, def:tr}
For finite $A$, the set of $L$ with $\mathrm{TR}_k^{(1)}(L)$ is eventually
periodic.
\end{lemma}

\begin{proof}
\leanok
\uses{lem:iterate, lem:reach-walk}
The vertex set of $T_k(\rho)$ is finite, hence so is the type of relations on
it; apply \ref{lem:iterate} to the step operator and transport along
\ref{lem:reach-walk}.
\end{proof}

\begin{theorem}[Theorem B, ratification kit gap 5.3]
\label{thm:B}
\lean{Reconstruction.Pruning.isEvPeriodic_safe}
\leanok
\uses{lem:unsafe-biunion, lem:evper-tr, def:evperiodic, def:unsafe-safe}
For every relation $\rho$ on a finite alphabet, $\mathrm{Safe}(\rho)$ is
eventually periodic: there are a threshold $S$ and a period $P > 0$ with
$L \in \mathrm{Safe}(\rho)$ iff $L+P \in \mathrm{Safe}(\rho)$, for every
$L \geq S$. Membership is therefore decided by a finite table, the certificate
an outsider can run on their own examples.
\end{theorem}

\begin{proof}
\leanok
\uses{lem:unsafe-biunion, lem:evper-tr}
$\mathrm{Unsafe}$ is a finite union of eventually periodic sets by
\ref{lem:unsafe-biunion} and \ref{lem:evper-tr}, hence eventually periodic with
threshold the max of the thresholds and period the lcm of the periods;
$\mathrm{Safe}$ is its complement among the positives.
\end{proof}

\chapter{The observational extension theorem and what blocks it}

This chapter exists to make one open goal legible. It is the only real
\texttt{sorry} in the development, and the reason it is open is an
\emph{infrastructure} gap rather than a mathematical one: the argument needs a
Yosida--Hewitt-style decomposition of finitely additive measures, and Mathlib
has no charge theory at all --- no purely finitely additive measures, no
decomposition, nothing to cite.

Nodes below carrying no Lean name are stated but not yet formalized. They are
the work items, in dependency order.

\section{The Stone route}

\begin{definition}[Stone space of a query system]
\label{def:stone-space}
\lean{stoneSpace}
\leanok
$\mathrm{St}(S)$ is the space of ultrafilters on $\Omega$, i.e. the
Stone--\v{C}ech compactification of $\Omega$ regarded as a discrete set. It is
compact, Hausdorff and totally disconnected.
\end{definition}

\begin{definition}[Stone embedding]
\label{def:stone-embedding}
\lean{stoneEmbedding}
\leanok
\uses{def:stone-space}
$\Omega \to \mathrm{St}(S)$, $\omega \mapsto$ the principal ultrafilter at
$\omega$. Its range is exactly the principal ultrafilters, and it has dense
range.
\end{definition}

\begin{definition}[Cylinder clopens]
\label{def:stone-clopens}
\lean{stoneClopens}
\leanok
\uses{def:stone-space}
The sets $\{u : E \in u\}$ for $E$ a cylinder event. These are clopen, form a
set semiring, and generate the $\sigma$-algebra used on $\mathrm{St}(S)$.
\end{definition}

\begin{definition}[Normalized compatible contents]
\label{def:ncc}
\lean{QuerySystem.NormalizedCompatibleContents}
\leanok
A family $\nu_i$ of finitely additive contents on the outcome spaces,
compatible under refinement, each of total mass $1$. Finitely additive only:
this is the whole point, since assuming $\sigma$-additivity would beg the
question.
\end{definition}

\begin{definition}[Collective exhaustion]
\label{def:collectively-exhaustive}
\lean{QuerySystem.CollectivelyExhaustive}
\leanok
\uses{def:ncc}
For every index $i$ and every decreasing sequence of measurable events
shrinking to $\emptyset$, some finer level $j \geq i$ witnesses convergence:
the charge of the pulled-back sequence at $j$ tends to $0$. This is the
valuation-layer condition; index-layer conditions provably do not suffice.
\end{definition}

\begin{theorem}[SP1]
\label{thm:sp1-iff}
\lean{QuerySystem.sp1_iff}
\leanok
\uses{def:collectively-exhaustive, def:ncc}
Collective exhaustion holds if and only if every level's content extends to a
genuine measure on that level's outcome space.
\end{theorem}

\begin{proof}
\leanok
Proved. This is the input the blocked steps below get to use: collective
exhaustion is available in the strong form ``each $\nu_i$ is countably
additive''.
\end{proof}

\section{The chain, and where it breaks}

\begin{theorem}[Step 1: the Stone measure exists and extends the content]
\label{thm:stone-measure-exists}
\lean{stone_measure_exists}
\leanok
\uses{def:stone-clopens, def:ncc}
There is a probability measure $\hat P$ on $\mathrm{St}(S)$ with
$\hat P(E) = m(E)$ for every cylinder clopen $E$, where $m$ is the additive
content transferred from $\nu$ along the Stone embedding.
\end{theorem}

\begin{proof}
\leanok
\uses{def:stone-clopens}
Transfer $\nu$ to an additive content on the cylinder clopens; prove
$\sigma$-subadditivity using compactness of the Stone space (every countable
cover of a clopen has a finite subcover); apply Carath\'eodory.

\emph{Record.} Until 2026-08-22 the conclusion of this theorem was only
``$\exists \hat P$, $\hat P$ is a probability measure'', which never mentions
$\nu$ --- and was therefore satisfied by a Dirac measure at any ultrafilter,
with the hypotheses unused. The extension the proof builds was discarded at
the statement boundary. The agreement clause above is what Step 3 actually
needs.
\end{proof}

\begin{lemma}[Step 2: the purely finitely additive part vanishes]
\label{lem:stone-support-principal}
\uses{thm:stone-measure-exists, thm:sp1-iff, def:stone-embedding}
Under collective exhaustion, $\hat P$ is concentrated on the range of the
Stone embedding: $\hat P(\mathrm{St}(S) \setminus \mathrm{ran}\,e) = 0$, where
$\mathrm{ran}\,e$ is the set of principal ultrafilters.

\textbf{[NOT FORMALIZED --- this is the Yosida--Hewitt step.]}
\end{lemma}

\begin{proof}
This is Yosida--Hewitt specialised to a discrete carrier. A bounded charge on
$\Omega$ splits as $\mu = \mu_c + \mu_p$ with $\mu_c$ countably additive and
$\mu_p$ purely finitely additive; under Stone duality $\mu_c$ is the part
carried by the principal ultrafilters and $\mu_p$ the part carried by the free
ones. Collective exhaustion, via \ref{thm:sp1-iff}, makes every level
countably additive, which is what forces $\mu_p = 0$.

The general Yosida--Hewitt theorem over an arbitrary Boolean algebra is
\emph{not} needed. On a discrete carrier the decomposition is concrete: the
countably additive part is the atomic part $A \mapsto \sum_{x \in A}\mu(\{x\})$
and the purely finitely additive part is what vanishes on every singleton.
Mathlib has neither, so both must be built.
\end{proof}

\begin{lemma}[Step 3: pushforward to $\Omega$]
\label{lem:stone-pushforward}
\uses{lem:stone-support-principal, def:stone-embedding}
Since the Stone embedding is injective and $\hat P$ is concentrated on its
range, the pushforward along its partial inverse is a probability measure
$\mu$ on $\Omega$.

\textbf{[NOT FORMALIZED.]}
\end{lemma}

\begin{lemma}[Step 4: the marginals are the given ones]
\label{lem:stone-marginals}
\uses{lem:stone-pushforward, thm:stone-measure-exists}
$\mu(\mathrm{Cyl}_i A) = \nu_i(A)$ for every query $i$ and measurable $A$,
by the agreement clause of \ref{thm:stone-measure-exists} and the fact that
cylinder clopens pull back to cylinders.

\textbf{[NOT FORMALIZED.]}
\end{lemma}

\begin{theorem}[Observational extension]
\label{thm:stone-observational-extension}
\lean{stone_observational_extension}
\leanok
\uses{lem:stone-marginals, lem:stone-pushforward, thm:observational-determination}
For an upper-directed query system with surjective evaluations and a
collectively exhaustive normalized compatible family $\nu$, there is a unique
probability measure $\mu$ on $\Omega$ with
$\mu(\mathrm{Cyl}_i A) = \nu_i(A)$ for every query $i$ and measurable $A$.
\end{theorem}

\begin{proof}
Existence is Steps 1--4; uniqueness is
\ref{thm:observational-determination}. The statement is formalized and the
proof is not: it is the one open goal in the development, and it is open at
Step 2.
\end{proof}

\section{Why this is open: the hypothesis gap, precisely}

The development already contains a \emph{proved}, axiom-free extension theorem.
It is worth stating exactly how it differs, because the difference is the whole
open problem.

\begin{theorem}[Observational extension, sequential form --- PROVED]
\label{thm:observational-extension-seq}
\lean{QuerySystem.observational_extension}
\leanok
\uses{def:compatible-marginals}
If $S$ is \emph{sequentially} upper-directed with surjective evaluations and
$\nu_i$ is a compatible family of probability \emph{measures}, then there is a
unique probability measure $P$ on $\Omega$ with
$\mathrm{map}(\mathrm{eval}_i)\,P = \nu_i$ for every $i$.
\end{theorem}

\begin{proof}
\leanok
Carath\'eodory on the cylinder semiring directly --- no Stone space. Build the
cylinder content, prove $\sigma$-subadditivity, extend, recover marginals.
\end{proof}

\begin{lemma}[$\sigma$-subadditivity from a common upper bound --- PROVED]
\label{lem:cylgen-sigma-subadd}
\lean{QuerySystem.cylGen_addContent_isSigmaSubadditive}
\leanok
\uses{thm:observational-extension-seq}
Under sequential upper-directedness the cylinder content is
$\sigma$-subadditive.
\end{lemma}

\begin{proof}
\leanok
The hypothesis is used exactly once, and this is the crux of the whole chapter:
given a countable cover, it produces a \emph{single} index $m$ above every
index appearing in it, after which $\sigma$-subadditivity of the one measure
$\nu_m$ finishes the argument. Every other step is bookkeeping.
\end{proof}

\begin{lemma}[THE OPEN PROBLEM, sharply stated]
\label{lem:cylgen-sigma-subadd-ce}
\uses{lem:cylgen-sigma-subadd, def:collectively-exhaustive, thm:sp1-iff}
Under \emph{ordinary} upper-directedness together with collective exhaustion,
the cylinder content is $\sigma$-subadditive.

\textbf{[OPEN. This, not Yosida--Hewitt, is the real frontier.]}
\end{lemma}

\begin{proof}
With \ref{lem:cylgen-sigma-subadd-ce}, Carath\'eodory finishes on $\Omega$
directly and \ref{thm:stone-observational-extension} follows with no Stone
space and no charge theory at all.

Why it is hard, and why the Stone route exists. Ordinary upper-directedness
bounds \emph{pairs} of indices, not countable families, so no single $\nu_m$
dominates a countable cover. That is a real obstruction, not a gap in
ingenuity: the corpus already exhibits
\texttt{fcContent\_not\_sigmaSubadditive}, a content whose index-layer
conditions hold and which is not $\sigma$-subadditive. Collective exhaustion is
the valuation-layer replacement, and \ref{thm:sp1-iff} converts it into
countable additivity at each single level --- but a countable cover of one
cylinder can involve unboundedly many levels, and CE as stated speaks about a
decreasing sequence at one level at a time. Bridging that is the work.

The Stone route buys the same conclusion differently: clopens of
$\mathrm{St}(S)$ are compact, so a countable cover admits a finite subcover and
no common level is ever needed. The price is that the resulting measure lives
on $\mathrm{St}(S)$, and returning it to $\Omega$ is what needs
Yosida--Hewitt (\ref{lem:stone-support-principal}).

So there are two ways to close the same goal, and they cost differently:
prove this lemma (pure measure theory, all pieces present), or build charge
theory and take the Stone route back (a missing Mathlib area, plus the
question of whether the range of the Stone embedding is even measurable in the
$\sigma$-algebra generated by the cylinder clopens --- for uncountable
$\Omega$ that is doubtful, since a measurable set there depends on only
countably many cylinder clopens).
\end{proof}

\begin{lemma}[The cylinder family is a set ring]
\label{lem:cylgen-setring}
\lean{QuerySystem.isSetRing_CylGen}
\leanok
\uses{def:cylinder, def:lower-directed}
Under upper-directedness the cylinder events are closed under union and
difference, not merely under the semiring operations.
\end{lemma}

\begin{proof}
\leanok
Lift both cylinders to a common level with \texttt{cyl\_refine}; union and
difference are then computed inside one outcome space and stay cylinders.

This is the first brick of \ref{lem:cylgen-sigma-subadd-ce}: the passage from
continuity at $\emptyset$ to $\sigma$-additivity
(\texttt{addContent\_iUnion\_eq\_sum\_of\_tendsto\_zero}) requires a
\emph{ring}, and only the semiring structure had been recorded.
\end{proof}

\section{Is the open goal true? A concern that must be settled first}

\begin{lemma}[Collective exhaustion is per-level $\sigma$-additivity]
\label{lem:ce-levelwise}
\lean{QuerySystem.ce_iff_levelwise_continuity}
\leanok
\uses{def:collectively-exhaustive, thm:sp1-iff}
$\nu$ is collectively exhaustive if and only if every $\nu_i$ is continuous at
$\emptyset$, i.e. $\sigma$-additive.
\end{lemma}

\begin{proof}
\leanok
The definition reads as though the finer level $j$ carries content --- ``some
$j \geq i$ witnesses the convergence''. It does not. Compatibility gives
$\nu_i(A) = \nu_j(\pi^{-1}A)$ outright, so the quantity tested at $j$ is
\emph{equal} to the one at $i$ and the existential is cosmetic.
\end{proof}

\begin{remark}[The truth of \ref{thm:stone-observational-extension} is open,
not just its proof]
\label{rmk:kolmogorov-concern}
\uses{lem:ce-levelwise, thm:observational-extension-seq}
Substituting \ref{lem:ce-levelwise} into
\ref{thm:stone-observational-extension}, its hypotheses read: compatible
marginals, each $\sigma$-additive, plus upper-directedness and surjective
evaluations. Its conclusion is the global extension. That is the Kolmogorov
extension theorem \emph{with no regularity or compactness hypothesis of any
kind} --- the outcome spaces carry only a $\sigma$-algebra, no topology.

Classically this is not sufficient: Andersen--Jessen (1948) give inverse
sequences of probability spaces with compatible $\sigma$-additive marginals
admitting no limit measure, in exactly this measurable-space-only setting. So
either \texttt{EvalSurjective} rules out every such system --- in which case it
is doing far more work than its name suggests, and identifying why is the key
idea of the missing proof --- or the statement is false and needs a further
hypothesis.

This is a concern, not a refutation: no counterexample has been constructed
here, and surjectivity of every $\mathrm{eval}_i$ is a strong condition that
the usual Andersen--Jessen constructions (thin or empty limits) violate.

\textbf{The decisive experiment.} Index by the FINITE subsets of $\mathbb{N}$
ordered by inclusion. That is upper-directed, and it is \emph{not} sequentially
upper-directed --- a countable family of finite sets has no finite upper bound
--- so it sits exactly in the gap between
\ref{thm:observational-extension-seq}, which is proved, and
\ref{thm:stone-observational-extension}, which is not. Realising an
Andersen--Jessen system over that index set with surjective evaluations would
settle the question negatively; showing it cannot be done would isolate the
mechanism the proof needs.

Note what this explains about the corpus: \ref{thm:observational-extension-seq}
was proved with \emph{sequential} upper-directedness, the hypothesis that lets
one level dominate an entire countable cover. That is the standard way of
avoiding a regularity assumption. The open theorem drops it and keeps only
collective exhaustion --- and collective exhaustion, by \ref{lem:ce-levelwise},
is weaker than it looks.
\end{remark}

\begin{remark}[Resolved: the statement is false]
\label{rmk:kolmogorov-refuted}
\uses{rmk:kolmogorov-concern, lem:ce-levelwise}
The concern of \ref{rmk:kolmogorov-concern} is settled, negatively.
\ref{thm:stone-observational-extension} is \textbf{false as stated}. The
Andersen--Jessen construction (Sparre Andersen and Jessen 1948; exposition in
K.~C.~Border, \emph{Kolmogorov Extension Problem}, Prop.~17, itself following
Halmos \emph{Measure Theory} pp.~68--70) instantiates every hypothesis and
refutes the conclusion.

\textbf{The construction.} Take $M_k$ decreasing with $X_k = M_k \cap [0,1]$
\emph{thick} in $[0,1]$ --- full outer measure --- and $X_k \downarrow
\emptyset$. Thickness makes $\Sigma_k = \{E \cap X_k : E$ Lebesgue
measurable$\}$ with $\lambda_k(E \cap X_k) = \lambda(E)$ a well defined
\emph{$\sigma$-additive} probability space. Index by $\mathbb{N}$ under $\leq$;
put $\mathrm{Outcome}_n = X_0 \times \cdots \times X_n$ with the projections as
bonding maps; then $\Omega = \prod_k X_k$. Define $P_n$ by pushing $\lambda_n$
forward along the diagonal $x \mapsto (x,\dots,x)$. These are Kolmogorov
consistent.

\textbf{The refutation.} The diagonal cylinders
$D_n = \{(x,\dots,x) \times \prod_{k>n} X_k : x \in X_n\}$ have
$P(D_n) = \lambda_n(X_n) = 1$ for every $n$, while $\bigcap_n D_n = \emptyset$
because $X_k \downarrow \emptyset$. No measure can give every term of a
decreasing sequence mass $1$ and the limit mass $0$.

\textbf{Why every hypothesis holds.} $\mathbb{N}$ is a chain, so
upper-directed --- and \emph{not} sequentially upper-directed, since
$(0,1,2,\dots)$ has no upper bound: exactly the gap between this theorem and
\ref{thm:observational-extension-seq}. The bonding maps and every
$\mathrm{eval}_n$ are projections onto initial segments, hence surjective
(each $X_k$ is thick, so nonempty), giving \texttt{EvalSurjective}. Each $P_n$
is a genuine measure, so collective exhaustion holds by \ref{lem:ce-levelwise}.
The variance matches: \texttt{QuerySystem} has $\pi :
\mathrm{Outcome}_j \to \mathrm{Outcome}_i$ for $i \leq j$, the inverse-system
direction.

\textbf{What fails, and what it costs.} Every level is $\sigma$-additive; the
failure lives entirely in the passage between levels, which a per-level
condition cannot see. So the claim that collective exhaustion characterises
\emph{$\sigma$-additive extensibility} is correct only for the per-level
reading proved in \ref{thm:sp1-iff}; read as global extensibility it is false,
and that reading appears in the prose around the definition. This is a defect in
a central definition of the paper, not in a Lean proof.

Repairs, in increasing order of ambition: assume sequential upper-directedness,
in which case \ref{thm:observational-extension-seq} already proves it; or
assume a compact class / perfect-measure condition on the marginals
(Marczewski), the classical hypothesis, and port that theorem instead.

\textbf{Status.} Verified on paper against the Lean hypotheses, hypothesis by
hypothesis. NOT yet formalized: doing so means constructing thick sets in Lean,
which needs a non-measurable-set construction and is a real project.
\end{remark}

\begin{remark}[Repaired, and what the repair costs]
\label{rmk:extension-repaired}
\uses{rmk:kolmogorov-refuted, thm:observational-extension-seq, thm:sp1-iff}
\leanok
\ref{thm:stone-observational-extension} is now \emph{proved}, with
\texttt{UpperDirected} strengthened to \texttt{SequentiallyUpperDirected}.
Receipts \texttt{[propext, Classical.choice, Quot.sound]}. The development is
\texttt{sorry}-free.

The proof does not go through the Stone space and needs no charge theory:
\ref{thm:sp1-iff} converts collective exhaustion into a genuine measure at each
level, compatibility transfers because each $\mu_i$ agrees with $\nu_i$ on every
measurable set, and \ref{thm:observational-extension-seq} finishes on $\Omega$
directly.

\textbf{What the hypothesis costs.} Sequential upper-directedness says the
completed query $\bigvee_n Q_n$ is itself a query --- something no finite
observer realises. It is natural exactly when a query is a $\sigma$-algebra of
resolvable events and the family is closed under countable joins, and it is not
a new idealisation: making each query a $\sigma$-algebra already assumes
completed countable operations, and $\bigcup_n E_n$ is as unphysical as
$\bigvee_n Q_n$. What it excludes is the finite-dimensional marginal family ---
precisely where \ref{rmk:kolmogorov-refuted} lives.

\textbf{Three positions, and the theorem for each.} Neither counterexample is
avoidable, so strength must be paid at one layer or the other:
\begin{itemize}
\item completed countable queries admitted:
  \ref{thm:observational-extension-seq} and
  \ref{thm:stone-observational-extension} --- proved;
\item only finite observers, contents rather than measures:
  \ref{thm:stone-measure-exists} --- proved, no directedness beyond
  upper, no CE. Free ultrafilters carrying mass are then not a pathology but
  the trace of observers who never complete their observations;
\item neither: false, by \ref{rmk:kolmogorov-refuted}.
\end{itemize}
The index-layer condition alone is also insufficient
(\texttt{counterexampleQS\_no\_finitary\_condition}), so a valuation-layer
condition and an index-layer condition are both required, with a counterexample
in each direction.
\end{remark}

\section{The obstruction, formalized}

\begin{definition}[Escaping tower]
\label{def:escaping-tower}
\lean{QuerySystem.EscapingTower}
\leanok
\uses{def:cylinder, def:ncc}
A decreasing sequence of cylinder events, each of full charge, with empty
intersection. Every finite stage carries all the mass; the limit carries none.
In Andersen--Jessen these are the diagonal cylinders over a decreasing sequence
of thick sets.
\end{definition}

\begin{theorem}[No extension over an escaping tower]
\label{thm:no-extension-tower}
\lean{QuerySystem.not_exists_extension_of_escapingTower}
\leanok
\uses{def:escaping-tower}
A query system carrying an escaping tower admits no extension of its charges to
a probability measure on $\Omega$.
\end{theorem}

\begin{proof}
\leanok
Continuity from above: each stage has measure $1$, the intersection has measure
$0$, and a constant sequence cannot converge to both.

Note the hypotheses: there are none on $S$. No directedness, no surjectivity,
no collective exhaustion. The tower alone is fatal --- which is why
strengthening the index layer was the only available repair for
\ref{thm:stone-observational-extension}. One cannot patch the conclusion or the
valuation layer around a tower; one can only assume an index layer rich enough
that no tower exists.
\end{proof}

\begin{remark}[What remains unformalized]
\label{rmk:tower-remaining}
\uses{thm:no-extension-tower, rmk:kolmogorov-refuted}
\ref{thm:no-extension-tower} is the framework-specific half of
\ref{rmk:kolmogorov-refuted} and is now machine-checked. What is not formalized
is the Andersen--Jessen system that supplies the tower: it needs a decreasing
sequence of \emph{thick} subsets of $[0,1]$ with empty intersection, built as
$X_k = (V + B_k) \cap [0,1]$ with $V$ a transversal of
$\mathbb{R}/(\mathbb{Z}+\alpha\mathbb{Z})$ and
$B_k = \{n+m\alpha : |n| \geq k,\ n \text{ even}\}$. Thickness follows from the
measure-theoretic Steinhaus theorem, which Mathlib has as
\texttt{div\_mem\_nhds\_one\_of\_haar\_pos}; the transversal is
\texttt{Quotient.out'}. Mathlib has neither thick sets nor a Vitali-style
transversal, so this is a real construction project.

Until it is done, the claim that the original statement was false rests on a
hand check against the source, not on the kernel. The mechanism does not.
\end{remark}

\begin{theorem}[The obstruction is not vacuous]
\label{thm:tower-inhabited}
\lean{QuerySystem.counterexampleTower}
\leanok
\uses{def:escaping-tower}
The finite--cofinite query system carries an escaping tower: with
$E_n = \mathbb{Q} \setminus \{q_0,\dots,q_n\}$, each $E_n$ is cofinite hence in
the hyperfilter hence of content $1$, while $\bigcap_n E_n = \emptyset$.
\end{theorem}

\begin{proof}
\leanok
Direct. The point is not the construction --- the same sequence already
appears in \texttt{counterexampleNCC\_not\_collectivelyExhaustive} --- but that
\ref{def:escaping-tower} is inhabited. A structure nobody can instantiate would
make \ref{thm:no-extension-tower} empty content, and this session has already
turned up one theorem whose statement was satisfiable by a Dirac measure.
\end{proof}

\begin{corollary}[The finite--cofinite system admits no extension]
\label{cor:cex-no-extension}
\lean{QuerySystem.counterexample_no_extension}
\leanok
\uses{thm:tower-inhabited, thm:no-extension-tower}
Recovered from the abstract obstruction rather than re-argued, which is the
point: the tower is the reusable content of that counterexample.

Note what it does \emph{not} show. This system fails collective exhaustion
(\texttt{counterexampleNCC\_not\_collectivelyExhaustive}), so it does not touch
\ref{thm:stone-observational-extension}. Refuting that needs a tower in a system
where every level \emph{is} $\sigma$-additive, which is what forces the thick
sets of \ref{rmk:tower-remaining} and why Andersen--Jessen is harder than this.
\end{corollary}

\chapter{Andersen--Jessen: the thick tower}

The counterexample that refutes observational extension under plain
upper-directedness needs one object: a decreasing family of subsets of $[0,1]$,
each of full outer measure, whose intersection is empty. Every level carries all
the measure, the limit carries none. That is the non-regularity phenomenon
itself, and it has no analogue over Polish spaces.

Built here by a shorter route than the source (Border, following
Halmos pp.~68--70), which uses a parity split. Parity caps out: a set with all
even differences lies in a coset of $2\mathbb{Z}+\alpha\mathbb{Z}$, and there are
only two such cosets, so it can never separate more than two pieces. Finiteness
has no such cap. See \texttt{notes/andersen\_jessen\_brief.md}.

\begin{definition}[The group and its parts]
\label{def:aj-A}
\lean{AndersenJessen.A}
\leanok
$A = \mathbb{Z} + \alpha\mathbb{Z}$ for $\alpha$ irrational, with $B_k$ its
elements of even integer coordinate and $|n| \geq k$, and $C_k$ the odd ones.
\end{definition}

\begin{lemma}[Distinctness --- the keystone]
\label{lem:aj-repr}
\lean{AndersenJessen.repr_unique}
\leanok
\uses{def:aj-A}
For $\alpha$ irrational, $n + m\alpha$ determines $n$ and $m$.
\end{lemma}

\begin{proof}
\leanok
Otherwise $\alpha = (n'-n)/(m-m')$ is rational. Everything about the tower rests
on this: it is what gives each element of $A$ a well-defined integer coordinate,
hence what makes $\bigcap_k B_k = \emptyset$ true.
\end{proof}

\begin{lemma}[Density]
\label{lem:aj-dense}
\lean{AndersenJessen.Afull_dense}
\leanok
\uses{def:aj-A}
$A$ is dense in $\mathbb{R}$.
\end{lemma}

\begin{proof}
\leanok
$2\mathbb{Z}+\alpha\mathbb{Z}$ is a subgroup, so by
\texttt{AddSubgroup.dense\_or\_cyclic} it is dense or cyclic; a generator $g$
would give $2 = pg$ and $\alpha = qg$, making $\alpha = 2q/p$ rational. Density
of $A$ follows by monotonicity. The source instead proves density of $B_k$ and
$C_k$ for every $k$ by an equidistribution argument; that is not needed.
\end{proof}

\begin{lemma}[The transversal]
\label{lem:aj-V}
\lean{AndersenJessen.V_unique}
\leanok
\uses{def:aj-A}
$V$, a set of coset representatives for $\mathbb{R}/A$, has the only property the
argument uses: two representatives congruent mod $A$ are equal.
\end{lemma}

\begin{proof}
\leanok
\texttt{Quotient.out} on $\mathbb{R} \mathbin{/} A$.
\end{proof}

\begin{theorem}[Finiteness beats parity]
\label{thm:aj-finite}
\lean{AndersenJessen.V_add_finite_measurable_null}
\leanok
\uses{lem:aj-V, lem:aj-dense}
For FINITE $F \subseteq A$, the set $V + F$ contains no measurable set of
positive measure.
\end{theorem}

\begin{proof}
\leanok
\uses{lem:aj-dense}
If it did, Steinhaus
(\texttt{Measure.sub\_mem\_nhds\_zero\_of\_addHaar\_pos}) puts a neighbourhood
of $0$ inside $F' - F'$. $F - F$ is finite while $A$ is dense, so some element of
$A$ lies in that neighbourhood and outside $F - F$; the transversal then forces
it into $F - F$ after all.

This is the replacement for the source's Proposition 13, and it is what gives
thickness at EVERY level rather than only at $k = 0$.
\end{proof}

\begin{theorem}[The thick tower]
\label{thm:aj-tower}
\lean{AndersenJessen.X_thick}
\leanok
\uses{thm:aj-finite, lem:aj-repr}
With $A$ enumerated, set $X_k = V + \{a_j : j \geq k\}$. Then $X_k$ is
decreasing (\texttt{X\_antitone}), $\bigcap_k X_k = \emptyset$
(\texttt{X\_iInter}), and every $X_k$ is thick: any measurable set disjoint from
it is null.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:aj-finite}
$X_k^c = V + \{a_0,\dots,a_{k-1}\}$, a finite subset of $A$, so
\ref{thm:aj-finite} applies at every $k$. The empty intersection is free from
the enumeration: each real is $v + a_j$ for exactly one $j$.
\end{proof}

\begin{remark}[What remains]
\label{rmk:aj-remaining}
\uses{thm:aj-tower, def:escaping-tower}
The tower is the hard half. What is left to reach an \ref{def:escaping-tower}
instance, and so a kernel-checked refutation: the trace measure on a thick set
(Proposition 14 --- $\lambda_X(E \cap X) = \lambda(E)$ well defined and
$\sigma$-additive, for which \ref{thm:aj-tower} is exactly the hypothesis), the
projective system over $X_0 \times \cdots \times X_n$ with the diagonal maps,
and the diagonal cylinders themselves.
\end{remark}

\begin{theorem}[Lemma NG, refutation half]
\label{thm:ng-refute}
\lean{Reconstruction.Pruning.NG_refutation}
\leanok
\uses{lem:ng, lem:ng-orbit}
For the directed 4-cycle at $L = 2$ and the non-generator $r = 2$
($\gcd(2,4) = 2$): $\mathrm{TR}_4^{(2)}(2)$ holds, $\mathrm{LISC}_2(2)$ holds,
and $\mathrm{LISC}_4(2)$ \emph{fails}.
\end{theorem}

\begin{proof}
\leanok
\uses{lem:ng-orbit}
The relation is a FUNCTION, so every vertex of the layered ring has out-degree
one and a walk is determined by its start; it returns to its starting state
after four steps. A winding-4 simple cycle needs eight pairwise distinct
positions, but positions $0$ and $4$ share a layer and a state. The
winding-2 cycle and the rotation-2 slot walk $\tau_i(j) = j + i$ are exhibited
directly.

So a rotation witness at a non-generator certifies winding $k/d = 2$, not
winding $k$ --- Theorem P's equivalence genuinely needs $r$ to generate, and
\ref{lem:ng-orbit} is sharp. This previously rested on the oracle rot-check;
it is now in the kernel.
\end{proof}

\chapter{Query Systems and Observational Foundations}

This formalization develops query systems: preordered families of measurable spaces with compatible refinement maps.

\section{Core Definitions}

\begin{definition}
\label{def:query}
\lean{Query}
\leanok
A \emph{query} is a measurable space representing the possible outcomes of asking a single question or making a single observation.
\end{definition}

\begin{definition}
\label{def:refine}
\lean{Refine}
\leanok
A \emph{refinement} between queries is a measurable function showing how outcomes of one query determine outcomes of another.
\end{definition}

\begin{definition}
\label{def:query-system}
\lean{QuerySystem}
\leanok
\uses{def:query, def:refine}
A \emph{query system} is a preordered family of queries with coherent refinement maps. It consists of:
\begin{itemize}
\item An index type $\iota$
\item A family of queries $q : \iota \to \text{Query}$
\item A preorder relation $\le$ on indices
\item Refinement maps $\pi : \forall \{i\ j\}, i \le j \to \text{Refine}(q_i, q_j)$
\end{itemize}
subject to functoriality conditions.
\end{definition}

\begin{definition}
\label{def:omega}
\lean{QuerySystem.Omega}
\leanok
The \emph{projective limit} $\Omega$ is the space of coherent families of query outcomes.
An element $\omega : \Omega$ assigns to each query $i$ an outcome $\omega(i)$, subject to the coherence condition:
whenever $i \le j$, we have $\omega(j) = \pi_{ij}(\omega(i))$.
\end{definition}

\begin{definition}
\label{def:cylinder}
\lean{QuerySystem.Cyl}
\leanok
A \emph{cylinder set} $\text{Cyl}(i, A)$ is the set of realizations where query $i$ yields an outcome in $A$.
These generate the observable $\sigma$-algebra.
\end{definition}

\begin{definition}
\label{def:compatible-marginals}
\lean{QuerySystem.CompatibleMarginals}
\leanok
\uses{def:query-system, def:refine}
A family $\nu$ of measures on query outcomes is \emph{compatible} if the measures commute with refinement maps:
whenever $i \le j$, the pushforward of $\nu_i$ along $\pi_{ij}$ equals $\nu_j$.
\end{definition}

\begin{definition}[Upper-directed]
\label{def:upper-directed}
\lean{QuerySystem.UpperDirected}
\leanok
A query system is \emph{upper-directed} if any two queries have a common
refinement: $\forall i\,j, \exists k, i \le k \land j \le k$.
\end{definition}

\begin{definition}[Sequentially upper-directed]
\label{def:seq-upper-directed}
\lean{QuerySystem.SequentiallyUpperDirected}
\leanok
\uses{def:upper-directed}
Every \emph{sequence} of queries has a common refinement:
$\forall u : \mathbb{N} \to \iota, \exists k, \forall n, u_n \le k$. Strictly
stronger than upper-directedness, which bounds pairs. It says the completed
query $\bigvee_n Q_n$ is itself a query.
\end{definition}

\begin{definition}[Surjective evaluations]
\label{def:eval-surjective}
\lean{QuerySystem.EvalSurjective}
\leanok
\uses{def:omega}
Every evaluation $\mathrm{eval}_i : \Omega \to \mathrm{Outcome}_i$ is
surjective: each local outcome extends to a coherent global realization. It
gives $\mathrm{Cyl}_k E_1 = \mathrm{Cyl}_k E_2 \Rightarrow E_1 = E_2$, which
the well-definedness and additivity of the cylinder content require.
\end{definition}

\subsection{Observational Determination}

\begin{lemma}
\label{lem:pi-system}
\lean{QuerySystem.isPiSystem_CylGen}
\leanok
\uses{def:cylinder, def:upper-directed}
Under upper-directedness, the cylinder family forms a $\pi$-system.
\end{lemma}

\begin{proof}
\leanok
Upper-directedness gives a common refinement of any two cylinder-generating queries; intersect the pulled-back events there.
\end{proof}

\begin{theorem}[Observational Determination]
\label{thm:observational-determination}
\lean{QuerySystem.observational_determination}
\leanok
\uses{lem:pi-system, def:omega}
Two (finite) measures on the projective limit $\Omega$ are equal if and only if
they have equal marginals on all queries, i.e.\ equal pushforwards along every
$\mathrm{eval}_i$. No compatibility hypothesis is consumed.
\end{theorem}

\begin{proof}
\leanok
Apply the $\pi$-$\lambda$ theorem to the cylinder $\sigma$-algebra. The cylinders form a $\pi$-system by \ref{lem:pi-system}, and the hypothesis gives agreement on all cylinders.
\end{proof}

\label{lem:premu-well-defined}
\lean{QuerySystem.preμ_wellDefined}
\leanok
\uses{def:compatible-marginals, def:upper-directed}

\chapter{The observational extension theorem, and the obstruction}

From a compatible family of finitely additive charges to a single probability
measure on $\Omega$. The extension theorem holds under \emph{sequential}
upper-directedness (\ref{thm:observational-extension-seq},
\ref{thm:stone-observational-extension}); under plain upper-directedness it is
false, and \S\ref{sec:sequentiality} exhibits the counterexample. The
mechanism of that failure --- an escaping tower --- is machine-checked
(\ref{thm:no-extension-tower}). On the Stone space there is a genuinely
finitely additive theorem, needing no collective exhaustion and no
directedness beyond upper (\ref{thm:stone-measure-exists}).

\section{Charges, collective exhaustion, and the Stone space}

\begin{definition}[Stone space of a query system]
\label{def:stone-space}
\lean{stoneSpace}
\leanok
$\mathrm{St}(S)$ is the space of ultrafilters on $\Omega$, i.e. the
Stone--\v{C}ech compactification of $\Omega$ regarded as a discrete set. It is
compact, Hausdorff and totally disconnected.
\end{definition}

\begin{definition}[Cylinder clopens]
\label{def:stone-clopens}
\lean{stoneClopens}
\leanok
\uses{def:stone-space}
The sets $\{u : E \in u\}$ for $E$ a cylinder event. These are clopen, form a
set semiring, and generate the $\sigma$-algebra used on $\mathrm{St}(S)$.
\end{definition}

\begin{definition}[Normalized compatible contents]
\label{def:ncc}
\lean{QuerySystem.NormalizedCompatibleContents}
\leanok
A family $\nu_i$ of finitely additive contents on the outcome spaces,
compatible under refinement, each of total mass $1$. Finitely additive only.
\end{definition}

\begin{definition}[Collective exhaustion]
\label{def:collectively-exhaustive}
\lean{QuerySystem.CollectivelyExhaustive}
\leanok
For every index $i$ and every decreasing sequence of measurable events
shrinking to $\emptyset$, some finer level $j \geq i$ witnesses convergence:
the charge of the pulled-back sequence at $j$ tends to $0$. Index-layer
conditions do not suffice
(\texttt{counterexampleQS\_no\_finitary\_condition}).
\end{definition}

\begin{lemma}[Collective exhaustion is per-level $\sigma$-additivity]
\label{lem:ce-levelwise}
\lean{QuerySystem.ce_iff_levelwise_continuity}
\leanok
\uses{def:collectively-exhaustive, def:ncc}
$\nu$ is collectively exhaustive if and only if every $\nu_i$ is continuous at
$\emptyset$, i.e. $\sigma$-additive.
\end{lemma}

\begin{proof}
\leanok
Compatibility gives $\nu_i(A) = \nu_j(\pi^{-1}A)$ outright, so the quantity
tested at the finer level $j$ is equal to the one at $i$ and the existential in
\ref{def:collectively-exhaustive} is cosmetic.
\end{proof}

\begin{theorem}[SP1]
\label{thm:sp1-iff}
\lean{QuerySystem.sp1_iff}
\leanok
\uses{def:collectively-exhaustive, def:ncc}
Collective exhaustion holds if and only if every level's content extends to a
genuine measure on that level's outcome space.
\end{theorem}

\begin{proof}
\leanok
By \ref{lem:ce-levelwise} collective exhaustion is continuity at $\emptyset$ of
each $\nu_i$, which for a content on a $\sigma$-algebra is countable
additivity.
\end{proof}

\section{The extension theorem}

\begin{theorem}[Observational extension, sequential form]
\label{thm:observational-extension-seq}
\lean{QuerySystem.observational_extension}
\leanok
\uses{lem:cylgen-sigma-subadd, lem:premu-well-defined, thm:observational-determination, def:compatible-marginals, def:seq-upper-directed, def:eval-surjective}
If $S$ is \emph{sequentially} upper-directed with surjective evaluations and
$\nu_i$ is a compatible family of probability \emph{measures}, then there is a
unique probability measure $P$ on $\Omega$ with
$\mathrm{map}(\mathrm{eval}_i)\,P = \nu_i$ for every $i$.
\end{theorem}

\begin{proof}
\leanok
\uses{lem:cylgen-sigma-subadd, lem:premu-well-defined, thm:observational-determination}
Carath\'eodory on the cylinder semiring. Build the cylinder content --- its
value on a cylinder is \texttt{pre}$\mu$ at a singleton presentation, so
\ref{lem:premu-well-defined} is what makes the content well defined --- prove
$\sigma$-subadditivity (\ref{lem:cylgen-sigma-subadd}), extend, recover
marginals; uniqueness is \ref{thm:observational-determination}.
\end{proof}

\begin{lemma}[$\sigma$-subadditivity from a common upper bound]
\label{lem:cylgen-sigma-subadd}
\lean{QuerySystem.cylGen_addContent_isSigmaSubadditive}
\leanok
\uses{def:cylinder, def:compatible-marginals, def:seq-upper-directed, def:eval-surjective}
Under sequential upper-directedness (with surjective evaluations and
compatible marginal measures) the cylinder content is $\sigma$-subadditive.
\end{lemma}

\begin{proof}
\leanok
Sequentiality is used exactly once: given a countable cover, it produces a
\emph{single} index $m$ above every index appearing in it, after which
$\sigma$-subadditivity of the one measure $\nu_m$ finishes the argument.
Ordinary upper-directedness bounds \emph{pairs} of indices, not countable
families, so no single $\nu_m$ dominates a countable cover; with ordinary
upper-directedness plus collective exhaustion in place of sequentiality this
lemma is \textbf{false}, by the Andersen--Jessen system of
\S\ref{sec:sequentiality}.
\end{proof}

\begin{theorem}[Observational extension from charges]
\label{thm:stone-observational-extension}
\lean{stone_observational_extension}
\leanok
\uses{thm:sp1-iff, thm:observational-extension-seq, def:ncc, def:collectively-exhaustive}
For a \emph{sequentially} upper-directed query system with surjective
evaluations and a collectively exhaustive normalized compatible family
$\nu$ of finitely additive charges, there is a unique
probability measure $\mu$ on $\Omega$ with
$\mu(\mathrm{Cyl}_i A) = \nu_i(A)$ for every query $i$ and measurable $A$.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:sp1-iff, thm:observational-extension-seq}
\ref{thm:sp1-iff} converts collective exhaustion into a genuine measure at each
level; compatibility transfers because each $\mu_i$ agrees with $\nu_i$ on every
measurable set; and \ref{thm:observational-extension-seq} finishes on $\Omega$,
its uniqueness clause included.
\end{proof}

\section{Sequentiality is not removable}
\label{sec:sequentiality}

Drop sequentiality to plain upper-directedness and the hypotheses of
\ref{thm:stone-observational-extension} read, via \ref{lem:ce-levelwise}:
compatible marginals, each $\sigma$-additive, plus upper-directedness and
surjective evaluations. That is Kolmogorov extension with no regularity or
compactness hypothesis of any kind --- the outcome spaces carry only a
$\sigma$-algebra --- and it is false. The Andersen--Jessen construction
(Sparre Andersen and Jessen 1948; exposition in K.~C.~Border,
\emph{Kolmogorov Extension Problem}, Prop.~17, following Halmos
\emph{Measure Theory} pp.~68--70) instantiates every hypothesis and refutes
the conclusion.

\emph{The construction.} Take $X_k \subseteq [0,1]$ \emph{thick} --- full
outer measure --- and decreasing with $X_k \downarrow \emptyset$. Thickness
makes $\Sigma_k = \{E \cap X_k : E \text{ Lebesgue measurable}\}$ with
$\lambda_k(E \cap X_k) = \lambda(E)$ a well defined \emph{$\sigma$-additive}
probability space. Index by $\mathbb{N}$ under $\leq$; put
$\mathrm{Outcome}_n = X_0 \times \cdots \times X_n$ with the projections as
bonding maps, so $\Omega = \prod_k X_k$; define $P_n$ by pushing $\lambda_n$
forward along the diagonal $x \mapsto (x,\dots,x)$. These are Kolmogorov
consistent.

\emph{The refutation.} The diagonal cylinders
$D_n = \{(x,\dots,x) \times \prod_{k>n} X_k : x \in X_n\}$ have
$P(D_n) = \lambda_n(X_n) = 1$ for every $n$, while $\bigcap_n D_n = \emptyset$
because $X_k \downarrow \emptyset$. No measure gives every term of a
decreasing sequence mass $1$ and the limit mass $0$.

\emph{Every hypothesis holds.} $\mathbb{N}$ is a chain, hence upper-directed,
and \emph{not} sequentially upper-directed, since $(0,1,2,\dots)$ has no upper
bound. Every $\mathrm{eval}_n$ is a projection onto an initial segment, hence
surjective (each $X_k$ is thick, so nonempty). Each $P_n$ is a genuine measure,
so collective exhaustion holds by \ref{lem:ce-levelwise}. Note $\Omega$ is the
subtype of \emph{coherent} families, not the bare product --- surjectivity onto
a projective limit can fail with every level nonempty --- but the bonding maps
are initial-segment restrictions, so coherence is automatically satisfiable and
nonemptiness of each $X_k$ suffices. The thick tower is machine-checked
(\ref{thm:aj-tower}) and the refutation is assembled at
\ref{thm:aj-no-extension}.

So the failure lives entirely in the passage between levels, where a per-level
condition cannot see it: collective exhaustion characterises $\sigma$-additive
extensibility \emph{per level} (\ref{thm:sp1-iff}) and not globally. The
alternative to assuming sequentiality is a compact-class or perfect-measure
condition on the marginals (Marczewski), the classical hypothesis.

\section{The finitely additive theorem on the Stone space}

\begin{theorem}[The Stone measure exists and extends the content]
\label{thm:stone-measure-exists}
\lean{stone_measure_exists}
\leanok
\uses{def:stone-clopens, def:ncc, def:upper-directed, def:eval-surjective}
There is a probability measure $\hat P$ on $\mathrm{St}(S)$ with
$\hat P(E) = m(E)$ for every cylinder clopen $E$, where $m$ is the additive
content transferred from $\nu$ along the Stone embedding. No collective
exhaustion and no directedness beyond upper is used.
\end{theorem}

\begin{proof}
\leanok
\uses{def:stone-clopens}
Transfer $\nu$ to an additive content on the cylinder clopens; prove
$\sigma$-subadditivity using compactness of the Stone space (every countable
cover of a clopen has a finite subcover); apply Carath\'eodory.
\end{proof}

\section{The obstruction, formalized}

\begin{definition}[Escaping tower]
\label{def:escaping-tower}
\lean{QuerySystem.EscapingTower}
\leanok
\uses{def:cylinder, def:ncc}
A decreasing sequence of cylinder events, each of full charge, with empty
intersection. In Andersen--Jessen these are the diagonal cylinders over a
decreasing sequence of thick sets.
\end{definition}

\begin{theorem}[No extension over an escaping tower]
\label{thm:no-extension-tower}
\lean{QuerySystem.not_exists_extension_of_escapingTower}
\leanok
\uses{def:escaping-tower, lem:cont-above}
A query system carrying an escaping tower admits no extension of its charges to
a probability measure on $\Omega$. No hypotheses on $S$ are used --- no
directedness, no surjectivity, no collective exhaustion.
\end{theorem}

\begin{proof}
\leanok
\uses{lem:cont-above}
The tower's cylinders are an antitone measurable sequence with empty
intersection and every stage of mass $1$, which is exactly what
\ref{lem:cont-above} forbids.
\end{proof}

\begin{theorem}[The obstruction is not vacuous]
\label{thm:tower-inhabited}
\lean{QuerySystem.counterexampleTower}
\leanok
\uses{def:escaping-tower}
The finite--cofinite query system carries an escaping tower: with
$E_n = \mathbb{Q} \setminus \{q_0,\dots,q_n\}$, each $E_n$ is cofinite hence in
the hyperfilter hence of content $1$, while $\bigcap_n E_n = \emptyset$.
\end{theorem}

\begin{proof}
\leanok
Direct.
\end{proof}

\begin{corollary}[The finite--cofinite system admits no extension]
\label{cor:cex-no-extension}
\lean{QuerySystem.counterexample_no_extension}
\leanok
\uses{thm:tower-inhabited, thm:no-extension-tower}
The finite--cofinite query system admits no extension of its charges to a

\S\ref{sec:sequentiality}.
\end{corollary}

\begin{proof}
\leanok
Instantiate \ref{thm:no-extension-tower} at the finite--cofinite escaping tower of \ref{thm:tower-inhabited}.
\end{proof}
\chapter{Andersen--Jessen: the thick tower and the diagonal layer}

The counterexample that refutes observational extension under plain
upper-directedness needs one object: a decreasing family of subsets of $[0,1]$,
each of full outer measure, whose intersection is empty.

The tower is built from \emph{finiteness} rather than the parity split of the
source (Border, following Halmos pp.~68--70). Parity caps out: a set with all
even differences lies in a coset of $2\mathbb{Z}+\alpha\mathbb{Z}$, of which
there are two, so it can never separate more than two pieces and gives
thickness only at $k = 0$. A finite set has no such cap, and
\ref{thm:aj-finite} gives thickness at every level.

\begin{definition}[The group]
\label{def:aj-A}
\lean{AndersenJessen.A, AndersenJessen.Afull}
\leanok
$A = \mathbb{Z} + \alpha\mathbb{Z}$ for $\alpha$ irrational (as a set, and as
the additive subgroup \texttt{Afull} of $\mathbb{R}$). Countable.
\end{definition}

\begin{lemma}[Density]
\label{lem:aj-dense}
\lean{AndersenJessen.Afull_dense}
\leanok
\uses{def:aj-A}
$A$ is dense in $\mathbb{R}$.
\end{lemma}

\begin{proof}
\leanok
$2\mathbb{Z}+\alpha\mathbb{Z}$ is a subgroup, so by
\texttt{AddSubgroup.dense\_or\_cyclic} it is dense or cyclic; a generator $g$
would give $2 = pg$ and $\alpha = qg$, making $\alpha = 2q/p$ rational. Density
of $A$ follows by monotonicity.
\end{proof}

\begin{lemma}[The transversal]
\label{lem:aj-V}
\lean{AndersenJessen.V_unique}
\leanok
\uses{def:aj-A}
$V$, a set of coset representatives for $\mathbb{R}/A$, has the only property the
argument uses: two representatives congruent mod $A$ are equal.
\end{lemma}

\begin{proof}
\leanok
\texttt{Quotient.out} on $\mathbb{R} \mathbin{/} A$.
\end{proof}

\begin{theorem}[Finiteness beats parity]
\label{thm:aj-finite}
\lean{AndersenJessen.V_add_finite_measurable_null}
\leanok
\uses{lem:aj-V, lem:aj-dense}
For FINITE $F \subseteq A$, the set $V + F$ contains no measurable set of
positive measure.
\end{theorem}

\begin{proof}
\leanok
\uses{lem:aj-dense}
If it did, Steinhaus
(\texttt{Measure.sub\_mem\_nhds\_zero\_of\_addHaar\_pos}) puts a neighbourhood
of $0$ inside $F' - F'$. $F - F$ is finite while $A$ is dense, so some element of
$A$ lies in that neighbourhood and outside $F - F$; the transversal then forces
it into $F - F$ after all.
\end{proof}

\begin{theorem}[The thick tower]
\label{thm:aj-tower}
\lean{AndersenJessen.X_thick, AndersenJessen.X_antitone, AndersenJessen.X_iInter}
\leanok
\uses{thm:aj-finite, lem:aj-V}
With $A$ enumerated, set $X_k = V + \{a_j : j \geq k\}$. Then $X_k$ is
decreasing (\texttt{X\_antitone}), $\bigcap_k X_k = \emptyset$
(\texttt{X\_iInter}), and every $X_k$ is thick: any measurable set disjoint from
it is null.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:aj-finite, lem:aj-V}
$X_k^c = V + \{a_0,\dots,a_{k-1}\}$, a finite subset of $A$, so
\ref{thm:aj-finite} applies at every $k$. The empty intersection needs only
the enumeration covering $A$ and the transversal property
(\ref{lem:aj-V}): each real decomposes as $v + a$ with $v$ forced, and $a$
eventually falls out of every tail.
\end{proof}

\section{The trace measure, and the refutation}

\begin{definition}[Thickness, relative to a base measure]
\label{def:thick-for}
\lean{ThickTrace.ThickFor}
\leanok
$X$ is \emph{thick for} $\mu$ when every measurable set disjoint from $X$ is
$\mu$-null. The base is arbitrary rather than Lebesgue so that the total mass is
finite: the marginals need mass $1$ and $\lambda(\mathbb{R}) = \infty$, while
$X_k \cap [0,1]$ is not thick in $\mathbb{R}$. Restricting the \emph{ambient}
measure to $\mathrm{Icc}\,0\,1$ leaves the levels exactly as
\ref{thm:aj-tower} builds them.
\end{definition}

\begin{theorem}[The trace measure on a thick set]
\label{thm:trace-measure}
\uses{def:thick-for, thm:aj-tower}
\lean{ThickTrace.traceMeasure, ThickTrace.traceMeasure_apply}
\leanok
For $X$ thick for $\mu$, the traces $\{E \cap X\}$ carry a measure with
$\lambda_X(E \cap X) = \mu(E)$ (Border, Prop.~14).
\end{theorem}

\begin{proof}
\leanok
\uses{thm:aj-tower}
The $\sigma$-algebra is the comap one, but the measure must be built by
\texttt{Measure.ofMeasurable}: \texttt{Measure.comap} returns $0$ for a thick
set with thick complement. Well-definedness is where thickness is spent --- two
measurable sets with the same trace differ by a measurable set disjoint from
$X$, hence null --- and $\sigma$-additivity follows because disjoint traces
force a.e.-disjoint representatives.
\end{proof}

\begin{theorem}[Andersen--Jessen admits no extension]
\label{thm:aj-no-extension}
\uses{thm:trace-measure, def:escaping-tower, thm:no-extension-tower, thm:diag-empty}
\lean{ASM.AJ_no_extension_unconditional}
\leanok
The Andersen--Jessen query system carries compatible $\sigma$-additive
probability marginals and admits no extension to a probability measure on
$\Omega$.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:trace-measure, thm:no-extension-tower, thm:diag-empty}
The levels carry trace measures (\ref{thm:trace-measure}) over the restricted
base, so each is a probability measure, and they are compatible along the
tower's inclusions. Pushing them along the diagonal gives Kolmogorov-consistent
marginals; the commutation $\mathrm{bond} \circ \mathrm{diag} = \mathrm{diag}
\circ \mathrm{incl}$ is definitional. The diagonal cylinders then form an
escaping tower: each carries full mass because the diagonal map lands in the
diagonal, and their intersection is empty by \ref{thm:diag-empty}.

A coherent family is not a sequence, so it is carried to one by
$\mathrm{toSeq}\,\omega : k \mapsto (\omega_k)_k$, reading off the top
coordinate at each level. Every entry lies in its own $X_k$ by the subtype
alone, while constancy needs both the coherence (to move coordinate $0$ down
from level $i$ to level $0$) and the base cylinders (to move it across level $i$
from coordinate $0$ to coordinate $i$). \ref{thm:diag-empty} then supplies the
emptiness, against \ref{thm:aj-tower}, and \ref{thm:no-extension-tower}
finishes.
\end{proof}

\begin{theorem}[The hypotheses, discharged]
\label{thm:aj-hypotheses}
\uses{thm:aj-no-extension, def:query-system}
\lean{ASM.sys_upperDirected, ASM.sys_evalSurjective}
\leanok
The Andersen--Jessen system is upper-directed, and every level's evaluation is
onto.
\end{theorem}

\begin{proof}
\leanok
$\iota = \mathbb{N}$ under $\leq$, so upper bounds are maxima. For surjectivity,
extend a given level-$n$ outcome by an arbitrary choice above $n$; the result is
coherent and restricts correctly.
\end{proof}

\begin{corollary}[The \texttt{UpperDirected} form is false]
\label{cor:upper-directed-false}
\uses{thm:aj-hypotheses, thm:aj-no-extension, def:seq-upper-directed}
\lean{ASM.upperDirected_extension_false}
\leanok
There is a query system satisfying \texttt{UpperDirected} and
\texttt{EvalSurjective} whose compatible $\sigma$-additive marginals admit no
extension.
\end{corollary}

\begin{proof}
\leanok
\uses{thm:aj-hypotheses}
The Andersen--Jessen system, with \ref{thm:aj-hypotheses} supplying the two
hypotheses and \ref{thm:aj-no-extension} the failure. At $\iota = \mathbb{N}$
the two hypotheses come apart: every finite set of levels has an upper bound, an
unbounded sequence of them does not.

Collective exhaustion is not discharged, so the form including that hypothesis
is not refuted by this corollary.
\end{proof}

\section{The diagonal layer}

\begin{definition}[Diagonal cylinders]
\label{def:diag-cylinders}
\lean{Diagonal.Diag}
\leanok
$D_n = \{\omega : \omega_i = \omega_0 \text{ for all } i \leq n\}$ inside
$\mathbb{N} \to \mathbb{R}$ with the product $\sigma$-algebra, and
$\Pi'(Y) = \{\omega : \omega_k \in Y_k \text{ for all } k\}$ (\texttt{Pi'}).
The cylinders decrease (\texttt{Diag\_antitone}), and a point of $D_n$ is
determined on $[0,n]$ by its first coordinate (\texttt{Diag\_eq\_of\_le}).
\end{definition}

\section{The delay system}

\begin{definition}[Delay query]
\label{def:delay-query}
\uses{def:query}
\lean{delayQuery, delayEval}
\leanok
Over a sensor alphabet $X$, the observation space is the bi-infinite stream
space $\Omega = X^{\mathbb{Z}}$. The query at $(d,\tau)$ reads $d$ samples
backwards at lag $\tau$:
\[
  \mathrm{eval}_{d,\tau}(\omega)
    = (\omega_0, \omega_{-\tau}, \omega_{-2\tau}, \dots, \omega_{-(d-1)\tau})
    \in X^d .
\]
\end{definition}

\begin{definition}[Delay refinement]
\label{def:delay-refine}
\uses{def:delay-query, def:refine}
\lean{delayLe, delayRefineMap}
\leanok
$(d,\tau) \leq (d',\tau')$ when $\tau' \mid \tau$ and $d' > (d-1)(\tau/\tau')$:
the finer lag divides the coarser one, and the finer window is long enough to
span the coarser one. The refinement map keeps the coordinates at positions
$0, \tau/\tau', \dots, (d-1)\tau/\tau'$.
\end{definition}

\begin{theorem}[The delay system is a query system]
\label{thm:delay-qs}
\uses{def:delay-query, def:delay-refine, def:query-system}
\lean{delayQuerySystem}
\leanok
$(d,\tau) \mapsto$ the delay query, with the refinement above, satisfies the
query-system axioms.
\end{theorem}

\begin{proof}
\leanok
\uses{def:delay-refine}
Reflexivity and transitivity of the order are divisibility and arithmetic;
$\pi_{\mathrm{refl}}$ and $\pi_{\mathrm{trans}}$ are index computations on
coordinate selections.
\end{proof}

\begin{theorem}[Upper-directed]
\label{thm:delay-upper}
\uses{thm:delay-qs, def:upper-directed}
\lean{delayQuerySystem.upperDirected}
\leanok
Any two delay queries have a common refinement.
\end{theorem}

\begin{proof}
\leanok
\uses{def:delay-refine}
Take lag $\gcd(\tau,\tau')$ and depth
$\max\{(d-1)(\tau/\gcd), (d'-1)(\tau'/\gcd)\} + 1$. The gcd divides both lags,
and the depth is chosen to span both windows.
\end{proof}

\section{The gap, exhibited}

\begin{theorem}[The delay system is NOT sequentially upper-directed]
\label{thm:delay-not-sud}
\uses{thm:delay-qs, def:seq-upper-directed}
\lean{delayQuerySystem.not_seqUpperDirected}
\leanok
There is no single query above the whole sequence $u_n = (n+2,\ 1)$.
\end{theorem}

\begin{proof}
\leanok
\uses{def:delay-refine}

\end{proof}

\section{Reconstruction, and the join to the query system}

The join rests on two conventions. The query system's parameter is the
\textbf{observation} space: \ref{def:delay-query} samples streams
$\mathbb{Z} \to Y$, and the state space never appears in it. And
\ref{def:delay-query} samples the past, at $0, -\tau, \dots, -(d-1)\tau$, while
reconstruction iterates $T$ forward; the two agree once reading the stream
backwards is reading the orbit forwards. Reading past times as \emph{backward}
iterates instead would need $T$ invertible, and for invertible $T$ the two
generate the same $\sigma$-algebra.

  \lean{SigmaEssential.Block}
  \leanok

\end{definition}

\begin{definition}[Intersection-closure: the pivot]
  \label{def:interclosed}
  \uses{def:sigma-class}
  \lean{SigmaEssential.InterClosed}
  \leanok
  $\mathrm{InterClosed}(d)$ says $A, A' \in d \Rightarrow A \cap A' \in d$. A
  Dynkin system with this property is exactly a $\sigma$-algebra, by Dynkin's

\end{definition}

\begin{lemma}[Distributivity is spent here, and only here]
  \label{lem:val-inter}
  \uses{def:interclosed, def:two-valued-state}
  \lean{SigmaEssential.TwoValuedState.val_inter}
  \leanok
  On an intersection-closed carrier a two-valued state is \emph{multiplicative}:
  if $A$ and $A'$ are true, so is $A \cap A'$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{def:interclosed}
  $A = (A \cap A') \sqcup (A \cap A'^{c})$ is a disjoint union inside $d$. Since
  $A \cap A'^{c} \subseteq A'^{c}$ and $A'$ is true, monotonicity makes
  $A \cap A'^{c}$ false; additivity on the disjoint pair forces $A \cap A'$ true.
\end{proof}

\section{MO$_2$: the smallest non-distributive witness}

\begin{definition}[Orthomodular lattice]
  \label{def:oml}
  \lean{QuerySystem.OrthomodularLattice}
  \leanok

  satisfying $x \wedge x^{\perp} = \bot$, $x \vee x^{\perp} = \top$, and the

\end{definition}

\begin{definition}[MO$_2$]

\section{MO$_2$ realized: the gap inside a $\sigma$-class}

MO$_2$ is a lattice; \ref{def:interclosed} is a condition on a $\sigma$-class of
sets. On four points there is a $\sigma$-class whose carrier is MO$_2$ and on
which the pivot fails.

\begin{definition}[The four-point carrier]
  \label{def:concrete-mo2}
  \uses{def:sigma-class, def:mo2}
  \lean{QuerySystem.ConcreteMO2.mo2Class}
  \leanok
  On $\Omega = \{0,1\}^{2}$ set $A = \{p : p_1 = 1\}$ and $B = \{p : p_2 = 1\}$,
  and let
  \[
    L \;=\; \{\emptyset, A, A^{c}, \Omega\} \;\cup\;
            \{\emptyset, B, B^{c}, \Omega\},
  \]
  the horizontal sum of two four-element Boolean blocks glued along
  $\{\emptyset, \Omega\}$. $L$ is a Dynkin system.
\end{definition}

\begin{proof}
  \leanok
  \uses{def:sigma-class}
  Complement-closure is immediate. For countable disjoint unions: two nonempty
  members of $L$ that are disjoint are complementary, because a member of one
  block always meets a member of the other --- $A \cap B$, $A \cap B^{c}$,
  $A^{c} \cap B$, $A^{c} \cap B^{c}$ are each a point. So a disjoint family in
  $L$ contains at most one nonempty member up to its complement, and its union
  is $\emptyset$, that member, or $\Omega$.
\end{proof}

\begin{theorem}[The carrier is MO$_2$]
  \label{thm:concrete-mo2-iso}
  \uses{def:concrete-mo2, def:mo2}
  \lean{QuerySystem.ConcreteMO2.rep, QuerySystem.ConcreteMO2.rep_le_iff,
        QuerySystem.ConcreteMO2.rep_ortho, QuerySystem.ConcreteMO2.rep_injective,
        QuerySystem.ConcreteMO2.rep_surjective}
  \leanok
  The indexing
  $\bot, a, a', b, b', \top \;\mapsto\; \emptyset, A, A^{c}, B, B^{c}, \Omega$
  is a bijection onto $L$; it carries the order of MO$_2$ to inclusion and
  $(\cdot)^{\perp}$ to set complement.
\end{theorem}

\begin{proof}
  \leanok
  \uses{def:concrete-mo2}
  Inclusion and complement are decidable over four points; injectivity follows
  from antisymmetry once the order correspondence is known, and surjectivity is
  the definition of $L$.
\end{proof}

\begin{definition}[Concrete compatibility, and a poor pair]
  \label{def:poor-pair}
  \uses{def:sigma-class}
  \lean{SigmaEssential.Blocks.Compat, SigmaEssential.Blocks.PoorPair}
  \leanok
  $A \leftrightarrow B$ (\emph{compatible}) means $A \cap B$ is itself in the
  carrier. The pair is \emph{poor} when $A \cap B$ is inhabited and yet the only
  carrier member inside it is $\emptyset$: the lattice meet is $\bot$ while the
  intersection is not.
\end{definition}

\begin{theorem}[The four-point carrier has a poor pair]
  \label{thm:concrete-mo2-poor}
  \uses{def:poor-pair, def:concrete-mo2}
  \lean{QuerySystem.ConcreteMO2.mo2Class_poorPair,
        QuerySystem.ConcreteMO2.singleton_not_mem}
  \leanok
  $A \cap B$ is the single point $(1,1)$, which is not in $L$, and no nonempty
  member of $L$ fits inside it. So $(A,B)$ is poor, and $A$ and $B$ lie in no
  common maximal block.
\end{theorem}

\begin{proof}
  \leanok
  \uses{def:concrete-mo2}
  Every nonempty member of $L$ has two points, so a member contained in a
  subsingleton is empty.
\end{proof}

\begin{corollary}[The pivot fails]
  \label{cor:concrete-mo2-not-interclosed}
  \uses{thm:concrete-mo2-poor, thm:concrete-mo2-iso, def:interclosed}
  \lean{QuerySystem.ConcreteMO2.mo2Class_not_interClosed}
  \leanok
  $L$ is not intersection-closed.
\end{corollary}

\begin{proof}
  \leanok
  \uses{thm:concrete-mo2-poor, thm:concrete-mo2-iso}
  $A$ and $B$ lie in $L$ (\ref{thm:concrete-mo2-iso}), so intersection-closure
  would put $A \cap B$ in $L$ as well. By \ref{thm:concrete-mo2-poor} the only
  member of $L$ inside $A \cap B$ is $\emptyset$, and $A \cap B$ is inhabited.
\end{proof}

\begin{theorem}[Four points are necessary, and $L$ attains the bound]
  \label{thm:interclosed-needs-four}
  \uses{def:interclosed, def:sigma-class, cor:concrete-mo2-not-interclosed}
  \lean{QuerySystem.ConcreteMO2.four_cells_nonempty,
        QuerySystem.ConcreteMO2.four_le_card_of_not_has_inter,
        QuerySystem.ConcreteMO2.interClosed_of_card_le_three,
        QuerySystem.ConcreteMO2.mo2Class_minimal}
  \leanok
  If $S$ and $T$ lie in a $\sigma$-class and $S \cap T$ does not, then
  $S \cap T$, $S \setminus T$, $T \setminus S$ and $(S \cup T)^{c}$ are all
  inhabited, so the carrier has at least four points. Every $\sigma$-class on
  three or fewer points is therefore intersection-closed, hence a
  $\sigma$-algebra; and $|\Omega| = 4$, so $L$ realizes the failure on the
  smallest carrier that admits one.
\end{theorem}

\begin{proof}
  \leanok
  \uses{def:sigma-class}
  Each cell is forced by a different closure property. $S \cap T = \emptyset$
  would put it in the carrier. $S \setminus T = \emptyset$ gives $S \subseteq T$
  and $S \cap T = S$; symmetrically $T \setminus S = \emptyset$ gives
  $S \cap T = T$. And $(S \cup T)^{c} = \emptyset$ makes $S^{c}$ and $T^{c}$
  disjoint members, so $S^{c} \cup T^{c}$ lies in the carrier and its complement
  is $S \cap T$. The four cells are pairwise disjoint, so one point from each
  gives four distinct points.
\end{proof}
\begin{theorem}[Meets exist; the gap is not their absence]
  \label{thm:concrete-mo2-meets}
  \uses{thm:concrete-mo2-poor}
  \lean{QuerySystem.ConcreteMO2.mo2Class_meetsExist,
        QuerySystem.ConcreteMO2.meet_AB_ne_inter}
  \leanok
  Any two members of $L$ have a greatest member of $L$ below both. In
  particular $A \wedge B = \emptyset$ while $A \cap B \neq \emptyset$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{def:concrete-mo2}
  Comparable pairs meet at the smaller one. For any other pair the intersection
  holds at most one point, so by the previous proof the only member below both
  is $\emptyset$.
\end{proof}

\section{The descent ladder: MO$_2$ carried to infinity}

The gap $a \wedge b = \bot$ with $a \not\le b^{\perp}$ relates \emph{one}
element to \emph{one} other; the descent question asks for one element
interleaved in the same way with an \emph{infinite orthogonal family}. Every
rung below is stated over a product of MO$_2$ copies with order, meet,
orthocomplement and bottom all computed coordinatewise, so each reduces to
\ref{thm:mo2-gap} at a single block.

\subsection{Rung 2: finite truncation}

\begin{definition}[Finite truncation $L_2^N$]
  \label{def:l2n}
  \uses{def:mo2}
  \lean{QuerySystem.L2N}
  \leanok
  $L_2^N := \mathrm{Fin}\,N \to \mathrm{MO}_2$, carrying the product lattice
  order: $x \le y$ iff $x_i \le y_i$ at every block $i$.
\end{definition}

\begin{definition}[Coordinatewise orthocomplement]
  \label{def:pi-ortho}
  \uses{def:l2n}
  \lean{QuerySystem.piOrtho}
  \leanok
  $(x^{\perp})_i := (x_i)^{\perp}$.
\end{definition}

\begin{definition}[The witness and the family]
  \label{def:descent-witness}
  \uses{def:l2n, def:pi-ortho}
  \lean{QuerySystem.p}
  \leanok
  $p := b$ on every block, and $a_i := a$ on block $i$, $\bot$ elsewhere
  (\texttt{QuerySystem.aFam}). The family is orthogonal because its members have
  disjoint supports.
\end{definition}

\begin{theorem}[$(\star)$ at every finite truncation]
  \label{thm:star-finite}
  \uses{def:descent-witness, thm:mo2-gap}
  \lean{QuerySystem.star_finite}
  \leanok
  For every $N$ and every block $i < N$,
  \[ p \sqcap a_i = \bot \quad\text{and}\quad p \not\le a_i^{\perp}. \]
\end{theorem}

\begin{proof}
  \leanok
  \uses{thm:mo2-gap}
  Both halves are coordinatewise tests. For the meet: off block $i$ the element
  $a_i$ is $\bot$, and on block $i$ the meet is $b \sqcap a = \bot$ by
  \ref{thm:mo2-gap}. For the non-orthogonality: $\le$ on the product is
  pointwise, so $p \le a_i^{\perp}$ would force $b \le a^{\perp}$ at block $i$,
  which \ref{thm:mo2-gap} refutes.
\end{proof}

\subsection{Rung 3: the infinite object, and what is assumed}

Navara's $L_2$ (PAMS 115 (1992) 425--431, p.~428) is assumed, not reformalized.
Eleven axioms record its carrier, that it is an orthomodular lattice, and that
it is built as a \textbf{sublogic of the product}, so its order,
orthocomplement, bottom and meet are the restrictions of the coordinatewise
ones. Two more name the witness $p = \bigvee_n b|_{C_n}$ and the family
$a_n = a|_{C_n}$. None of the eleven mentions $(\star)$.

\begin{definition}[Navara's $L_2$, blockwise]
  \label{def:navara-l2}
  \uses{def:mo2, def:oml}
  \lean{QuerySystem.L2}
  \leanok
  The carrier of the $(\beta)$-construction over a countable block index set,
  with block evaluation $\mathrm{blockVal} : L_2 \to \mathbb{N} \to
  \mathrm{MO}_2$ and the sublogic-of-product axioms
  \texttt{le\_iff\_blockwise}, \texttt{blockVal\_ortho}, \texttt{blockVal\_inf},
  \texttt{blockVal\_bot}. Cited, not proved.
\end{definition}

\begin{theorem}[$(\star)$ at infinity]
  \label{thm:star-infinite}
  \uses{def:navara-l2, thm:star-finite, thm:mo2-gap}
  \lean{QuerySystem.star_infinite}
  \leanok
  For Navara's witness $p$ and every $n$ in the countable orthogonal family,
  \[ p \sqcap a_n = \bot \quad\text{and}\quad p \not\le a_n^{\perp}. \]
  So $L_2$ is a concrete, $\sigma$-orthocomplete, non-Boolean orthomodular
  lattice in which one element is meet-zero against an infinite orthogonal
  family while orthogonal to no member of it.
\end{theorem}

\begin{proof}
  \leanok
  \uses{def:navara-l2, thm:mo2-gap}
  Verbatim the Rung-2 argument, with \texttt{le\_iff\_blockwise} supplying the
  coordinatewise order that the product structure gave for free. Antisymmetry of
  the blockwise order reduces the meet to a blockwise identity; the
  non-orthogonality is refuted at block $n$.
\end{proof}
\subsection{Rung 3 without the citation}

\begin{definition}[The concrete descent carrier]
  \label{def:concrete-descent}
  \uses{def:concrete-mo2, def:sigma-class}
  \lean{QuerySystem.ConcreteDescent.descentClass}
  \leanok
  On $\Omega = \mathbb{N} \times \{0,1\}^{2}$, the sets every fibre of which
  lies in the four-point carrier of \ref{def:concrete-mo2}: countably many
  disjoint copies of MO$_2$.
\end{definition}

\begin{proof}
  \leanok
  \uses{def:concrete-mo2}
  A Dynkin system. Complement and countable disjoint union are computed
  fibrewise --- the fibre of a complement is the complement of the fibre, the
  fibre of a union the union of the fibres, and disjointness passes to fibres
  --- so each closure property reduces to the same property of
  \ref{def:concrete-mo2}.
\end{proof}

\begin{theorem}[$(\star)$ at infinity, on a concrete carrier, with no cited axiom]
  \label{thm:concrete-descent}
  \uses{def:concrete-descent, thm:mo2-gap, thm:concrete-mo2-poor}
  \lean{QuerySystem.ConcreteDescent.star_concrete,
        QuerySystem.ConcreteDescent.aW_ortho,
        QuerySystem.ConcreteDescent.aW_pairwise_ne,
        QuerySystem.ConcreteDescent.not_interClosed}
  \leanok
  Put $p = \{q : q_2 \in B\}$ and $a_n = \{n\} \times A$. Then $\{a_n\}$ is an
  infinite pairwise-orthogonal family in the carrier, the greatest carrier
  element below both $p$ and $a_n$ is $\emptyset$, and
  $p \not\subseteq a_n^{c}$. The carrier is not intersection-closed.

  So a concrete, $\sigma$-complete, non-Boolean orthomodular carrier realizes
  $(\star)$ against an infinite orthogonal family --- the conclusion
  \ref{thm:star-infinite} draws from the eleven cited axioms of
  \ref{def:navara-l2}, here with none.
\end{theorem}

\begin{proof}
  \leanok
  \uses{def:concrete-descent, thm:mo2-gap}
  Orthogonality and distinctness are the block index: $a_n$ and $a_m$ live in
  different fibres. For the meet, a carrier element below both $p$ and $a_n$
  has empty fibre off block $n$, and at block $n$ its fibre sits inside
  $A \cap B$, a single point, so it is empty by \ref{thm:concrete-mo2-poor}.
  Non-orthogonality is witnessed at block $n$ by the point $(n, (1,1))$, which
  lies in $p$ and in $a_n$; that same point is the fibre of $p \cap a_n$, which
  the carrier does not contain.
\end{proof}

\subsection{The axioms are consistent}

An axiom set with no model proves everything, so the obligation is discharged by
exhibiting a model.

\begin{theorem}[Products of orthomodular lattices are orthomodular]
  \label{thm:prod-oml}
  \uses{def:oml}
  \lean{QuerySystem.instPiOrthomodularLattice}
  \leanok
  If each $\alpha_i$ is an orthomodular lattice then so is $\prod_i \alpha_i$,
  with lattice operations, bounds and orthocomplement all coordinatewise.
\end{theorem}

\begin{proof}
  \leanok
  Every law of the class is an equation or an implication between elements, and
  the product order, lattice and bounded-order structures are pointwise; so each
  law follows from the same law in each factor, applied at each coordinate.
\end{proof}

\begin{theorem}[A model of the Rung-3 axioms]
  \label{thm:descent-consistent}
  \uses{def:navara-l2, thm:prod-oml}
  \lean{QuerySystem.ConsistencyModel.axioms_consistent}
  \leanok
  Take $M := \mathbb{N} \to \mathrm{MO}_2$ with all operations coordinatewise,
  $p := b$ everywhere, $a_n := a$ on block $n$. Then every one of the eleven
  Rung-3 axioms is a \emph{theorem} about $M$, and $(\star)$ holds in it
  directly. Hence the axiom set is consistent.
\end{theorem}

\begin{proof}
  \leanok
  \uses{thm:prod-oml}
  Each blockwise axiom is the corresponding product lemma. The axiom asserting
  that $L_2$ is an orthomodular lattice --- without which $\sqcap$, $\le$,
  $^{\perp}$ and $\bot$ in the other ten are undefined --- is supplied by
  \ref{thm:prod-oml}, and the model's orthocomplement is definitionally that
  structure's (\texttt{mOrtho\_eq\_ortho}).
\end{proof}

\label{def:central}

\leanok

\end{definition}

\begin{theorem}[All four traces of a central set agree]
\label{thm:central-traces}
\uses{def:central, thm:normal-form}
\lean{SigmaEssential.Ulam.central_all_traces}
\leanok
A central $E$ has a single $\xi$ with $\mathrm{trace}(E,f) \approx \xi$ for every
coordinate $f$.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:normal-form}
Coordinates $0,1,2$ carry countably equal traces, $\kappa_3$ is false, and even
weight forces the fourth. Equal traces force equal codes, since
$\kappa_a \neq \kappa_b$ would make $\xi \approx \xi^{c}$ and $M$ countable.
\end{proof}

\begin{theorem}[$\xi$-triviality, the paper's $[E] \in \{0,1\}$]
\label{thm:xi-trivial}
\uses{thm:central-traces, def:central, thm:normal-form}
\lean{SigmaEssential.Ulam.central_xi_trivial}
\leanok
That common $\xi$ is countable or co-countable.
\end{theorem}

\begin{proof}
\leanok
\uses{def:central, thm:normal-form}

\end{proof}

\uses{thm:xi-trivial, thm:central-traces}

\leanok

\section{Inner regularity: what forces $\sigma$-additivity, and what refuses it}

The witness above is a finitely additive two-valued state with no $\sigma$-additive
extension. That raises the complementary question: what would \emph{force}
$\sigma$-additivity, and how close does the boundary run? Two answers, in

\subsection{Transport: a compact class forces the $\sigma$-condition}

\begin{definition}[Countably compact class]
  \label{def:countably-compact-class}
  \lean{SigmaEssential.Blocks.CountablyCompactClass}
  \leanok
  A family $\mathcal{K}$ of sets such that every sequence from $\mathcal{K}$
  with the finite intersection property has nonempty total intersection.
\end{definition}

\begin{definition}[Marczewski approximation, two-valued in-block form]
  \label{def:approximates-on}
  \uses{def:countably-compact-class, def:finadd-state}
  \lean{SigmaEssential.Blocks.ApproximatesOn}
  \leanok
  Every value-$1$ member $E$ of the family carries a value-$1$ inner witness
  sandwiched below a class member: $D \subseteq K \subseteq E$ with
  $K \in \mathcal{K}$.
\end{definition}

\begin{theorem}[Compact transport]
  \label{thm:compact-transport}
  \uses{def:approximates-on, def:countably-compact-class, def:sigma-class}
  \lean{SigmaEssential.Blocks.compact_transport,
        SigmaEssential.Blocks.IsMaxBlock.isSigmaOn_of_compact_class}
  \leanok
  On a carrier where meets exist, a finitely additive two-valued state
  approximated on every maximal block by a countably compact class satisfies the
  $\sigma$-condition on the whole carrier.
\end{theorem}

\begin{proof}
  \leanok
  \uses{def:approximates-on}
  Run inside one maximal block and glue. Tails of a disjoint sequence are
  value-$1$, so their inner witnesses have value-$1$ --- hence nonempty ---
  finite intersections; the sandwiching compacts therefore have the finite
  intersection property, and countable compactness contradicts the empty tail
  intersection. Marczewski 1953, 4(i), two-valued.
\end{proof}

\subsection{Forcing: (8.1) makes a finitely additive state Dirac}

\begin{definition}[A field of sets and a two-valued state on it]

  \lean{SigmaEssential.Blocks.SetField, SigmaEssential.Blocks.FieldState}
  \leanok

\end{definition}

\begin{definition}[Inner regularity (8.1)]
  \label{def:inner-regular-81}
  \uses{def:set-field}
  \lean{SigmaEssential.Blocks.InnerRegular81}
  \leanok
  Every value-$1$ member of the field contains a value-$1$ in-field
  \emph{compact}.
\end{definition}

\begin{theorem}[Theorem-let R: (8.1) forces Dirac]
  \label{thm:dirac-forcing}
  \uses{def:inner-regular-81, def:set-field}
  \lean{SigmaEssential.Blocks.dirac_forcing}
  \leanok
  On a Hausdorff space, a two-valued finitely additive state satisfying (8.1)
  has nonempty $\mathcal{K}_1$ (its value-$1$ in-field compacts), nonempty
  $D = \bigcap \mathcal{K}_1$, and \emph{equals} $\delta_\omega$ on the field
  for every $\omega \in D$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{def:inner-regular-81}
  Value-$1$ in-field compacts are $\cap$-stable (compact $\cap$ closed, and
  Hausdorff makes compacts closed), finite intersections are value-$1$ hence
  nonempty, and compactness of one fixed member turns an empty total
  intersection into an empty finite one.
\end{proof}

\section{The two sides: cycles in the layered ring, walks in the tuple digraph}

\begin{definition}[Layer-injective simple cycle]
\label{def:lisc}
\lean{Reconstruction.Pruning.IsLISC}
\leanok
For $L \geq 1$ the \emph{layered ring} $R_L(\rho)$ has vertex set
$\mathbb{Z}_L \times A$ and arcs $(i,a) \to (i+1 \bmod L, b)$ for every
$(a,b) \in \rho$. A simple cycle of \emph{winding} $k$ is a closed walk of
length $kL$ with all vertices distinct. $\mathrm{LISC}_k(L)$ holds when one
exists.

Encoded by the state sequence $W$ on positions $t < kL$: $\rho$-steps between
consecutive positions, a closing $\rho$-step, and the simplicity clause ---
same layer $(t \bmod L)$ plus same state implies same position.
\end{definition}

\begin{definition}[Tuple digraph and rotation walks]
\label{def:tr}
\lean{Reconstruction.Pruning.IsTR}
\leanok
$T_k(\rho)$ has as vertices the \emph{injective} $k$-tuples $u \in A^k$, with
arcs $u \to v$ iff $(u_j, v_j) \in \rho$ for every $j$. For $r \in \mathbb{Z}_k$
let $\sigma_r(u)_j = u_{j+r}$. Then $\mathrm{TR}_k^{(r)}(L)$ holds when some
length-$L$ walk in $T_k(\rho)$ runs from a $u$ to $\sigma_r(u)$.

Encoded as $\tau : \mathbb{N} \to \mathbb{Z}_k \to A$ with $\tau_i$ injective
for $i \leq L$, coordinatewise $\rho$-arcs, and endpoint
$\tau_L\,j = \tau_0\,(j+r)$.
\end{definition}

\begin{lemma}[Step 2: extraction]
\label{lem:extraction}
\lean{Reconstruction.Pruning.isTR_one_of_isLISC}
\leanok
\uses{def:lisc, def:tr}
For $0 < L$, $\mathrm{LISC}_k(L)$ implies $\mathrm{TR}_k^{(1)}(L)$.
\end{lemma}

\begin{proof}
\leanok
\uses{def:lisc, def:tr}
Read the $k$ strands at each layer: set $\tau_i[j]$ to the state of $W[jL+i]$.
Injectivity of each $\tau_i$ is forced by simplicity: distinct vertices at the
same layer have distinct states. The endpoint computes to $\sigma_1(\tau_0)$.
\end{proof}

\begin{lemma}[Step 3: assembly]
\label{lem:assembly}
\lean{Reconstruction.Pruning.isLISC_of_isTR_one}
\leanok
\uses{def:lisc, def:tr}
For $0 < L$, $\mathrm{TR}_k^{(1)}(L)$ implies $\mathrm{LISC}_k(L)$.
\end{lemma}

\begin{proof}
\leanok
\uses{def:lisc, def:tr}
Place position $t = jL+i$ at vertex $(i, \tau_i[j])$. Arcs within a strand are
the tuple-walk arcs; at a strand boundary the rotation endpoint supplies
$\tau_L[j] = \tau_0[j+1]$, which is the state at position $(j+1)L$, and at
$j = k-1$ this is the closure. Simplicity follows from the layer clock plus
injectivity of each $\tau_i$.
\end{proof}

\begin{lemma}[Step 4: generator conjugation]
\label{lem:conjugation}
\lean{Reconstruction.Pruning.isTR_rot_iff_isTR_one}
\leanok
\uses{def:tr}
If $r$ is a unit of $\mathbb{Z}_k$, that is $\gcd(r,k)=1$, then
$\mathrm{TR}_k^{(r)}(L)$ holds iff $\mathrm{TR}_k^{(1)}(L)$ does.
\end{lemma}

\begin{proof}
\leanok
\uses{def:tr}
The slot relabelling $P(u)[j] = u[j\bar r]$, with $\bar r = r^{-1}$, is a
bijection of injective tuples, preserves arcs (a fixed coordinate permutation),
and conjugates the rotations via $(j+r)\bar r = j\bar r + 1$.
\end{proof}

\begin{theorem}[Theorem P: pruning, phase-parametrized]
\label{thm:pruning}
\lean{Reconstruction.Pruning.pruning}
\leanok
\uses{lem:extraction, lem:assembly, lem:conjugation}
For every $\rho$, every $L \geq 1$, every $k \geq 1$ and every $r$ generating
$\mathbb{Z}_k$, $\mathrm{LISC}_k(L)$ holds iff $\mathrm{TR}_k^{(r)}(L)$ does.
\end{theorem}

\begin{proof}
\leanok
\uses{lem:extraction, lem:assembly, lem:conjugation}
Conjugate to $r=1$ by \ref{lem:conjugation}, then combine \ref{lem:extraction}
and \ref{lem:assembly}.
\end{proof}

\begin{lemma}[Lemma NG: assembly half]
\label{lem:ng}
\lean{Reconstruction.Pruning.isLISC_of_isTR_of_orbit}
\leanok
\uses{lem:assembly, def:tr, def:lisc}
Suppose the rotation $r$ closes an orbit of size $m$, witnessed by an injective
slot map $f : \mathbb{Z}_m \to \mathbb{Z}_k$ with $f(j+1) = f(j) + r$. Then a
rotation-$r$ tuple walk certifies $\mathrm{LISC}_m(L)$: winding $m$, not
winding $k$. For a generator $m = k$; for $\gcd(r,k) = d > 1$ it is the
strictly weaker $m = k/d$.
\end{lemma}

\begin{proof}
\leanok
\uses{lem:assembly}
Read the walk through $f$: this is a rotation-$1$ witness of width $m$, so
\ref{lem:assembly} assembles it. The instance $m=k$, $f = \mathrm{id}$, $r=1$
recovers Step 3.

Not formalized: that the $d$ orbit cycles are pairwise vertex-disjoint.
\end{proof}

\begin{definition}[Eventually periodic subset of the naturals]

\lean{Reconstruction.Pruning.IsEvPeriodic}

  \leanok
  \uses{def:cycle-stream}
  Interior positions are the cycle's own arcs; where the index rolls over the
  condition is the closing arc $\rho(W_{kL-1}, W_0)$.
\end{proof}

\begin{theorem}[Layer-injectivity is delay-injectivity]
  \label{thm:lisc-delay-witness}
  \uses{lem:cycle-stream-subshift, def:lisc, def:delay-query, thm:lag-algebra}
  \lean{QuerySystem.RelationalDelay.isLISC_delay_witness}
  \leanok
  If $\mathrm{LISC}_k(L)$ then there is a $\rho$-trajectory $\omega$, periodic
  of period $kL$, whose depth-$k$, lag-$L$ delay query is injective.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:cycle-stream-subshift, def:lisc}
  Take $\omega = \mathrm{cycleStream}\,W\,(kL)$; \ref{lem:cycle-stream-subshift}
  gives membership, and periodicity is immediate. The delay query at lag $L$ and
  depth $k$ samples $\omega$ at $0, -L, \dots, -(k-1)L$, which are the positions
  $0, L, \dots, (k-1)L$ of the cycle --- the $k$ positions sharing layer $0$. By
  \ref{def:lisc}'s simplicity clause these carry distinct states.
\end{proof}

\begin{corollary}[$\mathrm{Unsafe}$, in the delay vocabulary]
  \label{cor:unsafe-delay}
  \uses{thm:lisc-delay-witness, def:unsafe-safe}
  \lean{QuerySystem.RelationalDelay.unsafe_delay_witness}
  \leanok
  $L \in \mathrm{Unsafe}(\rho)$ yields a periodic $\rho$-trajectory on which
  the lag-$L$ delay query resolves $k \geq 2$ distinct strands.
\end{corollary}

\begin{proof}
  \leanok
  \uses{thm:lisc-delay-witness, def:unsafe-safe}
  \ref{def:unsafe-safe} unfolds to an $\mathrm{LISC}_k(L)$ with $k \geq 2$;
  apply \ref{thm:lisc-delay-witness}.
\end{proof}

\begin{lemma}[The hypothesis is satisfiable]
  \label{lem:lisc-inhabited}
  \uses{thm:lisc-delay-witness}
  \lean{QuerySystem.RelationalDelay.isLISC_complete_bool,
        QuerySystem.RelationalDelay.exists_delay_witness_complete_bool}
  \leanok
  The complete relation on $\{\bot,\top\}$ satisfies $\mathrm{LISC}_2(1)$, so
  the conclusion of \ref{thm:lisc-delay-witness} is inhabited.
\end{lemma}

\begin{proof}
  \leanok
  Steps and closure are immediate; simplicity is the two-case check that the
  cycle's two states differ.
\end{proof}

\subsection{The deterministic case, as a subclass}

\begin{lemma}[Delay queries see only non-positive times]
  \label{lem:delay-nonpos}
  \uses{def:delay-query}
  \lean{QuerySystem.RelationalDelay.delayEval_congr_nonpos}
  \leanok
  Streams agreeing at every $t \leq 0$ are indistinguishable to every delay
  query.
\end{lemma}

\begin{proof}
  \leanok
  \ref{def:delay-query} samples at $0, -\tau, \dots, -(d-1)\tau$.
\end{proof}

\begin{lemma}[The two halves of a deterministic trajectory]
  \label{lem:graph-subshift-halves}
  \uses{def:subshift, def:orbit-stream}
  \lean{QuerySystem.RelationalDelay.graphSubshift_neg,
        QuerySystem.RelationalDelay.graphSubshift_pos}
  \leanok
  For $\omega \in \Sigma_{\mathrm{graph}(T)}$: the non-positive half is the
  forward orbit of the base point, $\omega(-n) = T^n(\omega_0)$; the positive
  half is a coherent backward orbit, $T^n(\omega_n) = \omega_0$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{def:subshift}
  Induction in each direction, from the trajectory condition read forwards and
  backwards respectively.
\end{proof}

\begin{theorem}[Deterministic is a subclass]
  \label{thm:deterministic-subclass}
  \uses{lem:graph-subshift-halves, lem:delay-nonpos, def:orbit-stream, thm:delay-eval-orbit}
  \lean{QuerySystem.RelationalDelay.delayEval_comp_eq_orbitStream}
  \leanok
  For $\omega \in \Sigma_{\mathrm{graph}(T)}$ and an observable $h$, the delay
  data the relational picture assigns to $\omega$ equals the delay data the
  deterministic picture assigns to $\omega_0$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:graph-subshift-halves}
  Both sides evaluate at $-(k\tau)$, which \ref{lem:graph-subshift-halves} makes
  $T^{k\tau}(\omega_0)$.
\end{proof}

\begin{remark}[Base points outside the subshift]
  \label{rmk:deterministic-residue}
  \uses{thm:deterministic-subclass}
  \lean{QuerySystem.RelationalDelay.graphSubshift_base_mem_range}
  \leanok
  Being the base of a bi-infinite trajectory forces $\omega_0 \in
  \mathrm{range}(T^n)$ for every $n$. For non-surjective $T$ there are therefore
  points based at no relational trajectory whose delay data is still defined by
  \ref{def:orbit-stream}.
\end{remark}

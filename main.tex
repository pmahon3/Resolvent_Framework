\section{Local Linearizations}

Consider the Cauchy problem
\begin{align}
    \frac{du}{dt} + A(u(t)) &= 0, \quad t > 0, \label{eq:cauchy}\\[1mm]
    u(0) &= u_0,\quad u_0 \in U \subset X, \nonumber
\end{align}
where \(A\) is an operator (possibly nonlinear and state-dependent) that generates a contraction semigroup \(T(h)\) on the Banach space \(\mathcal{X}\). 

In many practical settings, one only has access to discrete observations of the state. Let these observations be arranged in the data matrix
\begin{align}
X_t &= \begin{bmatrix}
    x(t) \\
    x(t-h) \\
    x(t-2h) \\
    \vdots 
\end{bmatrix},
\end{align}
where the rows correspond to states sampled at time intervals of length \(h\).

A first approach is to attempt a global linearization via a least-squares formulation:
\begin{align}
    X_t &\approx X_{t-h}\, T(h), \label{eq:global_lin}\\[1mm]
    \text{or equivalently,} \quad u(t) &\approx T(h) \, x(t-h).
\end{align}
If the residuals from this approximation are well within ann acceptable and prescribed tolerance—then it is reasonable to assume that the dynamics are well represented by a linear operator. In such a case, one can recover the global generator via
\begin{align}
    A = \lim_{h\rightarrow 0^+} \frac{1}{h}\left(T(h) - I\right).
\end{align}
This approach is particularly effective when \(A\) is linear or nearly linear. The contraction property aids in establishing the existence and uniqueness of solutions, however it also means that a global linearization of \(A\) may be unattainable when its dependence on \(u(t)\) is significant with residuals remaining unacceptably large. 

A global linearization may fail to capture the system's behavior when \(A\) is state dependent. In such case a local linearization may be preferable. 

To construct a local approximation  we can introduce a weighting operator \(W\) designed to select or emphasize those rows in \(X_t\) that are dynamically relevant. The idea is to filter the data so that regions exhibiting dynamics similar to those surrounding \(x(t)\) are highlighted, typically by considering observations local to $x(t)$ under some metrization of $\mathcal{X}$.

If $W$ is constructed using a localization kernel, $k_\theta(x(j), x(i)$), where $\theta$ parametrizes what ``local'' means, then the weighted observations are,
\begin{align}
    W_\theta X_t,
\end{align}
and the local approximation is expressed as
\begin{equation}
    W_\theta X_t \approx W_\theta X_{t-h} \, \hat{T}(h,\theta, t), \label{eq:local_lin}
\end{equation}
where \(\hat{T}(h, \theta, t)\) is the locally linearized evolution operator valid in a neighborhood of \(x(t)\). If a weighted pseudoinverse is employed, one may write
\begin{align}
    A(x(t), \theta, h) = \left( \left( W_\theta X_{t-h} \right)^+ \left( W_\theta  X_t \right) - I \right)
\end{align}
where \(\left( W X_{t-h} \right)^+\) denotes the pseudoinverse of \(W X_{t-h}\). Now recovery of the generator $A(x(t))$ requires an additional spatial limit. 
\begin{align}
A(x(t)) = \lim_{h\rightarrow 0} \frac{1}{h}\left(\lim_{\theta\rightarrow 0}\,\hat{T}(h,\theta, t) - I\right)
\end{align}
If T is (i) deterministic ( i.e.   $\theta \to 0$ iff $k_\theta \to \delta$), and (ii) locally linear (i.e. $T = \sum_{k=0}^\infty \frac{t^k}{k!} A(x_{j}, \theta, h)^k $  converges where required),  then this is a nice framework for approximating the state-dependent generator \(A(x(t))\) in the presence of features precluding a global linear model. 

However, in the absence of any strong \emph{a priori} intuition regarding even the most general assumptions of the generating process, most notably stochasticity and the rate of decay of higher order terms, deciding how to approach these limits (and thus what constitutes a \emph{good} linearization) is not easy. 

\section{Locality in Infinitesimal Generators}


\subsection{The Evolution Kernel}

In the classical theory of \(C_0\) semigroups, the evolution operator \(T(h)\) composes a \(C_0\) semigroup generated by a linear operator \(A\)
\[
T(h) = e^{hA}.
\]
In this setting, the kernel \(K_h(\xi,\eta)\) is the Dirac delta
\[
K_h(\xi,\eta) \xrightarrow[h\to0]{} \delta(\xi-\eta).
\]

\begin{proof}
    Consider the integral operator

\begin{equation}
    (T(h) f) (\xi) = \int_\Omega K_h(\xi, \eta) f(\eta) d\eta
\end{equation}

with infinitesimal generator

\begin{align}
    (A f)(\xi)  &= \lim_{h\rightarrow 0} \frac{1}{h}\left(T(h) - I\right) f(\xi) \\
    &= \lim_{h\rightarrow 0} \frac{1}{h}\left((T(h)f)(\xi) -  f(\xi)\right) \\
    \intertext{By substitution}
    (A f)(\xi) &= \lim_{h\rightarrow 0} \frac{1}{h}\left(\int_\Omega K_h(\xi, \eta) f(\eta) d\eta - f(\xi)\right) \\
    \intertext{The goal is to rewrite the integral term so that \( f(\xi) \) appears inside the integral. We add and subtract \(  K_h(\xi, \eta) f(\xi) \) inside the integral,}
    (A f)(\xi) &= \lim_{h\rightarrow 0} \frac{1}{h} \left( \int_{\Omega} K_h(\xi, \eta) f(\eta) - K_h(\xi, \eta) f(\xi) +  K_h(\xi, \eta) f(\xi) d\eta - f(\xi) \right)\\
    \intertext{and factor,}
    (A f)(\xi) &= \lim_{h\rightarrow 0} \frac{1}{h} \left( \int_{\Omega} K_h(\xi, \eta) (f(\eta) - f(\xi)) d\eta + \int_{\Omega} K_h(\xi, \eta) f(\xi) d\eta - f(\xi) \right)\\
    &= \lim_{h\rightarrow 0} \frac{1}{h} \int_{\Omega} K_h(\xi, \eta) (f(\eta) - f(\xi)) d\eta + \lim_{h\rightarrow 0} \frac{1}{h} \left( \int_{\Omega} K_h(\xi, \eta) d\eta - 1 \right) f(\xi)
\end{align}

We can ensure that \(K_h(\xi,\eta)\) acts as a state transition probability density by imposing the normalization condition,

\begin{equation}
\int_{\Omega} K_h(\xi, \eta) d\eta = 1
\end{equation}

while at the same time eliminating the second term,

\begin{equation}
\lim_{h\rightarrow 0} \frac{1}{h} \left( \int_{\Omega} K_h(\xi, \eta) d\eta - 1 \right) f(\xi) = 0
\end{equation}

thus recovering the infinitesimal generator as a standard non-local operator,

\begin{equation}
(A f)(\xi) = \lim_{h\rightarrow 0} \frac{1}{h} \int_{\Omega} K_h(\xi, \eta) (f(\eta) - f(\xi)) d\eta
\end{equation}

\paragraph{Kernel Structure:}

The resolvent operator associated with \( A \) is given by:
\begin{align}
    R_{\lambda}(A) &= (\lambda I - A)^{-1}.
    \intertext{It is well known (see, e.g. Pazy) that the semigroup \(T(h)\) can be expressed in terms of the resolvent via the contour integral,}
    T(h) &= \frac{1}{2\pi i} \oint_{\Gamma} e^{h\lambda} R_{\lambda}(A) d\lambda\\
    \intertext{We equate the integral representations}
    \int_{\Omega} K_h(\xi, \eta) f(\eta) d\eta &= \frac{1}{2\pi i} \oint_{\Gamma} e^{h\lambda} R_{\lambda}(A) f(\xi) d\lambda \label{eq:omega_k_gamma_R}\\
    \intertext{Furthermore, Arendt \& Bukhvalov showed that if \( T(h) \) is an integral operator, then so is the resolvent,}
    (R_{\lambda} f)(\xi) &= \int_{\Omega} M_{\lambda}(\xi, \eta) f(\eta) d\eta \\
    \intertext{where \( M_{\lambda}(\xi, \eta) \) is the resolvent kernel. From (\ref{eq:omega_k_gamma_R}) using the integral resolvent,}
    \int_{\Omega} K_h(\xi, \eta) f(\eta) d\eta &= \frac{1}{2\pi i} \oint_{\Gamma} e^{h\lambda} \int_{\Omega} M_{\lambda}(\xi, \eta) f(\eta) d\eta \, d\lambda\\
    \intertext{Two cases follow regarding the absolute integrability of}\nonumber
    F(\lambda, \eta) &= e^{h\lambda} M_\lambda(\xi, \eta) f(\eta)
\end{align}

If $F \in L^1(\Gamma \times \Omega)$ then for
\begin{align}
    \oint_{\Gamma} \int_{\Omega} \left| F(\lambda, \eta)\right|& \, d\mu(\eta) \, d\nu(\lambda)
    \intertext{we may apply Fubini's theorem and interchange the order of integration}
    \int_{\Omega} K_h(\xi, \eta) f(\eta) d\eta &= \int_{\Omega}  \left( \frac{1}{2\pi i} \oint_{\Gamma}  e^{h\lambda} M_{\lambda}(\xi, \eta)  d\lambda \right) f(\eta) d\eta\\
    \intertext{Since this must hold for all \( f \), we identify the kernels,}
    K_h(\xi, \eta) &= \frac{1}{2\pi i} \oint_{\Gamma} e^{h\lambda} M_{\lambda}(\xi, \eta) d\lambda\\
    \intertext{Since \(T(0)=I\), the kernel at $h=0$ must satisfy}
    K_0(\xi, \eta) &= \delta(\xi-\eta)\\
    \intertext{In many classical settings, one can show that the resolvent satisfies}
    (\lambda I-A) R_\lambda(A) &= I\\
    \intertext{so that in kernel form}
    (\lambda I-A)M_\lambda(\xi,\eta) &= \delta(\xi-\eta)\\
    \intertext{\(M_\lambda(\xi,\eta)\) serves as the Green's function for the operator \(\lambda I-A\). Supposing that for values of \(\lambda\) in a suitable region the Neumann series expansion hold}
    (\lambda I-A)^{-1} &= \sum_{n=0}^{\infty} \frac{A^n}{\lambda^{n+1}}\\
    \intertext{Then the kernel \(M_\lambda(\xi,\eta)\) may be expressed as}\\
    M_\lambda(\xi,\eta) &= \sum_{n=0}^{\infty}\frac{A^n}{\lambda^{n+1}}\,\delta(\xi-\eta)\\
    \intertext{Substitute this expansion into}
    K_h(\xi,\eta) &= \frac{1}{2\pi i} \oint_{\Gamma} e^{h\lambda}\left( \sum_{n=0}^{\infty}\frac{A^n}{\lambda^{n+1}}\,\delta(\xi-\eta) \right) d\nu(\lambda)\\
    &= \delta(\xi-\eta) \sum_{n=0}^{\infty} A^n \left[ \frac{1}{2\pi i} \oint_{\Gamma} \frac{e^{h\lambda}}{\lambda^{n+1}} \, d\nu(\lambda) \right] \label{eq:series}\\
    \intertext{According to the residue theorem}
    \frac{1}{2\pi i} \oint_{\Gamma} \frac{e^{h\lambda}}{\lambda^{n+1}} \, d\nu(\lambda) &= \frac{h^n}{n!}\\
    \intertext{Substituting this back into \eqref{eq:series}, we obtain}
    K_h(\xi,\eta) &= \delta(\xi-\eta) \sum_{n=0}^{\infty} \frac{h^n}{n!} A^n = \delta(\xi-\eta)e^{hA}\\
    \intertext{Thus,}
    K_h(\xi,\eta) &= \delta(\xi-\eta)e^{hA}\\
    \intertext{which is equivalent to the operator identity,}
    T(h) &= e^{hA}
\end{align}
\end{proof}

\paragraph{The Nonlinear (State-Dependent) Case:}

When the system is nonlinear, the evolution is typically given by a state-dependent mapping,
\[
T(h)(\xi) = \Phi_h(\xi),
\]
we may still maintain an integral representation:
\[
(T(h)f)(\xi) = \int_{\Omega} K_h(\xi,\eta) f(\eta)\,d\eta.
\]
In order for the kernel to capture the correct local behavior of the flow, we require that for small \(h\),
\[
K_h(\xi,\eta) \to \delta\Bigl(\Phi_h(\xi)-\Phi_h(\eta)\Bigr).
\]
Since \(\Phi_0\) is the identity mapping, this condition reduces to
\[
\lim_{h\to0} K_h(\xi,\eta) = \delta(\xi-\eta),
\]
which, in the weak sense, means that for any test function \(f\),
\[
\lim_{h\to 0}\int_{\Omega}K_h(\xi,\eta)f(\eta)\,d\eta = f(\xi).
\]


\section{Locally--linear Stochastic Operators}

\subsection{Local linear model}

For each embedded state $\mathbf{x}\in \mathcal{X} $
fit a local globally weighted least--squares linear map
\begin{align}
A_{\theta}(\mathbf{x})
    &=
    \underset{A}{\arg\min}\;
        \sum_{j\in\mathcal{N}_{\theta}(\mathbf{x})}
        w_{\theta}(\mathbf{x},\mathbf{x}_{j})
        \,\lVert \mathbf{x}_{j+1} - A\,\mathbf{x}_{j} \rVert^{2}
\intertext{where $\mathcal{N}_{\theta}(\mathbf{x})$ is the neighbourhood chosen with
bandwidth (or kernel scale) $\theta$ and}
w_{\theta}(\mathbf{x},\mathbf{x}_{j}) & =\exp\Bigl({-\theta \lVert \mathbf{x}-\mathbf{x}_{j}\rVert}/\overline{\lVert \mathbf{x_i}-\mathbf{x}_{j}\rVert }\Bigr)
\intertext{The residuals}
\mathbf{r}_{j}
    &= \mathbf{x}_{j+1} - A_{\theta}(\mathbf{x})\,\mathbf{x}_{j}
    \qquad j\in\mathcal{N}_{\theta}(\mathbf{x})\\
\intertext{yield an empirical covariance}
\Sigma_{\theta}(\mathbf{x})
    &= \operatorname{Cov}_{j\in\mathcal{N}_{\theta}(\mathbf{x})}
      [ \mathbf{r}_{j} ]
\end{align}\\

\subsection{Stochastic transition kernel}

Replacing the Dirac mass of a deterministic map by a Gaussian centred on
the local linear prediction gives
\begin{align}
k_{\theta}(\mathbf{y}\mid\mathbf{x}) &=
    \mathcal{N}\!\bigl(\mathbf{y}\,;\;
        A_{\theta}(\mathbf{x})\,\mathbf{x},\;
        \Sigma_{\theta}(\mathbf{x})\bigr)\\
\intertext{a data--driven transition density that captures both drift
and anisotropic noise. A data driven stochastic Perron-Frobenius operator is given by the action of the kernel on probability densities $\rho$,}
(P_{\theta}\rho)(\mathbf{y})
    &=
    \int_{\mathcal{M}}
        k_{\theta}(\mathbf{y}\mid\mathbf{x})
        \,\rho(\mathbf{x})\;
    d\mathbf{x}
\end{align}

\subsection{Continuous--time interpretation}

For small sampling interval $\Delta t$ the update
\begin{align}
    \mathbf{x}_{t+\Delta t}
        &= A_{\theta}(\mathbf{x}_{t})\,\mathbf{x}_{t}
          + \boldsymbol{\eta}_{t},
        &
        \boldsymbol{\eta}_{t} &\sim \mathcal{N}
            \bigl(\mathbf{0},\,\Sigma_{\theta}(\mathbf{x}_{t})\bigr)         
\intertext{is a locally linear Itô SDE; subtract $\mathbf{x}_t$ and divide by the step size,}
    \frac{\mathbf{x}_{t+\Delta t}-\mathbf{x}_{t}}{\Delta t}
      &= \Bigl(\tfrac{A_{\theta}(\mathbf{x}_{t})-I}{\Delta t}\Bigr)\mathbf{x}_{t}
        \;+\; \frac{\boldsymbol{\eta}_{t}}{\Delta t}     
\intertext{Write the noise in the \emph{diffusion scaling}}
\end{align}
\begin{align}
    \boldsymbol{\eta}_{t} &= C(\mathbf{x}_{t})\sqrt{\Delta t}\ \boldsymbol{\xi}_{t}
     & \boldsymbol{\xi}_{t} &\sim \mathcal{N}(0,I)
\intertext{so that}
    \Sigma_{\theta}(\mathbf{x}_{t}) &= C(\mathbf{x}_{t})C^{\!\top} (\mathbf{x}_{t}) \Delta t   
\intertext{Define the local drift matrix}
    B(\mathbf{x}) &= A_{\theta}(\mathbf{x})-I   
\intertext{Taking $\Delta t\to0$ yields the Itô SDE}
    d\mathbf{X}_{t}
      &= B\bigl(\mathbf{X}_{t}\bigr)\,\mathbf{X}_{t}\,dt
        + C\bigl(\mathbf{X}_{t}\bigr)\,d\mathbf{W}_{t}
\end{align}
where $\mathbf{W}_{t}$ is a standard Wiener process.  The equation is
\emph{state--dependent linear}: for fixed~$x$ the map
$B(x)\,x$ is linear, but the coefficients vary smoothly over the
manifold.

If one keeps $\Sigma_{\theta}$ \emph{independent} of $\Delta t$, then
the result is a discrete--time Markov chain rather than the
time discretization of the continuous case.  Both views preserve the central
structure: 
\begin{align*}
    \text{``locally linear drift''} \;+\; \text{``additive Gaussian innovation''}
\end{align*}

 
Defining
$B(\mathbf{x}) = A_{\theta}(\mathbf{x}) - I$
and $C(\mathbf{x})C(\mathbf{x})^{\!\top}
      = \Sigma_{\theta}(\mathbf{x})/\Delta t$,
the infinitesimal generator acting on a smooth test function
$f$ is
\begin{align}
(\mathcal{L}f)(\mathbf{x})
    &= \nabla f(\mathbf{x})^{\!\top}\!
        B(\mathbf{x})\,\mathbf{x}
      + \tfrac12
        \operatorname{tr}\!\bigl(
            \Sigma_{\theta}(\mathbf{x})\,
            \nabla^{2} f(\mathbf{x})
          \bigr).
\end{align}

\subsection{Connection to Wold’s Decomposition}

For a covariance--stationary scalar process \(X_t\), Wold’s theorem states
\[
X_t \;=\; \sum_{j=0}^{\infty} \psi_j\,\varepsilon_{t-j} \;+\; \eta_t,
\]
where \(\{\varepsilon_t\}\) are orthogonal \emph{innovations},
\(\psi_0 = 1\), \( \sum \psi_j^2 < \infty \),
and \(\eta_t\) is a perfectly predictable (finite--rank) component.

\vspace{0.5em}
\noindent
In the locally--linear framework we obtain an analogous, \emph{state--dependent}
split at every embedded point \(\mathbf x_t\):
\[
\underbrace{\mathbf x_{t+1}}_{\text{next state}}
\;=\;
\underbrace{A_{\theta}(\mathbf x_t)\,\mathbf x_t}_{\text{predictable drift}}
\;+\;
\underbrace{\boldsymbol\varepsilon_t(\theta)}_{\text{innovation}},
\qquad
\boldsymbol\varepsilon_t(\theta)
= \mathbf x_{t+1} - A_{\theta}(\mathbf x_t)\,\mathbf x_t .
\]

\subsection{Fitting Algorithm}

Consider observations of state transition pairs $(y_i, x_i)_{i\in N}$ where $y_i = x_{i+1}$.

\begin{algorithm}[H] Data‐Driven Delay Embedding and Pointwise \(\theta\) Selection via Rose+PCA
\begin{algorithmic}[1]
\Require Univariate series \(\{x(t)\}\), over‐embed dimension \(D\), grid size \(K\), variance threshold \(\alpha\in(0,1)\).
\Ensure Reduced dimension \(d\), per‐anchor \(\{\theta_i^*\}\), final residuals \(\{\widetilde r_i\in\mathbb{R}^d\}\), coefficients \(\{\widetilde C_i\in\mathbb{R}^{d\times d}\}\).

\State \textbf{1. Build full \(D\)-dimensional delay embedding:}
\For{\(i=1\) to \(N\)} 
  \State \(\mathbf{x}_i \gets \bigl[x(t_i),\,x(t_i-1),\,x(t_i-2),\,\dots,\,x(t_i-(D-1))\bigr]\in\mathbb{R}^D.\)
\EndFor
\State \(E \gets [\mathbf{x}_1;\,\dots;\,\mathbf{x}_N]\in\mathbb{R}^{N\times D}.\)

\State \text{2. Construct \(\theta\)-grid}
\State \(\theta \gets [\theta_0 = 0; \ldots ; \theta_K = \theta_{max}]\)

\State \textbf{4. Compute Rose residuals \(r_{i,k}\in\mathbb{R}^D\):}
\For{\(i=1\) to \(N\)} 
  \For{\(k=1\) to \(K\)}
    \State Fit WLS at \(\mathbf{x}_i\) with kernel scale \(\theta_k\),  
      forecast next step \(\widehat{\mathbf{x}}_i\).  
    \State \(r_{i,k}\gets \mathbf{x}(t_i+1)\;-\;\widehat{\mathbf{x}}_i.\)
  \EndFor
\EndFor

\State \textbf{5. Select pointwise \(\theta_i^*\):}
\For{\(i=1\) to \(N\)} 
  \State \(\rho_i(k)\gets \|r_{i,k}\|_2,\quad k_i^*\gets \arg\min_{1\le k\le K} \rho_i(k).\)
  \State \(\theta_i^* \gets \theta_{k_i^*}.\)
\EndFor

\State \textbf{6. Collect selected residuals in \(\mathbb{R}^D\):}
\For{\(i=1\) to \(N\)} 
  \State \(r_i^* \gets r_{\,i,\,k_i^*}.\)
\EndFor
\State \(R^*\gets [\,r_1^*;\,\dots;\,r_N^*]\in\mathbb{R}^{N\times D}.\)

\State \textbf{7. PCA on \(R^*\) to determine reduced dimension \(d\):}
\State Compute SVD: \(R^* = U\,\Sigma\,V^\top.\)
\State Choose smallest \(d\) with \(\sum_{j=1}^d \sigma_j^2 \;/\;\sum_{j=1}^D \sigma_j^2 \ge \alpha.\)
\State \(V_d \gets V[:,1:d]\in\mathbb{R}^{D\times d}.\)

\State \textbf{8. Form reduced embedding in \(\mathbb{R}^d\):}
\For{\(i=1\) to \(N\)} 
  \State \(\widetilde{\mathbf{x}}_i \gets V_d^\top\,\mathbf{x}_i \in \mathbb{R}^d.\)
\EndFor

\State \textbf{9. Re‐fit Rose in \(\mathbb{R}^d\) at \(\theta_i^*\):}
\For{\(i=1\) to \(N\)} 
  \State Fit WLS on \(\widetilde{\mathbf{x}}_i\) with scale \(\theta_i^*\),  
    forecast \(\widehat{\widetilde{\mathbf{x}}}_i\).  
  \State \(\widetilde r_i \gets \widetilde{x}(t_i+1)\;-\;\widehat{\widetilde{\mathbf{x}}}_i\in\mathbb{R}^d.\)
  \State Extract coefficient \(\widetilde C_i\in\mathbb{R}^{d\times d}.\)
\EndFor

\State \textbf{10. Save outputs via \(\mathrm{save\_all}\):}
\State \(\bigl\{\widetilde{r}_i\bigr\}_{i=1}^N,\;\bigl\{\widetilde C_i\bigr\}_{i=1}^N,\;\{t_i\}_{i=1}^N,\;\{\theta_i^*\}_{i=1}^N.\)

\end{algorithmic}
\end{algorithm}

\section{A Categorical Scaffold for Local Linearization}
\label{sec:categorical}

Local linear models, residual covariances, and adaptive basis updates can all be
written algebraically, but a categorical formulation clarifies \emph{which parts
belong to the dynamics, which to the coordinate representation, and which to the
data–driven closure}.  By lifting the construction to functors, natural
transformations, and (co)monads we obtain

\begin{enumerate}[label=(\roman*)]
\item a precise separation of concerns
\item a route to operator‑theoretic results via functoriality
\item and a universal property (\emph{cofree} coalgebra) that characterises the
      residual‐frame augmentation
\end{enumerate}

\subsection{Components}

Throughout, $k\!\ge 2$ is fixed, so that $C^{k}$ flows have
well‑defined first and second jets.  All manifolds are assumed compact, and all
vector spaces are over~$\mathbb R$.

\begin{definition}[$\mathbf{Dyn}$]
  \label{def:dyn}
  The category $\mathbf{Dyn}$ consists of
  \begin{itemize}[leftmargin=1.75em]
  \item \textbf{objects} $(\mathcal M,\varphi^{t})$ where
        $\mathcal M$ is a compact $C^{k}$ manifold and
        $\varphi^{t}\!: \mathcal M\to\mathcal M$ a $C^{k}$ flow.
  \item \textbf{morphisms}
        $\Psi: (\mathcal M,\varphi^{t}) \to (\mathcal N,\psi^{t})$
        given by $C^{k}$ maps
        $\Psi:\mathcal M\rightarrow\mathcal N$ satisfying
        \begin{align}
            \Psi\circ\varphi^{t} &= \psi^{t}\circ\Psi
        \end{align}
        for all $t\in\mathbb R$. 
  \end{itemize}
\end{definition}

\begin{remark}
    Objects in $\mathbf{Dyn}$ are smooth dynamical systems. Morphisms $\Psi$ are \emph{dynamic‑preserving}
      (equivalently, \emph{equivariant} or \emph{trajectory‑intertwining}). They are dynamic-preserving in the sense that for smooth manifolds and flows:
    \begin{align}
        (\mathcal{M}, \varphi^t)&,\, (\mathcal{N}, \psi^t)  & t\in \R
    \end{align}
    the morphism,
    \begin{align}
        \Psi &= (\mathcal{M}, \varphi^t) \rightarrow  (\mathcal{N}, \psi^t) 
    \end{align}
    is a $C^k$ $(k \geq 1)$ map that is equivariant,
    \begin{align}
        \Psi(\varphi^t(x)) &= \psi^t(\Psi(x)) & \forall x \in \mathcal{M}
    \intertext{so that $\Psi$ transports trajectories  $\varphi^t$ onto trajectories $\psi^t$:}
        \Psi \circ \varphi^t &= \psi^t \circ \Psi
    \end{align}
\end{remark}


\begin{example}
  \label{ex:dyn_examples}
  Take the two-torus with linear flow,
  \begin{align}
  &(\mathbb{T}^2, \varphi^t), & \varphi^t(x,y) &= (x + \alpha t,\; y + \beta t)\pmod 1
  \intertext{and the circle with rigid rotation,}
  &(\mathbb{S}^1, \psi^t), & \psi^t(\theta) &= \theta + \alpha t \pmod 1
  \intertext{Let,}
  &\Psi : \mathbb{T}^2 \to \mathbb{S}^1, & \Psi(x,y) &= x \bmod 1
  \intertext{For all $(x,y) \in \mathbb{T}^2$ $t \in \R$:}
  \Psi(\varphi^t(x,y)) &= \Psi (x + \alpha t, y + \beta t)\\
  &= (x + \alpha t) \pmod 1\\
  &= \psi^t(\Psi(x,y))
  \intertext{so,}
  \Psi \circ \varphi^t &= \psi^t \circ \Phi, & \forall t &\in \R
\end{align}
and $\Psi$ is a dynamic-preserving $C^\infty$ map and morphism of $\mathbf{Dyn}$.
\end{example}

\begin{remark}[Discrete--time systems via suspension]
Although $\mathbf{Dyn}$ is defined only for \emph{flows}, any
discrete--time diffeomorphism $T\colon\mathcal M\!\to\!\mathcal M$ can
be embedded in $\mathbf{Dyn}$ through its \emph{suspension flow}. 

Form the mapping torus
\begin{align}
\operatorname{Sus}(\mathcal M,T)
   &:= (\mathcal M\times[0,1])/\bigl(x,1\bigr)\!\sim\!\bigl(Tx,0\bigr),
\intertext{and define the flow}
\varphi^{t}[(x,u)]&=[(T^{\lfloor t+u\rfloor}x,\;u+t-\lfloor t+u\rfloor)]
\intertext{Then $(\operatorname{Sus}(\mathcal M,T),\varphi^{t})$ is an object of $\mathbf{Dyn}$ whose \emph{time--$1$ map} reproduces $T$:}
\varphi^{1}[(x,0)]&=[(Tx,0)]
\end{align}
This construction extends to morphisms by  $\Sigma(f)[(x,u)]=[(f(x),u)]$, producing a functor $\Sigma\!: \mathbf{Dyn}_{\mathbb Z}\to\mathbf{Dyn}$. As such, results for continuous--time objects apply \emph{mutatis mutandis} to discrete systems via $\Sigma$.
\end{remark}

  \item \textbf{objects} $V$ finite-dimensional real vector spaces.
  \item \textbf{morphisms}  $L:  V \to W$ linear maps.

\end{definition}

\begin{definition}[Embedding functor]
Fix, for every Dyn–object $(\mathcal M,\varphi^{t})$
a generic embedding
$\Phi_{\mathcal M}:\mathcal M\hookrightarrow\mathbb R^{m(\mathcal M)}$.
Define the functor
\[
F:\mathbf{Dyn}\longrightarrow\mathbf{Vect}
\quad\text{by}\quad
\begin{cases}
F\bigl(\mathcal M,\varphi^{t}\bigr)=\mathbb R^{m(\mathcal M)},\\
F(\Psi)=\displaystyle
  \Phi_{\mathcal N}\!\circ\!\Psi\!\circ\!
  \Phi_{\mathcal M}^{-1}
    :\mathbb R^{m(\mathcal M)}\longrightarrow\mathbb R^{m(\mathcal N)}.
\end{cases}
\]
with identity $F(\mathrm{id}_{(\mathcal M,\varphi^{t})})=\mathrm{id}_{\mathbb R^{m(\mathcal M)}}$.

Functoriality is immediate from
$\Phi_{\mathcal N}\circ(\Psi_{2}\!\circ\!\Psi_{1})
 =F(\Psi_{2})\!\circ F(\Psi_{1})$, which gives the following diagram:

        & \mathbb R^{m(\mathcal N)}
    \end{tikzcd}
\]

\end{definition}

\begin{definition}[$\mathbf{Dyn}$ Component--wise product]
\label{def:dyn_prod_object}
Let $\bigl(\mathcal M_{i},\varphi^{t}_{i}\bigr)_{i=1}^{n}$ be objects of
$\mathbf{Dyn}$.  Define

\begin{align}
\mathcal M \;:=\; \prod_{i=1}^{n}\mathcal M_{i},
\qquad
\varphi^{t}(x_{1},\dots,x_{n})
       \;:=\;\bigl(\varphi^{t}_{1}(x_{1}),\dots,\varphi^{t}_{n}(x_{n})\bigr),
\end{align}
and projection maps
\(
\pi_{j}(x_{1},\dots,x_{n})=x_{j}\quad(j=1,\dots,n).
\)
We call
\begin{align}
(\mathcal M,\varphi^{t})
\end{align}
the \emph{component--wise product} of the given objects.
\end{definition}

\begin{lemma}[Universal property of finite products in $\mathbf{Dyn}$]
\label{lem:dyn_prod}
For the system $(\mathcal M,\varphi^{t})$ and projections
$\pi_{j}$ defined above, the pair
\begin{align}
\bigl((\mathcal M,\varphi^{t}),\;\{\pi_{j}\}_{j=1}^{n}\bigr)
\end{align}
is the categorical product of 
$\bigl(\mathcal M_{i},\varphi^{t}_{i}\bigr)_{i=1}^{n}$
in the category $\mathbf{Dyn}$.
\end{lemma}

\begin{proof}

Because each $\pi_{j}$ is smooth,
\begin{align}
\pi_{j}\!\bigl(\varphi^{t}(x_{1},\dots,x_{n})\bigr)
      =\varphi^{t}_{j}(x_{j})
      =\varphi^{t}_{j}\!\bigl(\pi_{j}(x_{1},\dots,x_{n})\bigr),
\end{align}
so $\pi_{j}$ intertwines the flows and is therefore a morphism in
$\mathbf{Dyn}$.

\medskip
\noindent
Let $(\mathcal N,\psi^{t})$ be any object and suppose we are given
morphisms
$f_{j}:(\mathcal N,\psi^{t})\to(\mathcal M_{j},\varphi_{j}^{t})$
for $j=1,\dots,n$.
Define
\begin{align}
f:\mathcal N\longrightarrow\mathcal M,
\qquad
f(y):=\bigl(f_{1}(y),\dots,f_{n}(y)\bigr).
\intertext{The map $f$ is $C^{k}$ because each $f_{j}$ is.
Equivariance follows component--wise:}
f(\psi^{t}y)
   =\bigl(f_{1}(\psi^{t}y),\dots\bigr)
   =\bigl(\varphi_{1}^{t}f_{1}(y),\dots\bigr)
   =\varphi^{t}\bigl(f(y)\bigr).
\end{align}
Hence $f$ is a morphism in $\mathbf{Dyn}$ and
$\pi_{j}\circ f = f_{j}$ by construction.

\medskip
\noindent
If $g:(\mathcal N,\psi^{t})\to(\mathcal M,\varphi^{t})$ is another
morphism with the same property $\pi_{j}\circ g=f_{j}$ for every $j$,
then for each $y\in\mathcal N$
\[
g(y)=\bigl(\pi_{1}g(y),\dots,\pi_{n}g(y)\bigr)
    =\bigl(f_{1}(y),\dots,f_{n}(y)\bigr)=f(y),
\]
so $g=f$.

\medskip
\noindent
Since a unique mediating morphism exists for every candidate
cone, $(\mathcal M,\varphi^{t})$ with the projections $\pi_{j}$ satisfies
the universal property of finite products in $\mathbf{Dyn}$.
\end{proof}

\begin{lemma}[Coproducts]
\label{lem:dyn_coprod}
For any family $\bigl(\mathcal M_{i},\varphi^{t}_{i}\bigr)_{i\in I}$,
the disjoint union
\[
\Bigl(\bigsqcup_{i\in I}\mathcal M_{i},\;
      \varphi^{t}(x)=\varphi^{t}_{i}(x)\text{ if }x\in\mathcal M_{i}\Bigr)
\]
with the canonical injections
$\iota_{i}:\mathcal M_{i}\hookrightarrow\bigsqcup_{j}\mathcal M_{j}$
is the coproduct in $\mathbf{Dyn}$.
\end{lemma}

\begin{proof}
Disjoint unions of compact manifolds are compact manifolds, the flow is
$C^{k}$ on each component, and the injections are equivariant by
construction.  The universal property follows from the universal
property of disjoint unions in smooth manifolds.
\end{proof}

\begin{lemma}[Regular equalisers]
\label{lem:dyn_equaliser}
Let $f,g:(\mathcal M,\varphi^{t})\to(\mathcal N,\psi^{t})$ be morphisms.
Assume $f$ and $g$ are transverse to the diagonal
$\Delta_{\mathcal N}\subset\mathcal N\times\mathcal N$.  Then
\[
E:=\{x\in\mathcal M\mid f(x)=g(x)\}
\]
is an embedded $C^{k}$ sub‑manifold, invariant under $\varphi^{t}$.  The
inclusion $i:E\hookrightarrow\mathcal M$ with the restricted flow
$\varphi^{t}_{|E}$ realises the equaliser of $(f,g)$ in
$\mathbf{Dyn}$.
\end{lemma}

\begin{proof}
Transversality implies $E$ is an embedded sub‑manifold.  Equivariance of
$f$ and $g$ gives $f(\varphi^{t}x)=\psi^{t}f(x)=\psi^{t}g(x)=g(\varphi^{t}x)$,
so $E$ is flow‑invariant.  The inclusion $i$ is a morphism, and any
morphism $h:(\mathcal K,\chi^{t})\to(\mathcal M,\varphi^{t})$ with
$fh=gh$ factors uniquely through $i$ by the usual smooth‑manifold
argument.
\end{proof}

\begin{remark}[Limitations of completeness]
\label{rem:dyn_incomplete}
The category of $C^{k}$ manifolds is neither complete nor cocomplete; an
infinite product need not be a manifold, and general coequalisers (or
non‑regular equalisers) can fail to carry a smooth Hausdorff structure.
Consequently $\mathbf{Dyn}$ is \emph{finitely} complete and has
coproducts, but it does not possess arbitrary small limits or colimits.
\end{remark}


\begin{lemma}[Limits and colimits]
  \label{lem:dyn_limits}
  $\mathbf{Dyn}$ has all small limits and colimits; they are inherited
  point‑wise from the category of $C^{k}$ manifolds.
\end{lemma}

\begin{proof}[Sketch]
  Products exist because finite products of compact manifolds are compact
  manifolds; the flow acts component‑wise.  Equalisers are submanifolds defined
  by equalising maps that commute with the flow.  Dually for colimits.
\end{proof}

The remainder of this section builds functors \emph{out of} $\mathbf{Dyn}$ and
uses natural transformations and comonads to formalise local linearisation and
its residual correction.


\subsection{Local linearisation as a natural transformation}
Write $\Pi_d:\mathbf{Dyn}\to\mathbf{Dyn}$ for the functor that forgets all
$\ge\!2$‑jet data.  Weighted least squares at each anchor
$x$ induces the component
\[
  \eta_{(\mathcal M,\varphi^t)}(x)\;:=\;B_x
  : F(\mathcal M,\varphi^t)\;\longrightarrow\;F\!\circ\!\Pi_d(\mathcal M,\varphi^t),
\]
yielding a natural transformation $\eta:F\!\Rightarrow\!F\!\circ\!\Pi_d$.

\subsection{Residual frame field as a cofree coalgebra}
Define a comonad
$\mathsf G:(\mathbf{Vect},\otimes)\to(\mathbf{Vect},\otimes)$ by
$\mathsf G(V)=V\oplus\bigl(\text{span}\,Q_x\bigr)$.
The map $x\mapsto(B_x,Q_x)$ endows each fibre with the structure of a
$\mathsf G$‑coalgebra, cofree under the variance‑fraction constraint
$\rho$.  In string‑diagram notation:

\[
  \begin{tikzcd}[row sep=huge]
    F(\mathcal M,\varphi^t)
      \arrow[d, "\eta"']
      \arrow[rr, dashed, "\Delta" description]
      && \mathsf G F(\mathcal M,\varphi^t)
          \arrow[d, "\mathsf G\eta"] \\
    F\!\circ\!\Pi_d(\mathcal M,\varphi^t)
      \arrow[rr, dashed, "\mathsf G\Delta"']
      && \mathsf G F\!\circ\!\Pi_d(\mathcal M,\varphi^t)
  \end{tikzcd}
\]
The dashed arrows are the Rose augmentation; commutativity is enforced by
letting $\mathsf G$ be \emph{cofree} with respect to~$\rho$.

% ---------- 1.  Measure & Probability Foundations ----------
\subsection{Measure \& Probability Foundations}

\begin{definition}[$\sigma$–Algebra]\label{def:sigma-algebra}
Given a set $X$, a \emph{$\sigma$–algebra} $\mathcal F\subseteq 2^X$ is
a non‑empty collection of subsets closed under complementation and
countable unions (hence also countable intersections).
\end{definition}

\begin{definition}[Measurable Space]\label{def:measurable-space}
A \emph{measurable space} is a pair $(X,\mathcal F)$ where
$\mathcal F$ is a $\sigma$–algebra on $X$
(Definition~\ref{def:sigma-algebra}).
\end{definition}

\begin{definition}[Measurable Map]\label{def:measurable-map}
If $(X,\mathcal F)$ and $(Y,\mathcal G)$ are measurable spaces, a map
$f:X\to Y$ is \emph{measurable} if $f^{-1}(B)\in\mathcal F$ for all
$B\in\mathcal G$.
\end{definition}

\begin{definition}[Measure \& Probability Measure]\label{def:measure}
Let $(X,\mathcal F)$ be a measurable space.
A \emph{measure} is a countably additive map
$\mu:\mathcal F\to[0,\infty]$ with $\mu(\varnothing)=0$.
It is a \emph{probability measure} if $\mu(X)=1$.
We write $PX$ for the set of all probability measures on
$(X,\mathcal F)$.
\end{definition}

\begin{definition}[Push‑Forward Measure]\label{def:pushforward}
Given a measurable $f:(X,\mathcal F)\to(Y,\mathcal G)$ and
$\mu\in P X$, the \emph{push‑forward} measure
$f_{*}\mu\in P Y$ is defined by
$(f_{*}\mu)(B)=\mu\bigl(f^{-1}(B)\bigr)$ for $B\in\mathcal G$.
\end{definition}

\begin{definition}[Pull‑Back Measure]\label{def:pullback-measure}
Given a measurable $f:(X,\mathcal F)\to(Y,\mathcal G)$ and
$\mu\in P X$, the \emph{pull‑back} measure
$f^{*}\mu\in P Y$ is defined by $f^{*}\mu(A)\;:=\;\mu\!\bigl(f(A)\bigr)$ for $A\in\mathcal F$.

\end{definition}


% ---------- 2.  Category Theory ----------
\subsection{Category Theory}

\begin{definition}[Category]\label{def:category}
A \emph{category} $\mathcal C$ consists of a class of objects,
hom‑sets $\mathrm{Hom}_{\mathcal C}(A,B)$ for each ordered pair
$(A,B)$, identity morphisms $\mathrm{id}_A$, and associative
composition, satisfying the usual unit laws.
\end{definition}

\begin{definition}[Functor]\label{def:functor}
Let $\mathcal C,\mathcal D$ be categories.
A \emph{functor} $F:\mathcal C\to\mathcal D$ assigns to each object
$A$ of $\mathcal C$ an object $FA$ of $\mathcal D$ and to each morphism
$f:A\to B$ a morphism $Ff:FA\to FB$, preserving identities and
composition.
\end{definition}

\begin{definition}[Finite products in a category]
\label{def:finite_product}
Let $\mathcal{C}$ be a category and
$\{\,X_{i}\,\}_{i=1}^{n}$ a finite family of objects in  $\mathcal{C}$
($n\ge 0$).

\begin{itemize}[leftmargin=2em]
\item
A \emph{product} of the family is a pair
\[
\Bigl(X,\;\{\pi_{i}\}_{i=1}^{n}\Bigr),
\quad
\pi_{i}:X\longrightarrow X_{i}\;\text{ in }\mathcal{C},
\]
such that for every object $Y$ and every collection of morphisms
$\,\{\,f_{i}:Y\to X_{i}\,\}_{i=1}^{n}$ there exists a \emph{unique}
morphism

\[
f:Y\longrightarrow X
\qquad\text{with}\qquad
\pi_{i}\circ f = f_{i}\quad(i=1,\dots,n).
\]
\item
When it exists, the product is unique up to a unique isomorphism; we
write
\(
X \;=\; \prod_{i=1}^{n} X_{i}
\)
and call the $\pi_{i}$ the \emph{projection morphisms}.
\item
For $n=0$ the empty product is any \emph{terminal object}
$1\in\mathcal{C}$ (unique up to unique isomorphism).
\end{itemize}

\[
\begin{tikzcd}[column sep=large,row sep=large]
&[-5pt]
    Y
    \arrow{dl}[swap]{f_{1}}
    \arrow[dashed]{d}{f}
    \arrow{dr}{f_{2}}
&
\\
X_{1} & \arrow[l, "\pi_1"' near start] X_{1} \times X_{2}  \arrow[r, "\pi_2" near start] & X_{2}
\end{tikzcd}
\]

\noindent
The dashed arrow~\(f\) is the \emph{unique} morphism rendering all
triangles commutative: \(\pi_{i}\circ f = f_{i}\) for
\(i=1,\dots,n\).  This diagram visualises the universal property that
characterises the product object~\(X=\prod_{i=1}^{n} X_{i}\).

\end{definition}

\begin{definition}[Natural Transformation]\label{def:natural}
Given functors $F,G:\mathcal C\to\mathcal D$, a
\emph{natural transformation} $\eta:F\Rightarrow G$ assigns to every
object $A$ a morphism $\eta_A:FA\to GA$ such that
$Gf\circ\eta_A=\eta_B\circ Ff$ for all $f:A\to B$ in $\mathcal C$.
\end{definition}

\begin{definition}[Monad]\label{def:monad}
A \emph{monad} $(T,\eta,\mu)$ on a category $\mathcal C$ comprises an
endofunctor $T:\mathcal C\to\mathcal C$, a unit natural transformation
$\eta:\mathrm{id}_{\mathcal C}\Rightarrow T$ and a multiplication
$\mu:T^2\Rightarrow T$ satisfying $\mu\circ T\eta=\mu\circ\eta T=
\mathrm{id}_T$ and $\mu\circ T\mu=\mu\circ\mu T$.
\end{definition}

\begin{definition}[Category $\mathbf{Meas}$]\label{def:meas}
Objects are measurable spaces; morphisms are
measurable maps (Definition~\ref{def:measurable-map}).
\end{definition}

\begin{definition}[Markov Kernel]\label{def:markov-kernel}
Let $(X,\mathcal F)$ and $(Y,\mathcal G)$ be measurable spaces.
A \emph{Markov kernel} $\kappa:X\rightsquigarrow Y$ is a map
$\kappa:X\times\mathcal G\to[0,1]$ such that
\begin{enumerate}
  \item for each $x\in X$, $B\mapsto\kappa(B\mid x)$ is a probability
        measure on $(Y,\mathcal G)$;
  \item for each $B\in\mathcal G$, $x\mapsto\kappa(B\mid x)$ is
        $\mathcal F$‑measurable.
\end{enumerate}
\end{definition}

\begin{definition}[Category $\mathbf{Stoch}$]\label{def:stoch}
Objects are measurable spaces.  Morphisms are Markov kernels
(Definition~\ref{def:markov-kernel}).  Composition is
\emph{Chapman–Kolmogorov} integration:
\[
(\kappa\ast\lambda)(C\!\mid x)=\int_Y
\kappa(C\!\mid y)\,\lambda(dy\!\mid x).
\]
\end{definition}

\begin{definition}[Giry Monad]\label{def:giry-monad}
Define $G:\mathbf{Meas}\to\mathbf{Meas}$ by
$G(X,\mathcal F)=(PX,\mathcal{P\!F})$, where $\mathcal{P\!F}$ is the
weak $\sigma$–algebra generated by
$\mu\mapsto\mu(B)\;(B\in\mathcal F)$.
For $f:(X,\mathcal F)\to(Y,\mathcal G)$ set $Gf=f_{*}$
(Definition~\ref{def:pushforward}).
The \emph{unit} $\eta_X:X\to G X$ is $x\mapsto\delta_x$ and the
\emph{multiplication} $\mu_X:GGX\to GX$ is
$\mu_X(\Phi)(B)=\int_{PX}\nu(B)\,d\Phi(\nu)$.
With these, $(G,\eta,\mu)$ is a monad on $\mathbf{Meas}$
(Definition~\ref{def:monad}); its Kleisli category is
$\mathbf{Stoch}$ (Definition~\ref{def:stoch}).
\end{definition}

\begin{definition}[Stochastic Dynamical System]\label{def:coalgebra}
A \emph{stochastic dynamical system} is a
\emph{$G$‑coalgebra}: a measurable map $c:X\to GX$
(Definition~\ref{def:giry-monad}).  Iterating $c$ via Kleisli
composition yields a family of kernels
$c^{(n)}:X\rightsquigarrow X$; see Definition~\ref{def:transition}.
\end{definition}


% ---------- 3.  Markov Kernels & Stochastic Processes ----------
\subsection{Markov Kernels \& Stochastic Processes}

\begin{definition}[Transition Kernel Family]\label{def:transition}
A family $\{\kappa_t\}_{t\ge0}$ of kernels
$\kappa_t:X\rightsquigarrow X$ is a
\emph{transition kernel family} if
\[
\kappa_{t+s}=\kappa_t\ast\kappa_s,\qquad
\kappa_0(x,\cdot)=\delta_x.
\]
\end{definition}

\begin{definition}[Discrete‑Time Markov Chain]\label{def:dtmc}
A \emph{discrete‑time Markov chain} on state space $X$ is a pair
$(\kappa,\mu_0)$ where $\kappa:X\rightsquigarrow X$ is a kernel and
$\mu_0\in PX$ an initial distribution.  The law of $X_{n+1}$ given
$X_n=x$ is $\kappa(\,\cdot\mid x)$.
\end{definition}


% ---------- 4.  Semigroup & Generator Theory ----------
\subsection{Semigroup \& Generator Theory}

\begin{definition}[Bounded Linear Operators]\label{def:bounded-linear-operators} 
For a Banach space $B$, the set $\mathcal L(B)$ consists of all linear maps $A: B \rightarrow B$ with finite operator norm $|| A ||_{\mathcal L(B)} := \sup_{f\neq0} \frac{||Af||_B}{||f||_B}  $

Equipped with this norm,  is itself a Banach space.
    
\end{definition}

\begin{definition}[Operator Semigroup]\label{def:operator-semigroup}
Let $B$ be a Banach space.  
A family $\{P_t\}_{t\ge 0}\subset\mathcal L(B)$ is called an
\emph{operator semigroup} (on $B$) if 
\[
P_0 = \mathrm{id}_B,
\qquad
P_{t+s} = P_t\,P_s 
\quad \text{for all } s,t \ge 0 .
\]
\end{definition}

\begin{definition}[Uniformly Continuous Semigroup]\label{def:uniform-semigroup}
A family $\{P_t\}_{t\ge 0}\subset\mathcal L(B)$ is a \emph{uniformly continuous semigroup} if
\[
\lim_{t\downarrow 0}\|P_t-\mathrm{id}\|_{\mathcal L(B)} = 0,
\qquad
P_0=\mathrm{id},\qquad
\;\;\text{for all } t\ge 0
\]
Equivalently, $P_t = e^{\,tA}$ for a \emph{bounded} operator $A\in\mathcal L(B)$ called the (bounded) generator, and the map $t\mapsto P_t$ is continuous in operator norm.
\end{definition}


\begin{definition}[Strongly Continuous Semigroup]\label{def:c0} A family
$\{P_t\}_{t\ge0}\subset\mathcal L(B)$ is a
\emph{strongly continuous semigroup} ($C_0$–semigroup) if
\[
     \lim_{t\downarrow0}\|P_t f-f\|_B=0, \qquad \text{for all }f\in B,\,t\geq 0
\]
\end{definition}

\begin{definition}[Markov Semigroup]\label{def:semigroup}
A $C_0$–semigroup $\{P_t\}_{t\ge0}$ on $L^p(X,\lambda)$ is a
\emph{Markov semigroup} if each $P_t$ is positive and $P_t1=1$.
\end{definition}

\begin{definition}[Infinitesimal Generator]\label{def:generator}
For a $C_0$–semigroup $\{P_t\}_{t\ge0}$ the
\emph{generator} is the (possibly unbounded) operator
\[
Lf=\lim_{t\downarrow0}\frac{P_t f-f}{t},
\]
with domain $D(L)=\{f:\text{limit exists in }\|\cdot\|\}$.
\end{definition}

% FOOTER: the three spaces and what the projection is

\begin{scope}
  \draw[gray!40] (-0.6,-2.05)--(16.0,-2.05);
  \node[anchor=north west,align=left,font=\scriptsize,text width=16.4cm] at (-0.6,-2.2)
    {\textbf{The spaces.} The lattice $L$ (e.g.\ the pentagon: $5$ atoms $+$ loop
     relations) is an order-theoretic object, not a vector space; it indexes the
     others. A \emph{state} assigns a number to each element, so states live in
     $\mathbb R^{L}$ --- for the pentagon, pinned by the $5$ atom-values, a polytope
     $S(L)\subseteq[0,1]^{5}\subseteq\mathbb R^{5}$. The dispersion-free states
     $S_{\mathrm{df}}$ are its $\{0,1\}$-valued \emph{vertices}; $\overline{\mathrm{conv}}
     (S_{\mathrm{df}})$ is a sub-polytope, \emph{also in $\mathbb R^{5}$} --- high-dimensional
     and undrawable. \textbf{The projection.} Panels (2)--(3) draw not that polytope but
     its image under a linear functional $g:\mathbb R^{L}\!\to\mathbb R$ (a weighted sum of
     atom-values; in (2), $g=f=\sum_i s(a_i)$). Since $g$ is linear,
     $g\bigl(\overline{\mathrm{conv}}(S_{\mathrm{df}})\bigr)$ is an \emph{interval}, and
     $g(w)$ outside it forces $w$ outside the hull --- a one-way certificate.
     \textbf{Caveat.} $g$ is a \emph{chosen} separating functional for this $w$, not a
     canonical projection: by Hahn--Banach, $w$ is contextual \emph{iff some} such $g$
     sends it past the hull's image; one $g$ detects the states violating \emph{that}
     inequality, not all contextual states.};
\end{scope}
